R24, the target repository's instructions as granted context. At admission the runtime reads a fixed list from the base it snapshotted (`CLAUDE.md`, `AGENTS.md`, `.claude/rules/*.md`, and each skill's `SKILL.md` under `.claude/skills/` and `.agents/skills/`) under R20's rules: a link or a non-regular entry, a file over 32 KiB, a set over 32 KiB, or bytes that are not UTF-8 refuse admission as `instructions-refused`, naming the file. `AdmissionRecord.repoInstructions` pins each file's path, SHA-256, text, and whether its front matter was dropped; front matter is never read. `ContextGrant` gains `'repo-instructions'`, allowed at `spec`, `plan`, `build`, and `review` and never at `test-design`, and `CapabilityScope.repoInstructions` grants it per role, `false` when absent. A granted seat at L0 to L2 receives the set first in its stable prefix, without any file a withheld sensitive class matches; above L2 the set narrows the run through `narrowRun` and arrives in the variable suffix. `runTask` takes a list of released inputs.

/**
 * Reads the target repository's own instructions from the base admission
 * snapshotted (R24, D-A-RI-01), under R20's rules (D-R24-02).
 *
 * The list is fixed: `CLAUDE.md` and `AGENTS.md` at the root, each
 * `.claude/rules/*.md`, and the `SKILL.md` of each directory under
 * `.claude/skills/` and `.agents/skills/`. Nothing else is read: no skill's
 * `scripts/`, `references/`, or `assets/`, and no file a listed one points
 * at. The only input is the base's path, so nothing a workspace holds is
 * read (I3).
 *
 * Every limit is a refusal that names the file, never a skip or a truncation
 * (I5): a link anywhere on a listed path, a listed path that is not a regular
 * file, a file over the byte cap or a set over the set's cap (checked before
 * a byte is decoded), and bytes that are not UTF-8. A path that is absent
 * holds nothing (D-R24-03).
 *
 * Front matter is never read. `splitFrontMatter` drops it from what a seat
 * is given, and the file is marked, so no field in a target's file widens or
 * narrows a grant (I4, D-R24-04).
 */
import { createHash } from 'node:crypto';
import type { BigIntStats } from 'node:fs';
import { lstat, readdir } from 'node:fs/promises';
import { join } from 'node:path';
import { readListed, ROLE_FILE_MAX_BYTES } from './compile.js';

/** The most bytes one instruction file may hold: R20's cap for a role file. */
export const REPO_INSTRUCTION_FILE_MAX_BYTES = ROLE_FILE_MAX_BYTES;

/**
 * The most bytes the whole set may hold. The set shares the Claude Code
 * driver's one system argv element with a role of up to 32 KiB and the
 * granted context, and Linux refuses one past 128 KiB (D-R20-03, D-R24-03).
 */
export const REPO_INSTRUCTIONS_MAX_BYTES = 32 * 1024;

export type RepoInstructionCode = 'unreadable' | 'not-regular' | 'too-large' | 'not-utf8' | 'set-too-large';

/** One file the reader refused, by its path in the base. */
export interface RepoInstructionRefusal {
  readonly path: string;
  readonly code: RepoInstructionCode;
  readonly message: string;
}

/** One file as read: its whole text, the hash of its bytes, and whether it opened with front matter. */
export interface ReadInstructionFile {
  readonly path: string;
  readonly sha256: string;
  readonly text: string;
  readonly frontMatterDropped: boolean;
}

/** Every file in the fixed order, or every file refused. Never some of each. */
export type RepoInstructionRead =
  | { readonly ok: true; readonly files: readonly ReadInstructionFile[] }
  | { readonly ok: false; readonly refused: readonly [RepoInstructionRefusal, ...RepoInstructionRefusal[]] };

/**
 * A first line `---`, up to and including the next line `---`. A byte-order
 * mark before it is dropped with it. A file whose first line is `---` and
 * that never closes it has no front matter, and is delivered whole.
 */
const FRONT_MATTER = /^\uFEFF?---[ \t]*\r?\n(?:[\s\S]*?\r?\n)?---[ \t]*(?:\r?\n|$)/u;

/** `text` without its front matter, which is never read, and whether it had any. */
export function splitFrontMatter(text: string): { readonly body: string; readonly dropped: boolean } {
  const match = FRONT_MATTER.exec(text);
  return match === null ? { body: text, dropped: false } : { body: text.slice(match[0].length), dropped: true };
}

type Listed = { readonly kind: 'absent' } | { readonly kind: 'link' } | { readonly kind: 'other'; readonly stats: BigIntStats };

async function listed(path: string): Promise<Listed | { readonly kind: 'error'; readonly error: unknown }> {
  try {
    const stats = await lstat(path, { bigint: true });
    return stats.isSymbolicLink() ? { kind: 'link' } : { kind: 'other', stats };
  } catch (error) {
    if ((error as { code?: unknown }).code === 'ENOENT' || (error as { code?: unknown }).code === 'ENOTDIR') return { kind: 'absent' };
    return { kind: 'error', error };
  }
}

function describe(error: unknown): string {
  return error instanceof Error ? error.message : String(error);
}

/**
 * Reads the fixed list from `base`. A refused file refuses the whole set: a
 * run is never admitted with some of the target's instructions.
 */
export async function readRepoInstructions(base: string): Promise<RepoInstructionRead> {
  const files: ReadInstructionFile[] = [];
  const refusals: RepoInstructionRefusal[] = [];
  let total = 0;
  let overSet = false;

  /** A directory on a listed path: its entries, sorted, or none where it is absent or not a directory. A link is refused. */
  const directory = async (rel: string): Promise<string[]> => {
    const at = await listed(join(base, ...rel.split('/')));
    if (at.kind === 'absent') return [];
    if (at.kind === 'link') {
      refusals.push({ path: rel, code: 'not-regular', message: `'${rel}' is a link; a repository instruction path is read only through directories and files of the base itself` });
      return [];
    }
    if (at.kind === 'error') {
      refusals.push({ path: rel, code: 'unreadable', message: `'${rel}' could not be read: ${describe(at.error)}` });
      return [];
    }
    if (!at.stats.isDirectory()) return [];
    try {
      return (await readdir(join(base, ...rel.split('/')))).sort();
    } catch (error) {
      refusals.push({ path: rel, code: 'unreadable', message: `'${rel}' could not be read: ${describe(error)}` });
      return [];
    }
  };

  /** A listed file: read where present, refused where it is a link or anything but a regular file. */
  const file = async (rel: string): Promise<void> => {
    const path = join(base, ...rel.split('/'));
    const at = await listed(path);
    if (at.kind === 'absent') return;
    if (at.kind === 'error') {
      refusals.push({ path: rel, code: 'unreadable', message: `repository instruction file '${rel}' could not be read: ${describe(at.error)}` });
      return;
    }
    if (at.kind === 'link' || !at.stats.isFile()) {
      refusals.push({ path: rel, code: 'not-regular', message: `repository instruction path '${rel}' is not a regular file` });
      return;
    }
    const read = await readListed(path, at.stats, REPO_INSTRUCTION_FILE_MAX_BYTES, `repository instruction file '${rel}'`);
    if (!read.ok) {
      refusals.push({ path: rel, code: read.code, message: read.message });
      return;
    }
    total += read.bytes.byteLength;
    if (total > REPO_INSTRUCTIONS_MAX_BYTES && !overSet) {
      overSet = true;
      refusals.push({ path: rel, code: 'set-too-large', message: `with '${rel}' the repository instructions are ${String(total)} bytes or more; the limit for the set is ${String(REPO_INSTRUCTIONS_MAX_BYTES)}` });
      return;
    }
    files.push({ path: rel, sha256: createHash('sha256').update(read.bytes).digest('hex'), text: read.text, frontMatterDropped: splitFrontMatter(read.text).dropped });
  };

  await file('CLAUDE.md');
  await file('AGENTS.md');
  const claude = await directory('.claude');
  if (claude.includes('rules')) {
    for (const name of await directory('.claude/rules')) if (name.endsWith('.md')) await file(`.claude/rules/${name}`);
  }
  const roots: string[] = [];
  if (claude.includes('skills')) roots.push('.claude/skills');
  if ((await directory('.agents')).includes('skills')) roots.push('.agents/skills');
  for (const root of roots) {
    for (const name of await directory(root)) {
      const rel = `${root}/${name}`;
      const at = await listed(join(base, ...rel.split('/')));
      // A link in a skills directory is refused wherever it points; a file there is no skill, and holds nothing.
      if (at.kind === 'link') refusals.push({ path: rel, code: 'not-regular', message: `'${rel}' is a link; a skill is a directory of the base itself` });
      else if (at.kind === 'error') refusals.push({ path: rel, code: 'unreadable', message: `'${rel}' could not be read: ${describe(at.error)}` });
      else if (at.kind === 'other' && at.stats.isDirectory()) await file(`${rel}/SKILL.md`);
    }
  }
  const [first, ...rest] = refusals;
  if (first !== undefined) return { ok: false, refused: [first, ...rest] };
  return { ok: true, files };
}

const ORDER = ['CLAUDE.md', 'AGENTS.md', '.claude/rules/style.md', '.claude/skills/greet/SKILL.md', '.agents/skills/tidy/SKILL.md'];
const MARKERS = ['r24-claude-marker', 'r24-agents-marker', 'r24-rule-marker', 'r24-skill-marker', 'r24-agents-skill-marker'];
const NEVER = ['r24-script-marker', 'r24-reference-marker', 'r24-outside-marker', 'allowed-tools', 'paths:'];

async function plant(root: string, files: Readonly<Record<string, string>>): Promise<void> {
  for (const [path, text] of Object.entries(files)) {
    await mkdir(dirname(join(root, path)), { recursive: true });
    await writeFile(join(root, path), text);
  }
}

function sha256(text: string | Uint8Array): string {
  return createHash('sha256').update(text).digest('hex');
}

/** A scope with the grant set, or left as the rig grants it. */
function granted(scope: CapabilityScope, on = true): CapabilityScope {
  return { ...scope, repoInstructions: on };
}

async function grantedPolicy(builder = true, reviewer = true): Promise<Policy> {
  return linePolicy({}, {
    roles: { [BUILDER]: granted(lineScope(['build', 'verify'], 'standard'), builder), [REVIEWER]: granted(lineScope(['review'], 'deep'), reviewer) },
  });
}

/** A builder and a reviewer each recording what it is given; the rig makes the builder do the fixture's work. */
async function observed(rig: LineRig, extra: Parameters<LineRig['components']>[0] = {}): Promise<{ builder: ObservedDriver; reviewer: ObservedDriver; components: Awaited<ReturnType<LineRig['components']>> }> {
  const builder = await stubDriver();
  const reviewer = await stubDriver({ family: 'reviewer-family' });
  return { builder, reviewer, components: await rig.components({ driver: builder, reviewer, ...extra }) };
}

async function admissionOf(rig: LineRig, vault: Vault): Promise<AdmissionRecord> {
  return readRecord<AdmissionRecord>(vault, (await rig.state()).admission);
}

function requestOf(requests: readonly TaskRequest[], role: RoleId, what: string): TaskRequest {
  const found = requests.find((r) => r.role.role === role);
  if (found === undefined) throw new Error(`R24: no ${what} request was made`);
  return found;
}

/** Throws where any of `markers` is, or is not, in `text`. */
function holds(text: string, markers: readonly string[], present: boolean, what: string): void {
  for (const marker of markers) {
    if (text.includes(marker) !== present) throw new Error(`R24: ${what} ${present ? 'lacks' : 'holds'} '${marker}'`);
  }
}

/** Both prefix and suffix of every request in `requests`. */
function textOf(requests: readonly TaskRequest[]): string {
  return requests.map((r) => `${r.stablePrefix}\n${r.variableSuffix}`).join('\n');
}

/** `driver`, whose builder writes `CLAUDE.md` into its workspace and claims it, as an agent rewriting the instructions would. */
function plantingInWorkspace(driver: ObservedDriver, sandbox: SandboxProvider): ObservedDriver {
  return new Proxy(driver, {
    get(target, key, receiver) {
      if (key !== 'runTask') return Reflect.get(target, key, receiver) as unknown;
      return async (req: TaskRequest): Promise<TaskResult> => {
        const wrote = await sandbox.exec(req.sandbox, ['node', '-e', "require('node:fs').writeFileSync('CLAUDE.md', 'r24-planted-marker\\n')"]);
        if (wrote.exitCode !== 0) throw new Error(`R24: the planted copy did not land: ${wrote.stderr}`);
        const result = await target.runTask(req);
        return { ...result, claim: { ...result.claim, filesChanged: [...new Set([...result.claim.filesChanged, 'CLAUDE.md'])] } };
      };
    },
  });
}

/** Refused at admission as `instructions-refused`, naming `path` with `code`, with no run state left behind. */
async function refusesNaming(rig: LineRig, path: string, code: string, what: string): Promise<void> {
  const { startRun } = await api();
  const { builder, components } = await observed(rig);
  const outcome = await startRun(await rig.request(components, { policy: await grantedPolicy() }));
  if (outcome.ok || outcome.reason !== 'instructions-refused') throw new Error(`R24: ${what} was ${outcome.ok ? 'admitted' : `refused as ${outcome.reason}`}, not as instructions-refused`);
  if (!outcome.refused.some((r) => r.path === path && r.code === code)) throw new Error(`R24: ${what} was refused as ${JSON.stringify(outcome.refused)}, not naming ${path} as ${code}`);
  if (builder.requests.length > 0) throw new Error(`R24: ${what} reached the driver before it was refused`);
  const { isNoRunState } = await import('@olympus-ai/vault');
  try {
    await components.vault.readRunState(rig.runId);
  } catch (error) {
    if (isNoRunState(error)) return;
    throw error;
  }
  throw new Error(`R24: ${what} left run state behind`);
}

    * `runs/<id>/objects/<kind>/` or `nightly/` is named by its hash; every
    * other is named by a run, a number, or an id. An entry that is neither a
    * file nor a directory is not a record this store writes, and stops the

   async *listObjects(): AsyncIterable<StoredObject> {
     const store = this.#store;
     async function* walk(dir: string, rel: readonly string[]): AsyncGenerator<StoredObject> {
       let entries: Dirent[];
       try {
         entries = await readdir(dir, { withFileTypes: true });
       } catch (error) {
         if (isAbsent(error) && rel.length === 0) return;
         throw error;

   /**
    * Compare-and-swap, arbitrated by the filesystem. The next version's file is
    * created with `wx`, so of any number of writers holding the same
    * `ifVersion` exactly one creates it and the rest are refused with `EEXIST`.
    * There is deliberately no read-then-compare here: that shape loses an
    * update whenever two writers interleave between the read and the write, and
    * it passes a sequential test while doing so.
    *
    * `ifVersion` is authoritative and `s.version` is ignored; the contract left
    * the choice to this unit (D-S1, S1 spec section 2). The precondition below
    * is an extra refusal, not the concurrency control: it rejects a version
    * that never existed, so a caller cannot jump the run forward and leave a
    * hole in its history.
    */
   async commitRunState(s: RunState, ifVersion: string): Promise<RunState> {
     if (!VERSION.test(ifVersion)) {
       throw new Error(`LocalVault: '${ifVersion}' is not a run state version`);
     }
     const states = join(runDir(this.#store, s.runId), 'state');
     const current = Number(ifVersion);
     if (!Number.isSafeInteger(current)) throw new Error(`LocalVault: run state version ${ifVersion} is out of range`);
 
     const held = await highestIn(states);
     if (current === 0 ? held !== null : held === null || held < current) {
       const at = held === null ? 'no committed state' : `version ${String(held)}`;
       throw new Error(`LocalVault: run ${s.runId} has ${at}, so version ${ifVersion} was never current; nothing stored`);
     }
 
     // The history it replaces, read before anything is written: a commit may extend it and never shorten or rewrite it (codex-5).
     const previous = current === 0 ? [] : requireRunState(await readJson(join(states, `${ifVersion}.json`)), `${ifVersion}.json`).stations;
     const problem = stationsProblem(previous, (s as { stations?: unknown }).stations);
     if (problem !== undefined) throw new Error(`LocalVault: refusing run ${s.runId}'s state: ${problem}; nothing stored`);
 
     const next = String(current + 1);
     const stored: RunState = { ...s, version: next };
     if (!(await createExclusive(join(states, `${next}.json`), encode(stored)))) {
       throw new Error(
         `LocalVault: run ${s.runId} has already moved past version ${ifVersion}; another writer committed ${next} first. ` +
           'Nothing was stored. Re-read the state and commit again.',
       );
     }
     return stored;
   }

+/**
+ * The one check every reader of an admission record's repository
+ * instructions makes (A-R24-03): each file names a path in the fixed list,
+ * once, and its hash is the hash of its text, so the record holds exactly the
+ * versions it names and a resume delivers what admission read (I3, I5).

+/** A path the reader can list: the two root files, a rule, or a skill's `SKILL.md`. */
+const LISTED = /^(?:CLAUDE\.md|AGENTS\.md|\.claude\/rules\/[^/]+\.md|\.(?:claude|agents)\/skills\/[^/]+\/SKILL\.md)$/u;
+
+/** Whether `path` is one the fixed list can name. */
+export function isRepoInstructionPath(path: string): boolean {
+  return LISTED.test(path) && !path.split('/').some((part) => part === '.' || part === '..');

+
+/** Why `value` is not a repository instruction set a record can hold, or undefined. Absent is the empty set. */
+export function repoInstructionsProblem(value: unknown): string | undefined {
+  if (value === undefined) return undefined;
+  if (!Array.isArray(value)) return 'its repository instructions are not a list';
+  const seen = new Set<string>();
+  for (const file of value as unknown[]) {
+    const f = (typeof file === 'object' && file !== null ? file : {}) as Record<string, unknown>;
+    if (typeof f.path !== 'string' || typeof f.sha256 !== 'string' || typeof f.text !== 'string' || typeof f.frontMatterDropped !== 'boolean') {
+      return 'a repository instruction file carries no path, hash, text, and front-matter mark';
+    }
+    if (!isRepoInstructionPath(f.path)) return `'${f.path}' is not a path the repository instruction list names`;
+    if (seen.has(f.path)) return `repository instruction file '${f.path}' is listed twice`;
+    seen.add(f.path);
+    if (createHash('sha256').update(f.text, 'utf8').digest('hex') !== f.sha256) {
+      return `repository instruction file '${f.path}': its hash is not the hash of its text`;
+    }
+  }
+  return undefined;
