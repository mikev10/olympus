R37: a failed gate tells the retry, and the run, why. A build task's next iteration after a failed gate carries each failed check's id and cause in its variable suffix (`gateRetrySuffix` in `core`), never the check's output. `LastFailure` is now a union discriminated by `kind` (`attempt-failed` or `gate-failed`), read for any park that names a task; a gate's check that did not start carries its evidence's reason as `untrustedReason`. The event stream publishes `gate-failed`, and `DriveHooks` gains `onGateFailed`.

       write(event);

   /** A person's dismissal: once, of a candidate neither admitted nor expired, with a cause and no text (A-R16-05). */
   const dismissIntake = async (id: string, body: DismissBody): Promise<CandidateSummary> => {
     const entry = /^[0-9a-f]{64}$/.test(id) ? await components.vault.readCandidate(id as CandidateId) : null;
     if (entry?.candidate == null) throw new HttpError(404, 'not-found', `no intake candidate ${id}`);
     if (entry.expiredAt !== null) throw new HttpError(409, 'conflict', `candidate ${id} expired at ${entry.expiredAt}; an expired candidate is already recorded as not worked`);
     if (!(await components.vault.dismissCandidate(entry.candidate.id, body.cause, new Date().toISOString()))) {
       throw new HttpError(409, 'conflict', `candidate ${id} was admitted or dismissed before`);
     }
     const after = await components.vault.readCandidate(entry.candidate.id);
     if (after?.candidate == null) throw new Error(`server: candidate ${id} is gone after its dismissal`);
     return summary({ ...after, candidate: after.candidate });
   };
 
   /**
    * `GET /readiness`: the scan's report for one revision, as a stream. A bad
    * revision is refused before the stream opens; then `started` names the
    * commit, a comment every 15 s keeps a client waiting through a scan that
    * outlasts its header timeout, and one `report` or `failed` event ends it.
    * Read-only: nothing is written, and admission does not read the ceiling.
    */
   const readiness = async (revision: string, res: ServerResponse): Promise<void> => {
     const scanner = options.readiness ?? null;
     if (scanner === null) throw new HttpError(409, 'refused', 'this server scans no repository; readiness is served by factory-host over its FACTORY_REPOSITORY');
     let commit: string;
     try {
       commit = await scanner.resolve(revision);
     } catch (error) {
       throw new HttpError(400, 'bad-request', error instanceof Error ? error.message : String(error));
     }
     res.writeHead(200, { 'content-type': 'text/event-stream; charset=utf-8', 'cache-control': 'no-store', connection: 'keep-alive' });
     open.add(res);
     res.write(`event: started\ndata: ${JSON.stringify({ revision, commit })}\n\n`);
     const beat = setInterval(() => { res.write(': scanning\n\n'); }, 15_000);
     try {
       const report = await scanner.scan(commit);
       res.write(`event: report\ndata: ${JSON.stringify(report)}\n\n`);
     } catch (error) {
       res.write(`event: failed\ndata: ${JSON.stringify({ message: error instanceof Error ? error.message : String(error) })}\n\n`);
     } finally {
       clearInterval(beat);
       open.delete(res);
       res.end();
     }
   };

     const url = new URL(req.url ?? '/', 'http://server.invalid');
     const parts = url.pathname.split('/').filter((p) => p !== '');

       return;

     }

       send(res, 200, await components.vault.readNightly());
       return;
     }

       return;
     }
     if (parts[0] === 'readiness' && parts.length === 1) {
       if (method !== 'GET') throw new HttpError(405, 'method-not-allowed', `${method} ${url.pathname}`);
       await readiness(url.searchParams.get('revision') ?? 'HEAD', res);
       return;
     }
     if (parts[0] !== 'runs' || parts.length > 3) throw new HttpError(404, 'not-found', `no route ${url.pathname}`);
     if (parts.length === 1) {
       if (method !== 'POST') throw new HttpError(405, 'method-not-allowed', `${method} ${url.pathname}`);
       send(res, 201, await create(createBody(await readJson(req))));
       return;
     }
     const runId = decodeURIComponent(parts[1] ?? '') as RunId;
     const action = parts[2];
     if (action === undefined) {
       if (method !== 'GET') throw new HttpError(405, 'method-not-allowed', `${method} ${url.pathname}`);
       send(res, 200, await view(runId));
       return;
     }
     if (action === 'report') {
       if (method !== 'GET') throw new HttpError(405, 'method-not-allowed', `${method} ${url.pathname}`);
       send(res, 200, await report(runId));
       return;
     }
     if (action === 'telemetry') {
       if (method !== 'GET') throw new HttpError(405, 'method-not-allowed', `${method} ${url.pathname}`);
       send(res, 200, await telemetry(runId));
       return;
     }
     if (action === 'events') {
       if (method !== 'GET') throw new HttpError(405, 'method-not-allowed', `${method} ${url.pathname}`);
       await events(runId, res);
       return;
     }
     if (action === 'approve' || action === 'cancel' || action === 'reject') {
       if (method !== 'POST') throw new HttpError(405, 'method-not-allowed', `${method} ${url.pathname}`);
       if (action === 'approve') send(res, 200, await approve(runId, approveBody(await readJson(req))));
       else if (action === 'reject') send(res, 200, await reject(runId, rejectBody(await readJson(req))));
       else send(res, 200, await cancel(runId));
       return;
     }
     throw new HttpError(404, 'not-found', `no route ${url.pathname}`);
   };
 
   const server: Server = createServer((req, res) => {
     route(req, res).catch((error: unknown) => {
       const http = error instanceof HttpError ? error : new HttpError(500, 'internal', describe(error));
       if (res.headersSent) {
         res.end();
         return;
       }
       sendError(res, http);
     });
   });
 
   return {
     listen: async (port, host) => {
       // The schedule is the service's: what a stop missed is recorded before the first request is taken (D-A-TF-08).
       scheduler ??= await startSchedules(async () => {
         const loaded = await loadPolicyFile(options.policyFile);
         return loaded.ok ? loaded.policy : null;
       }, components.vault, intake);
       return new Promise((resolveUrl, reject) => {
         server.once('error', reject);
         server.listen(port, host, () => {
           server.off('error', reject);
           const address = server.address();
           if (address === null || typeof address === 'string') {
             reject(new Error('api server: listening on something other than a TCP port'));
             return;
           }
           const shown = address.family === 'IPv6' ? `[${address.address}]` : address.address;
           resolveUrl(`http://${shown}:${String(address.port)}`);
         });
       });
     },
     close: async () => {
       scheduler?.stop();
       scheduler = null;
       for (const res of open) res.end();
       open.clear();
       await new Promise<void>((done) => {
         server.close(() => { done(); });
         server.closeAllConnections();
       });
       await Promise.all([...runs.values()].flatMap((entry) => (entry.drive === null ? [] : [entry.drive])));
     },
   };
 }

export const BUILD_RETRY_TOLD_FAILED_CHECKS: LocalAssertion = runtime({
  id: 'I7.build-retry-told-failed-checks-not-output',
  title:
    "a builder whose first change fails hello-written receives, in its second request's variable suffix, that check's id and the cause the gate derived, exit-code, and no byte of what the check printed, in the suffix, the stable prefix, or the role (#159)",
  run: async () => {
    await withLine('r37-retry-', async (rig) => {
      await writeManifest(rig.dirs, [LOUD_REPRODUCTION]);
      const { StubSandboxProvider } = await import('@olympus-ai/sandbox');
      const sandbox = new StubSandboxProvider();
      let builds = 0;
      // The first change writes the wrong content; every later one the right content.
      const driver = await stubDriver({
        during: async (req: TaskRequest) => {
          if (req.role.role !== ('builder' as RoleId)) return;
          builds += 1;
          const content = builds === 1 ? 'goodbye' : 'hello';
          const wrote = await sandbox.exec(req.sandbox, ['node', '-e', `require('node:fs').writeFileSync(['hello', 'txt'].join('.'), ${JSON.stringify(content)})`]);
          if (wrote.exitCode !== 0) throw new Error(`R37 control: the builder's change did not land: ${wrote.stderr}`);
        },
      });
      const components = await rig.components({ driver, sandbox }, { work: false });
      const { startRun } = await api();

  id: 'I5.unstarted-check-is-the-last-failure',
  title:
    "a run whose required check cannot start at verify parks with that check, its cause no-result, and the reason its evidence recorded as its last failure in status, and the line tells events of each failed gate in the same shape (#158)",
  run: async () => {

      const inner = new StubSandboxProvider();
      // The check starts over the base, so test-design's exit holds; once a build has run, it cannot be started.
      let armed = false;
      const command = JSON.stringify(HELLO_REPRODUCTION.command);
      const sandbox: SandboxProvider = new Proxy(inner, {
        get(target, key, receiver) {
          if (key !== 'exec') return Reflect.get(target, key, receiver) as unknown;
          return async (...args: Parameters<SandboxProvider['exec']>) => {
            if (armed && JSON.stringify(args[1]) === command) throw new Error(UNSTARTED_MARKER);
            return target.exec(...args);
          };

      const driver = await stubDriver({ during: (req: TaskRequest) => { if (req.role.role === ('builder' as RoleId)) armed = true; return Promise.resolve(); } });

      const told: unknown[] = [];

      if (told.length !== STATION_CONTRACTS.build.maxIterations || told.some((t) => JSON.stringify(t) !== shape)) {
        throw new Error(`R37: events were told of ${String(told.length)} failed gates as ${JSON.stringify(told)}`);
      }
    });
  },
});
