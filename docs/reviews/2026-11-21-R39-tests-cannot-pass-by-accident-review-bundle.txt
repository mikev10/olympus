===== packages/conformance/src/registry/all-text.ts =====
import { readdir, readFile } from 'node:fs/promises';
import { join } from 'node:path';

/** Reads one file as text; a parameter only so a test can hand it one that fails. */
export type TextReader = (path: string) => Promise<string>;

const readText: TextReader = (path) => readFile(path, 'utf8');

/**
 * Every file under `dir`, read as text. A file that cannot be read rejects
 * the call (I5, #93): a negative check over this text, such as "the Vault does
 * not hold the token", would otherwise pass on a file it never read.
 */
export async function allText(dir: string, read: TextReader = readText): Promise<string> {
  let out = '';
  for (const entry of await readdir(dir, { recursive: true, withFileTypes: true })) {
    if (entry.isFile()) out += await read(join(entry.parentPath, entry.name));
  }
  return out;
}

===== packages/conformance/src/registry/vault-surface.ts =====
/**
 * What a constructed Vault exposes at run time beyond its named operations
 * (I1, #65). Read from the instance and its whole prototype chain up to
 * `Object.prototype`, by `Reflect.ownKeys`, so an instance field, a
 * symbol-keyed method, an inherited helper, and an accessor are all seen: a
 * read of the class prototype's own string names misses every one of them.
 *
 * Returns one line per problem; an empty list is the only pass.
 */
export function vaultSurfaceProblems(instance: object, named: readonly string[]): string[] {
  const problems: string[] = [];
  const found = new Set<string>();
  for (const key of Reflect.ownKeys(instance)) {
    problems.push(`an instance property ${describeKey(key)}, which no prototype check sees`);
  }
  for (let proto: object | null = Object.getPrototypeOf(instance) as object | null; proto !== null && proto !== Object.prototype; proto = Object.getPrototypeOf(proto) as object | null) {
    for (const key of Reflect.ownKeys(proto)) {
      if (key === 'constructor') continue;
      if (typeof key === 'symbol' || !named.includes(key)) {
        problems.push(`${describeKey(key)}, which is not a named operation`);
        continue;
      }
      const descriptor = Reflect.getOwnPropertyDescriptor(proto, key);
      if (descriptor === undefined || !('value' in descriptor) || typeof descriptor.value !== 'function') {
        problems.push(`${describeKey(key)}, which is not a plain method: an accessor or a value under a named operation's name runs code the contract does not declare`);
        continue;
      }
      found.add(key);
    }
  }
  for (const name of named) {
    if (!found.has(name)) problems.push(`no plain method for the named operation '${name}'`);
  }
  return problems;
}

function describeKey(key: string | symbol): string {
  return typeof key === 'symbol' ? `the symbol-keyed ${key.toString()}` : `'${key}'`;
}

/**
 * Docker labels a caller asks the local provider to put on every container
 * and network it creates (#19, D-R39-05). A label is how what a provider made
 * is found after the process that made it is gone: a run killed before its
 * teardown leaves objects nothing else can tell from a neighbour's.
 *
 * Checked once, at `create`, before a daemon is asked to make anything. A key
 * outside Docker's lower-case reverse-DNS form, one under a prefix Docker
 * reserves, or a value with a control character is refused (I5), never
 * dropped or rewritten.
 */
import { refuse } from './refusal.js';

const KEY = /^[a-z0-9](?:[a-z0-9-]*[a-z0-9])?(?:\.[a-z0-9](?:[a-z0-9-]*[a-z0-9])?)*$/u;
const RESERVED = ['com.docker.', 'io.docker.', 'org.dockerproject.'];
// eslint-disable-next-line no-control-regex -- the control characters are what is refused
const CONTROL = /[\u0000-\u001f\u007f]/u;

/** The `--label key=value` arguments for `labels`, in key order; empty when there are none. */
export function labelArgs(labels: Readonly<Record<string, string>> | undefined): readonly string[] {
  if (labels === undefined) return [];
  const args: string[] = [];
  for (const key of Object.keys(labels).sort()) {
    const value = labels[key];
    if (!KEY.test(key)) refuse('labels', `the label key ${JSON.stringify(key)} is not lower-case letters, digits, dots, and interior hyphens`);
    if (RESERVED.some((prefix) => key.startsWith(prefix))) refuse('labels', `the label key ${JSON.stringify(key)} is under a prefix Docker reserves`);
    if (typeof value !== 'string' || CONTROL.test(value)) refuse('labels', `the label ${key} has a value that is not text without control characters`);
    args.push('--label', `${key}=${value}`);
  }
  return Object.freeze(args);
}
