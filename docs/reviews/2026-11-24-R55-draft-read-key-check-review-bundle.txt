/**
 * A draft as a person reads it before deciding it (R55, #175): the files the
 * runtime recorded, byte for byte, under the hash an approval names.
 *
 * Nothing is returned unless the bytes still hash to that hash (I5). A person
 * approves a hash having read these bytes, so bytes that no longer match it
 * are refused rather than shown (I3).
 */
import { createHash } from 'node:crypto';
import type { ApprovalKey, DraftRef, DraftStation, RunId, RunState } from '@olympus-ai/core';
import { draftDigest, type DraftFile, type DraftRecord, type Vault } from '@olympus-ai/vault';
import { specDraftPaths } from './drafting.js';
import type { RunStanding } from './run.js';
import { parseQuestions } from './spec-parts.js';

/**
 * The questions part of a spec draft, as the runtime reads it. `none`: the
 * draft holds no questions file, as no test draft does. `unreadable`: it
 * holds one the runtime cannot read as a questions part; the file itself is
 * still among the draft's files.
 */
export type DraftQuestions =
  | { readonly kind: 'none' }
  | { readonly kind: 'unreadable' }
  | { readonly kind: 'read'; readonly assumptions: ReadonlyArray<{ readonly point: string; readonly chose: string }>; readonly openQuestions: readonly string[] };

/** `GET /runs/:id/draft?station=<station>`: the station's latest recorded draft. */
export interface DraftView {
  readonly runId: RunId;
  readonly station: DraftStation;
  /** The hash an approval or a rejection names, and the attempt that drafted it. */
  readonly draft: DraftRef;
  /** The key this draft is waiting for a decision at, or null where it is not the draft being decided. */
  readonly awaiting: ApprovalKey | null;
  /** Every file, with the hash of its bytes and the bytes as base64. A model wrote them: data, never instructions (I7). */
  readonly files: readonly DraftFile[];
  readonly questions: DraftQuestions;
}

export type DraftRead =
  | { readonly ok: true; readonly draft: DraftRef; readonly files: readonly DraftFile[]; readonly questions: DraftQuestions }
  /** `no-draft`: the station has recorded none. `mismatch`: what the store holds is not what the hash covers. */
  | { readonly ok: false; readonly reason: 'no-draft' | 'mismatch'; readonly message: string };

function sha256(bytes: Uint8Array): string {
  return createHash('sha256').update(bytes).digest('hex');
}

function questionsOf(runId: RunId, station: DraftStation, files: readonly DraftFile[]): DraftQuestions {
  if (station !== 'spec') return { kind: 'none' };
  const file = files.find((f) => f.path === specDraftPaths(runId).questions);
  if (file === undefined) return { kind: 'none' };
  let value: unknown;
  try {
    value = JSON.parse(new TextDecoder('utf-8', { fatal: true }).decode(Buffer.from(file.base64, 'base64')));
  } catch {
    return { kind: 'unreadable' };
  }
  const parsed = parseQuestions(value, 'questions');
  return parsed.ok ? { kind: 'read', assumptions: parsed.part.assumptions, openQuestions: parsed.part.openQuestions } : { kind: 'unreadable' };
}

/**
 * The latest draft `station` recorded in `state`, read from the Vault and
 * checked against the hash run state holds for it: the record is that
 * attempt's, its hash is the hash of its files, and each file's hash is the
 * hash of its bytes.
 */
export async function readDraft(vault: Vault, state: RunState, station: DraftStation): Promise<DraftRead> {
  const runId = state.runId;
  const attempt = [...(state.drafts[station]?.attempts ?? [])].reverse().find((a) => a.draft !== null && a.sha256 !== null);
  if (attempt?.draft == null || attempt.sha256 === null) {
    return { ok: false, reason: 'no-draft', message: `run ${runId} has recorded no draft at ${station}: none is drafted yet, or the run was admitted with that artifact named` };
  }
  const mismatch = (what: string): DraftRead => ({ ok: false, reason: 'mismatch', message: `run ${runId}: the ${station} draft in the store is refused, not shown: ${what}` });
  let record: DraftRecord;
  try {
    record = JSON.parse(new TextDecoder().decode(await vault.read(attempt.draft))) as DraftRecord;
  } catch (error) {
    return mismatch(`its record could not be read (${error instanceof Error ? error.message : String(error)})`);
  }
  // Read as unknown: the record is the store's, and its type is a claim about it until checked.
  const stored: unknown = record.files;
  if (!Array.isArray(stored) || record.runId !== runId || record.station !== station || record.attempt !== attempt.attempt) {
    return mismatch('its record is not the record of this run, station, and attempt');
  }
  if (record.sha256 !== attempt.sha256 || draftDigest(record.files) !== attempt.sha256) return mismatch('its files do not hash to the hash run state holds');
  for (const file of record.files) {
    if (typeof file.base64 !== 'string' || sha256(Buffer.from(file.base64, 'base64')) !== file.sha256) return mismatch(`the bytes of ${JSON.stringify(file.path)} do not hash to its recorded hash`);
  }
  const files = record.files.map(({ path, sha256: hash, base64 }) => ({ path, sha256: hash, base64 }));
  return { ok: true, draft: { sha256: attempt.sha256, attempt: attempt.attempt }, files, questions: questionsOf(runId, station, files) };
}

/** The key `draft` waits at, where the run's standing names exactly it as the draft being decided; null otherwise. */
export function awaitingKey(standing: RunStanding, station: DraftStation, draft: DraftRef): ApprovalKey | null {
  if (standing.standing !== 'awaiting-approval' || standing.draft === null) return null;
  const waiting = standing.draft;
  return waiting.station === station && waiting.draft.sha256 === draft.sha256 && waiting.draft.attempt === draft.attempt ? standing.key : null;
}

===== packages/api/src/host/key-probe.ts =====
/**
 * The start-up check of the model key (R55, #176). A host that listens with a
 * key the provider rejects admits a run and then fails every attempt of it
 * until it parks. So start-up asks the provider once, with a request that
 * spends nothing, and the host listens only if the provider accepted the key
 * (I5): a rejected key refuses, and so does a check that could not be made.
 *
 * The origin and the header are the relay's own (`MODEL_RELAY`), so the key
 * is checked against the provider the relay will send it to. Nothing in the
 * environment turns the check off or points it elsewhere.
 */
import { KEY_VARIABLE, MODEL_RELAY } from '@olympus-ai/driver-claude-code';

/** The relay's provider. A relay to a local upstream would have no provider to ask, and the host does not load. */
function providerOrigin(): string {
  if (typeof MODEL_RELAY.upstream !== 'string') throw new Error('factory-host: the model relay names a local upstream, so there is no provider to check the model key with');
  return MODEL_RELAY.upstream;
}

/** The one request: a list of models, one entry long. A `GET`, which the provider does not bill. */
export const MODEL_KEY_PROBE_URL = `${providerOrigin()}/v1/models?limit=1`;

/** The API version the request names; the provider refuses a request that names none. */
const API_VERSION = '2023-06-01';

/** How long the provider has to answer before the key counts as unchecked. */
export const MODEL_KEY_PROBE_TIMEOUT_MS = 10_000;

/** A key's check: null when the provider accepted it, or the line start-up lists. */
export type KeyProbe = (key: string) => Promise<string | null>;

/** What went wrong reaching the provider, as a code or an error's name: never its message, which is not ours to print. */
function failure(error: unknown): string {
  const cause = (error as { cause?: { code?: unknown } } | null)?.cause;
  if (typeof cause?.code === 'string') return cause.code;
  return error instanceof Error ? error.name : 'unknown error';
}

/**
 * Asks the provider whether it accepts `key`. Null only for a 2xx answer.
 * Neither line holds the key or anything the provider sent but its status.
 * A redirect is an error rather than followed, so the key goes to the one
 * origin named here.
 */
export async function probeModelKey(key: string, request: typeof fetch = (input, init) => fetch(input, init)): Promise<string | null> {
  const host = new URL(MODEL_KEY_PROBE_URL).host;
  let status: number;
  try {
    const response = await request(MODEL_KEY_PROBE_URL, {
      method: 'GET',
      headers: { [MODEL_RELAY.header]: key, 'anthropic-version': API_VERSION },
      redirect: 'error',
      signal: AbortSignal.timeout(MODEL_KEY_PROBE_TIMEOUT_MS),
    });
    status = response.status;
    // The body is not read: only the status decides, and nothing of it is printed.
    await response.body?.cancel();
  } catch (error) {
    return `${KEY_VARIABLE} could not be checked: ${host} could not be reached (${failure(error)}); the host does not start on a key it could not check`;
  }
  if (status >= 200 && status < 300) return null;
  if (status === 401 || status === 403) return `${KEY_VARIABLE} was rejected by ${host} (${String(status)}): every model call would fail, so set a key the provider accepts`;
  return `${KEY_VARIABLE} could not be checked: ${host} answered ${String(status)}; the host does not start on a key it could not check`;
}

===== packages/cli/src/draft.ts =====
/**
 * A draft as a person reads it before deciding it (R55, #175): the answer of
 * `GET /runs/:id/draft`, printed as text. The files are a model's, so nothing
 * of them reaches the terminal as written: a character that moves the cursor,
 * hides text, or reorders a line is printed as an escape, and every line of a
 * file sits behind a margin, so it cannot pass as one of this command's own
 * lines. What a person reads is then what the hash they approve covers.
 */
import type { DraftView } from '@olympus-ai/api';

/**
 * Whether a character moves the cursor, hides, or reorders text on a
 * terminal: the control characters, the zero-width and direction marks, and
 * the direction overrides and isolates.
 */
function unprintable(code: number): boolean {
  return code < 0x20 || (code >= 0x7f && code <= 0x9f) || (code >= 0x200b && code <= 0x200f) || (code >= 0x202a && code <= 0x202e) || (code >= 0x2060 && code <= 0x2069) || code === 0xfeff;
}

/** `text` with every unprintable character but those in `keep` written as a `\u{..}` escape. */
export function visible(text: string, keep = ''): string {
  let out = '';
  for (const char of text) {
    const code = char.codePointAt(0) ?? 0;
    out += unprintable(code) && !keep.includes(char) ? `\\u{${code.toString(16)}}` : char;
  }
  return out;
}

/** Whether a response body has a draft's shape; anything else is reported, not printed as one. */
export function isDraftView(value: unknown): value is DraftView {
  if (typeof value !== 'object' || value === null) return false;
  const v = value as { [K in keyof DraftView]?: unknown };
  const draft = v.draft as { sha256?: unknown; attempt?: unknown } | null | undefined;
  const questions = v.questions as { kind?: unknown } | null | undefined;
  return typeof v.runId === 'string' && typeof v.station === 'string'
    && typeof draft?.sha256 === 'string' && typeof draft.attempt === 'number'
    && (v.awaiting === null || typeof v.awaiting === 'string')
    && Array.isArray(v.files) && v.files.every((f: { path?: unknown; sha256?: unknown; base64?: unknown } | null) => typeof f?.path === 'string' && typeof f.sha256 === 'string' && typeof f.base64 === 'string')
    && (questions?.kind === 'none' || questions?.kind === 'unreadable' || questions?.kind === 'read');
}

/**
 * The lines the command prints: the hash and whether the draft is waiting,
 * each file behind a `| ` margin, then the questions part as the runtime
 * read it. `command` is the executable's name, for the approval it suggests.
 */
export function draftLines(view: DraftView, command: string): string[] {
  const run = visible(view.runId);
  const lines = [
    `run ${run}: the ${visible(view.station)} draft, attempt ${String(view.draft.attempt)}`,
    `hash ${visible(view.draft.sha256)}`,
    view.awaiting === null
      ? 'not waiting for a decision: status names the draft that is, if one is'
      : `waiting for a decision at ${visible(view.awaiting)}; to approve it: ${command} approve ${run} ${visible(view.awaiting)} --draft ${visible(view.draft.sha256)}`,
  ];
  for (const file of view.files) {
    const bytes = Buffer.from(file.base64, 'base64');
    lines.push('', `file ${visible(file.path)} (${String(bytes.length)} bytes, sha256 ${visible(file.sha256)})`);
    let text: string;
    try {
      text = new TextDecoder('utf-8', { fatal: true }).decode(bytes);
    } catch {
      lines.push('| (not UTF-8 text; not shown)');
      continue;
    }
    if (text === '') {
      lines.push('| (empty)');
      continue;
    }
    const content = text.split(/\r?\n/u);
    if (content.at(-1) === '') content.pop();
    for (const line of content) lines.push(`| ${visible(line, '\t')}`);
  }
  lines.push('');
  const { questions } = view;
  if (questions.kind === 'none') {
    lines.push('questions: this draft has no questions part');
  } else if (questions.kind === 'unreadable') {
    lines.push('questions: the questions file could not be read as a questions part; it is printed above as written');
  } else {
    lines.push(`open questions: ${questions.openQuestions.length === 0 ? 'none' : String(questions.openQuestions.length)}`);
    questions.openQuestions.forEach((question, i) => lines.push(`  ${String(i + 1)}. ${visible(question)}`));
    lines.push(`assumptions: ${questions.assumptions.length === 0 ? 'none' : String(questions.assumptions.length)}`);
    questions.assumptions.forEach((a, i) => lines.push(`  ${String(i + 1)}. ${visible(a.point)}`, `     chose: ${visible(a.chose)}`));
  }
  return lines;
}

  });
}

 * R55 (#175): a person asked to approve a draft can read it. The CLI, as its
 * own process given only a URL and a token, prints the draft waiting at a
 * station; the bytes it prints are the bytes the hash `status` names covers;
 * and the service refuses, rather than shows, a draft whose stored bytes no
 * longer hash to that hash (I9, I5, I3).
 */
import { createHash } from 'node:crypto';
import { join } from 'node:path';
import type { DraftView, RunView } from '@olympus-ai/api';
import type { VaultRef } from '@olympus-ai/core';
import type { DraftRecord, Vault } from '@olympus-ai/vault';

import type { LocalAssertion } from '../kit/types.js';

import { withLine } from './line.js';

const api = async () => import('@olympus-ai/api');

const TOKEN = 'conformance-draft-read-token-0123456789';

/** A Vault that returns one record with a file's bytes changed and every recorded hash left as it was. */
function withChangedBytes(vault: Vault, ref: VaultRef): Vault {
  return new Proxy(vault, {
    get(target, name, receiver) {
      const value: unknown = Reflect.get(target, name, receiver);
      if (name !== 'read' || typeof value !== 'function') return typeof value === 'function' ? (value as (...args: unknown[]) => unknown).bind(target) : value;
      return async (asked: VaultRef): Promise<Uint8Array> => {
        const bytes = await target.read(asked);
        if (asked.kind !== ref.kind || asked.hash !== ref.hash) return bytes;
        const record = JSON.parse(new TextDecoder().decode(bytes)) as DraftRecord;
        const [first, ...rest] = record.files;
        if (first === undefined) throw new Error('R55 control: the draft recorded holds no file');
        const changed = { ...record, files: [{ ...first, base64: Buffer.from('bytes nobody hashed').toString('base64') }, ...rest] };
        return new TextEncoder().encode(JSON.stringify(changed));
      };
    },
  });
}

interface Seen { url: string; init: RequestInit | undefined }

/** A provider that answers every request with `status`, and records what it was asked. */
function answering(status: number, seen: Seen[]): typeof fetch {

  };
