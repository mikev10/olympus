R15: checks run on the base before build. `CheckSpec` splits by kind: an
acceptance check must declare `onBase: 'fails' | 'passes'`, a behavioral check
may, and no other kind can. Admission refuses an undeclared acceptance check,
a declaration on another kind, and a manifest with no required declared check
declared `'fails'`. At the `test-design` exit the line runs every declared
check over a fresh read-only copy of the base with no egress, before any
driver is invoked, records the results once through the new
`Vault.recordBaseRun` (read back with `readBaseRun`, `VaultRefKind`
`'base-run'`), and refuses the exit with the new `StationRefusal` reason
`'base-mismatch'` when a check does not do what it declared, does not start,
or runs over suites that cannot be counted. A resume judges the stored record
and runs nothing again. `EvidenceBundle` gains a required `baseRun` reference.

/**
 * The rule that turns the base record into a refusal or none (D-A-BF-01).
 * The line runs the declared checks over the base at the `test-design`
 * exit and records what they did; this reads that record against what each
 * check declared, in the runtime's terms alone (I2). It is exported so the
 * rule is assertable without a run.
 */
import type { BaseMismatch, StationRefusal, VaultRef } from '@olympus-ai/core';
import type { CheckSpec, DeclaredCheckSpec } from '@olympus-ai/integrity';
import type { BaseRunRecord } from '@olympus-ai/vault';
import { SUITE_KINDS } from './gate.js';

/** The checks the base run runs: every check that declares what it does there, in the manifest's order (D-R15-02). */
export function declaredChecks(checks: readonly CheckSpec[]): DeclaredCheckSpec[] {
  return checks.filter((check): check is DeclaredCheckSpec => check.onBase !== undefined);
}

/**
 * Each declared check that did not do on the base what it declared. A check
 * with no result is `no-result`, never a failure: a test that never ran did
 * not fail (I5). One that ran and did the opposite of its declaration is
 * `contrary`. A suite check, or one that pins `expectedSuiteCount`, that did
 * what it declared over suites that could not be counted, or counted below
 * that number, is `suite-count`: the runner may never have loaded the test,
 * so its failure reproduces nothing. This is `verify`'s gate rule. A
 * behavioral check that pins no count runs the product rather than a runner,
 * so its exit code alone decides.
 */
export function baseMismatches(checks: readonly CheckSpec[], record: BaseRunRecord): BaseMismatch[] {
  const mismatched: BaseMismatch[] = [];
  for (const check of declaredChecks(checks)) {
    const result = record.checks.find((r) => r.checkId === check.id);
    if (result === undefined) {
      mismatched.push({ checkId: check.id, declared: check.onBase, exitCode: null, cause: 'no-result' });
      continue;
    }
    const failed = result.exitCode !== 0;
    if (failed !== (check.onBase === 'fails')) {
      mismatched.push({ checkId: check.id, declared: check.onBase, exitCode: result.exitCode, cause: 'contrary' });
      continue;
    }
    const counted = result.suiteCount;
    const unknown = counted === null && (SUITE_KINDS.has(check.kind) || check.expectedSuiteCount !== undefined);
    if (unknown || (counted !== null && check.expectedSuiteCount !== undefined && counted < check.expectedSuiteCount)) {
      mismatched.push({ checkId: check.id, declared: check.onBase, exitCode: result.exitCode, cause: 'suite-count' });
    }
  }
  return mismatched;
}

/** The refusal of the `test-design` exit for a base record, or undefined when every declared check did what it declared. */
export function baseRefusal(checks: readonly CheckSpec[], record: BaseRunRecord, ref: VaultRef): StationRefusal | undefined {
  const [first, ...rest] = baseMismatches(checks, record);
  if (first === undefined) return undefined;
  const mismatched: [BaseMismatch, ...BaseMismatch[]] = [first, ...rest];
  const named = mismatched.map((m) => `${m.checkId} (declared ${m.declared}, ${describe(m)})`).join(', ');
  return {
    ok: false,
    reason: 'base-mismatch',
    mismatched,
    baseRun: ref,
    message: `checks did not do on the base what they declared: ${named}`,
  };
}

function describe(m: BaseMismatch): string {
  switch (m.cause) {
    case 'contrary':
      return `exited ${String(m.exitCode)}`;
    case 'no-result':
      return 'did not start';
    case 'suite-count':
      return `exited ${String(m.exitCode)} over suites that could not be counted or counted short`;
  }
}

/**

 * the `test-design` exit, before any driver is invoked, and the exit is
 * refused unless each does there what the locked manifest declares
 * (D-A-BF-01, D-R15-01, D-R15-02).
 *
 * The hello fixture's `hello-written` check is a behavioral check declared
 * `'fails'`: the base has no `hello.txt`, and the rig's builder writes one
 * (`doingTheWork`). The assertions here make it an acceptance check, the
 * kind that must declare, over a workspace given a suite (`withSuite`).
 */

/** A provider that counts what it provisions, so an assertion can see whether a check ran again. */
async function counting(): Promise<{ provider: SandboxProvider; provisioned: () => number }> {
  const { StubSandboxProvider } = await import('@olympus-ai/sandbox');
  const inner = new StubSandboxProvider();
  let n = 0;
  const provider = new Proxy(inner, {
    get(target, key, receiver) {
      if (key === 'provision') {
        return async (...args: Parameters<SandboxProvider['provision']>) => {
          n += 1;
          return target.provision(...args);
        };
      }
      return Reflect.get(target, key, receiver) as unknown;
    },
  });
  return { provider, provisioned: () => n };
}

async function decisionsOf(vault: Vault, rig: LineRig): Promise<EnforcementDecision[]> {
  return Promise.all((await vault.readDecisions(rig.runId)).map((ref) => readRecord<EnforcementDecision>(vault, ref)));
}

/**
 * The fixture declares no test framework, so its reproduction is behavioral
 * (D-R15-02). An acceptance check is a suite check and needs suites the
 * adapters can count, so an assertion about one gives the workspace a vitest
 * manifest and one test file.
 */
async function withSuite(rig: LineRig): Promise<void> {
  await writeFile(join(rig.dirs.artifacts, 'package.json'), JSON.stringify({ name: 'hello', private: true, devDependencies: { vitest: '4.1.11' } }));
  await writeFile(join(rig.dirs.artifacts, 'hello.test.mjs'), "import { test } from 'vitest';\ntest('hello', () => {});\n");
}

/** Runs the fixture with `checks` as its manifest and returns the refusal, the driver's requests, and the components. */
async function refusedAtTestDesign(rig: LineRig, checks: readonly unknown[], context: string) {
  const { startRun } = await api();
  await withSuite(rig);
  await writeManifest(rig.dirs, checks);
  const driver = await stubDriver();
  const components = await rig.components({ driver, reviewer: driver });
  const refusal = refusalOf(await startRun(await rig.request(components)), context);
  if (refusal.at !== 'test-design' || refusal.transition.reason !== 'base-mismatch') {
    throw new Error(`${context}: the run stopped at ${refusal.at} as '${refusal.transition.reason}', not at the test-design exit as base-mismatch`);
  }
  if (driver.requests.length !== 0) throw new Error(`${context}: the driver was invoked ${String(driver.requests.length)} times before the base was judged`);
  return { refusal: refusal.transition, components };
}

async function refusedAtAdmission(rig: LineRig, checks: readonly unknown[], path: string, code: string): Promise<void> {
  const { startRun } = await api();
  await writeManifest(rig.dirs, checks, { asWritten: true });

  const outcome = await startRun(await rig.request(await rig.components({ sandbox: provider })));
  if (outcome.ok || outcome.reason !== 'invalid-request') {
    throw new Error(`I5: ${JSON.stringify(checks)} was ${outcome.ok ? 'admitted and finished' : `refused as '${outcome.reason}'`}, not refused at admission`);
  }
  if (!outcome.problems.some((p) => p.path === path && p.code === code)) {
    throw new Error(`I5: admission named ${JSON.stringify(outcome.problems.map((p) => `${p.path}:${p.code}`))}, not ${path}:${code}`);
  }
  if (provisioned() !== 0) throw new Error(`I5: a refused admission provisioned ${String(provisioned())} sandboxes`);
}
