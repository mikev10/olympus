R3: the browser behavioral adapter, and the browser it observes through.

`LocalDockerProvider.browse` starts one container per call in the sandbox's
network namespace (`--network container:<sandbox>`) and not its filesystem or
process tree, from a browser image pinned by digest and run by its content ID.
The container has the probe's limits and a memory and process ceiling: a
read-only root, every capability dropped, no new privileges, an unprivileged
user, nothing mounted. Chrome runs without its own process sandbox, which
cannot start inside those limits; the container is the boundary. The browser
is driven over a pipe, never a port, and its program arrives on standard
input.

A request is a port and a list of steps: `goto` (a path), `click`, `fill`,
`press`, and `observe`. A step names its element by role and accessible name,
by label, or by visible text, and by nothing else. Every request the page
makes to a host other than the loopback is refused inside the container and
its host reported. The browser observes and decides nothing.

`BrowserBehavioralAdapter` starts a scenario's `serve` detached, browses it,
and compares on the host. A refused host fails the scenario unless its
`mayBlock` lists it, a document that navigates off the loopback always fails
it, and a step that could not be done fails it and leaves every later
observation unmade. One browser scenario per handle. Stack detection carries
the adapter when the provider has a `browse` and names `behavioral:browser`,
with the provider as the reason, otherwise.

Contract amendments: `SandboxProvider.browse` (optional; A-R3-01) and the
types it speaks, `BrowseTarget`, `BrowseStep`, `BrowseRequest`,
`BrowseObservation`, `BrowseBlockedHost`, and `BrowseResult` (A-R3-02).
`StubSandboxProvider` has no `browse`.

No real run builds a behavioral adapter of any kind yet. Wiring scenarios into
the line is R76.

===== packages/adapters/src/browser.ts =====
/**
 * A browser scenario, served inside the sandbox, driven by a browser the
 * product cannot reach, and judged on the host (R3).
 *
 * The scenario owns its server, as an HTTP scenario does (D-P11-03): `serve`
 * is started detached in the sandbox, and the provider's browser, a container
 * in the sandbox's network namespace and out of its filesystem, carries out
 * the steps and reports what was on the page. The comparison against
 * `expected` happens here, in the runtime's own process. A browser inside the
 * product's container would be one the product could replace, and what it
 * saw would be the product's report of itself (I2).
 *
 * A step names its element by what a person sees: a role and accessible
 * name, a label, or visible text. Nothing here reads a selector, a script, or
 * a URL, so a scenario locked before `build` does not depend on markup the
 * agent chooses afterwards (I3).
 *
 * One scenario per handle, per adapter, for the reason the HTTP adapter
 * gives: the first scenario's server holds the sandbox until it ends.
 *
 * `expected` comes from locked acceptance criteria and never from the
 * implementation. This adapter cannot check that; it takes the value it is
 * handed and refuses one it cannot read.
 */
import type { CheckResult, ExpectationMismatch, ExpectationOutcome } from '@olympus-ai/integrity';
import type { BrowseObservation, BrowseRequest, BrowseResult, BrowseStep, BrowseTarget, SandboxHandle, SandboxProvider } from '@olympus-ai/sandbox';
import { DEFAULT_READY_WITHIN_MS } from './http.js';
import { refuse } from './refusal.js';
import type { BehavioralAdapter, BehavioralScenario } from './types.js';

export interface BrowserInput {
  readonly serve: readonly [string, ...string[]];
  readonly port: number;
  readonly readyWithinMs: number;
  readonly steps: readonly [BrowseStep, ...BrowseStep[]];
  /** The hosts the locked scenario expects the page to ask for and do without, lower-cased. A refused host not listed here fails the scenario (D-A-BR-05). */
  readonly mayBlock: readonly string[];
}

/**
 * What one `observe` step must show. Only the fields named are compared, and
 * at least one is named. Unless `matched` is named, the target must have
 * matched exactly one element: an expectation is never held against one
 * element picked from several, or against none (D-R3-08).
 */
export interface BrowserExpectedObservation {
  readonly matched?: number;
  readonly visible?: boolean;
  readonly text?: string;
  /** `null` holds the element to having no value at all, as anything but a form field does. */
  readonly value?: string | null;
  readonly path?: string;
  readonly title?: string;
}

export interface BrowserExpected {
  readonly observations: readonly BrowserExpectedObservation[];
}

const INPUT_KEYS: ReadonlySet<string> = new Set(['serve', 'port', 'readyWithinMs', 'steps', 'mayBlock']);
const EXPECTED_KEYS: ReadonlySet<string> = new Set(['observations']);
const OBSERVATION_KEYS: ReadonlySet<string> = new Set(['matched', 'visible', 'text', 'value', 'path', 'title']);
const STEP_KEYS: Readonly<Record<BrowseStep['do'], ReadonlySet<string>>> = {
  goto: new Set(['do', 'path']),
  click: new Set(['do', 'target']),
  fill: new Set(['do', 'target', 'value']),
  press: new Set(['do', 'key']),
  observe: new Set(['do', 'target']),
};
/** How much of an observed text a mismatch quotes; the whole observation is in the result beside it. */
const QUOTE_LIMIT = 1024;

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function unknownKeys(value: Record<string, unknown>, allowed: ReadonlySet<string>, label: string): void {
  const extra = Object.keys(value).filter((key) => !allowed.has(key));
  if (extra.length > 0) {
    refuse('unrecognised-scenario', `${label} has ${extra.map((k) => `${label}.${k}`).join(', ')}, which this adapter does not know; it is refused rather than ignored`);
  }
}

/** An array's entries by index, holes included as undefined, so none is an entry nothing checked. */
function dense(value: readonly unknown[]): unknown[] {
  return Array.from(value);
}

function strings(value: unknown, label: string): string[] {
  const items = Array.isArray(value) ? dense(value) : [];
  if (!Array.isArray(value) || !items.every((item): item is string => typeof item === 'string' && item.trim() !== '')) {
    refuse('unrecognised-scenario', `${label} must be an array of non-empty strings`);
  }
  return items;
}

function integer(value: unknown, label: string, min: number, max: number): number {
  if (typeof value !== 'number' || !Number.isInteger(value) || value < min || value > max) {
    refuse('unrecognised-scenario', `${label} must be an integer from ${String(min)} to ${String(max)}`);
  }
  return value;
}

function text(value: unknown, label: string): string {
  if (typeof value !== 'string' || value.trim() === '') refuse('unrecognised-scenario', `${label} must be a non-empty string`);
  return value;
}

function readTarget(value: unknown, at: string): BrowseTarget {
  const forms = 'a role and name, a label, or a text';
  if (!isRecord(value)) refuse('unrecognised-scenario', `${at} must name its element by ${forms}`);
  const keys = Object.keys(value).sort().join(',');
  if (keys === 'name,role') {
    if (typeof value.name !== 'string') refuse('unrecognised-scenario', `${at}.name must be a string`);
    return { role: text(value.role, `${at}.role`), name: value.name };
  }
  if (keys === 'label') return { label: text(value.label, `${at}.label`) };
  if (keys === 'text') return { text: text(value.text, `${at}.text`) };
  // A selector, a script, or a URL is a key none of the three forms has.
  refuse('unrecognised-scenario', `${at} must name its element by exactly one of ${forms}; it has ${Object.keys(value).join(', ') || 'no key'}`);
}

function readStep(value: unknown, i: number): BrowseStep {
  const at = `input.steps[${String(i)}]`;
  if (!isRecord(value)) refuse('unrecognised-scenario', `${at} must be an object`);
  const kind = value.do;
  if (kind !== 'goto' && kind !== 'click' && kind !== 'fill' && kind !== 'press' && kind !== 'observe') {
    refuse('unrecognised-scenario', `${at}.do must be one of goto, click, fill, press, observe`);
  }
  unknownKeys(value, STEP_KEYS[kind], at);
  if (kind === 'goto') {
    if (typeof value.path !== 'string' || !value.path.startsWith('/') || /\s/u.test(value.path)) {
      refuse('unrecognised-scenario', `${at}.path must be a path beginning with / and holding no whitespace; the host is always the sandbox's loopback`);
    }
    return { do: 'goto', path: value.path };
  }
  if (kind === 'press') return { do: 'press', key: text(value.key, `${at}.key`) };
  const target = readTarget(value.target, `${at}.target`);
  if (kind === 'fill') {
    if (typeof value.value !== 'string') refuse('unrecognised-scenario', `${at}.value must be a string`);
    return { do: 'fill', target, value: value.value };
  }
  return { do: kind, target };
}

function readObservation(value: unknown, i: number): BrowserExpectedObservation {
  const at = `expected.observations[${String(i)}]`;
  if (!isRecord(value)) refuse('unrecognised-scenario', `${at} must be an object`);
  unknownKeys(value, OBSERVATION_KEYS, at);
  // An entry that names nothing would hold the page to nothing.
  if (Object.keys(value).length === 0) refuse('unrecognised-scenario', `${at} names no field, so it would hold the page to nothing`);
  if (value.visible !== undefined && typeof value.visible !== 'boolean') refuse('unrecognised-scenario', `${at}.visible must be true or false`);
  for (const field of ['text', 'path', 'title'] as const) {
    if (value[field] !== undefined && typeof value[field] !== 'string') refuse('unrecognised-scenario', `${at}.${field} must be a string`);
  }
  if ('value' in value && value.value !== null && typeof value.value !== 'string') refuse('unrecognised-scenario', `${at}.value must be a string or null`);
  // visible, text, and value are read from one element, so an entry that expects none or several cannot also name them.
  if (value.matched !== undefined && value.matched !== 1 && (value.visible !== undefined || value.text !== undefined || 'value' in value)) {
    refuse('unrecognised-scenario', `${at}.matched is not 1, so ${at} cannot also name visible, text, or value: there is no one element to read them from`);
  }
  return {
    ...(value.matched === undefined ? {} : { matched: integer(value.matched, `${at}.matched`, 0, 1_000_000) }),
    ...(typeof value.visible === 'boolean' ? { visible: value.visible } : {}),
    ...(typeof value.text === 'string' ? { text: value.text } : {}),
    ...('value' in value ? { value: value.value as string | null } : {}),
    ...(typeof value.path === 'string' ? { path: value.path } : {}),
    ...(typeof value.title === 'string' ? { title: value.title } : {}),
  };
}

export function readBrowserScenario(scenario: BehavioralScenario): { input: BrowserInput; expected: BrowserExpected } {
  if (typeof scenario.id !== 'string' || scenario.id === '') refuse('unrecognised-scenario', 'scenario.id must be a non-empty string');
  const { input, expected } = scenario;
  if (!isRecord(input)) refuse('unrecognised-scenario', `scenario ${scenario.id}: input must be an object`);
  unknownKeys(input, INPUT_KEYS, 'input');
  const [command, ...args] = strings(input.serve, 'input.serve');
  if (command === undefined) refuse('unrecognised-scenario', 'input.serve must be a non-empty array of non-empty strings');
  const port = integer(input.port, 'input.port', 1, 65535);
  const readyWithinMs = input.readyWithinMs === undefined ? DEFAULT_READY_WITHIN_MS : integer(input.readyWithinMs, 'input.readyWithinMs', 1, 600_000);
  if (!Array.isArray(input.steps)) refuse('unrecognised-scenario', 'input.steps must be a non-empty array');
  const [first, ...rest] = dense(input.steps).map(readStep);
  if (first === undefined) refuse('unrecognised-scenario', 'input.steps must be a non-empty array');
  // The browser starts on no page, so a step before the first goto has nothing to act on.
  if (first.do !== 'goto') refuse('unrecognised-scenario', 'input.steps[0] must be a goto: the browser starts on no page');
  const observes = [first, ...rest].filter((step) => step.do === 'observe').length;
  if (observes === 0) refuse('unrecognised-scenario', 'input.steps has no observe step, so the scenario would judge nothing');
  const mayBlock = input.mayBlock === undefined ? [] : strings(input.mayBlock, 'input.mayBlock').map((host) => host.trim().toLowerCase());

  if (!isRecord(expected)) refuse('unrecognised-scenario', `scenario ${scenario.id}: expected must be an object`);
  unknownKeys(expected, EXPECTED_KEYS, 'expected');
  if (!Array.isArray(expected.observations)) refuse('unrecognised-scenario', 'expected.observations must be an array');
  const observations = dense(expected.observations).map(readObservation);
  if (observations.length !== observes) {
    refuse('unrecognised-scenario', `expected.observations has ${String(observations.length)} entries for ${String(observes)} observe steps in input.steps; each observe step needs one`);
  }
  return { input: { serve: [command, ...args], port, readyWithinMs, steps: [first, ...rest], mayBlock }, expected: { observations } };
}

function quote(observed: string): string {
  if (observed.length <= QUOTE_LIMIT) return observed;
  return `${observed.slice(0, QUOTE_LIMIT)}… (${String(observed.length - QUOTE_LIMIT)} more characters in the result)`;
}

function compareObservation(expected: BrowserExpectedObservation, observed: BrowseObservation, at: string, out: ExpectationMismatch[]): void {
  if (observed.kind === 'not-reached') {
    out.push({ field: at, expected: 'an observation of the page', observed: `none was made: ${observed.reason}` });
    return;
  }
  const wanted = expected.matched ?? 1;
  if (observed.matched !== wanted) {
    out.push({
      field: `${at}.matched`,
      expected: expected.matched === undefined ? 'exactly 1 element, since no element is picked from several or from none' : String(wanted),
      observed: String(observed.matched),
    });
  }
  // What an element shows is compared only when the target named one element. Otherwise the mismatch on matched above is the failure.
  if (observed.matched === 1) {
    if (expected.visible !== undefined && observed.visible !== expected.visible) {
      out.push({ field: `${at}.visible`, expected: String(expected.visible), observed: String(observed.visible) });
    }
    if (expected.text !== undefined && observed.text !== expected.text) out.push({ field: `${at}.text`, expected: expected.text, observed: quote(observed.text) });
    if ('value' in expected && observed.value !== expected.value) {
      out.push({ field: `${at}.value`, expected: expected.value ?? '(no value)', observed: observed.value === null ? '(no value)' : quote(observed.value) });
    }
  }
  if (expected.path !== undefined && observed.path !== expected.path) out.push({ field: `${at}.path`, expected: expected.path, observed: quote(observed.path) });
  if (expected.title !== undefined && observed.title !== expected.title) out.push({ field: `${at}.title`, expected: expected.title, observed: quote(observed.title) });
}

/**
 * The runtime's comparison. Pure, so what decides a verdict is testable
 * without a browser. A port that never answered, a step that could not be
 * done, a refused host the scenario does not list, a document that tried to
 * leave the loopback, and an observation that was not made each fail it, and
 * none is read as a pass for want of something to compare (I5).
 */
export function compareBrowser(expected: BrowserExpected, mayBlock: readonly string[], observed: BrowseResult): ExpectationOutcome {
  const mismatches: ExpectationMismatch[] = [];
  if (!observed.ready) {
    mismatches.push({ field: 'ready', expected: 'the server listening on its port', observed: 'nothing accepted a connection before readyWithinMs passed' });
  } else {
    if (observed.failedStep !== null) {
      mismatches.push({ field: `steps[${String(observed.failedStep.index)}]`, expected: 'the step done', observed: observed.failedStep.reason });
    }
    for (const refused of observed.blocked) {
      const host = refused.host.toLowerCase();
      if (refused.navigation) {
        // Listed or not: a page that leaves the loopback has left what the scenario can observe.
        mismatches.push({ field: `blocked.${host}`, expected: "the document staying on the sandbox's loopback", observed: `the document tried to navigate to ${host}` });
      } else if (!mayBlock.includes(host)) {
        mismatches.push({
          field: `blocked.${host}`,
          expected: 'no request to a host the scenario does not list in mayBlock',
          observed: `${String(refused.requests)} refused, and the page was shown without what it asked ${host} for`,
        });
      }
    }
    expected.observations.forEach((wanted, i) => {
      const at = `observations[${String(i)}]`;
      const seen = observed.observations[i];
      if (seen === undefined) mismatches.push({ field: at, expected: 'an observation of the page', observed: 'no observation' });
      else compareObservation(wanted, seen, at, mismatches);
    });
    if (observed.observations.length > expected.observations.length) {
      mismatches.push({ field: 'observations', expected: `${String(expected.observations.length)} observations`, observed: String(observed.observations.length) });
    }
  }
  const [first, ...rest] = mismatches;
  return first === undefined ? { held: true } : { held: false, mismatches: [first, ...rest] };
}

/** A provider with a browser. Narrowed once, at construction, so `run` never finds it missing. */
type BrowsingProvider = SandboxProvider & { browse(h: SandboxHandle, request: BrowseRequest): Promise<BrowseResult> };

export function canBrowse(provider: SandboxProvider): provider is BrowsingProvider {
  return typeof provider.browse === 'function';
}

export class BrowserBehavioralAdapter implements BehavioralAdapter {
  readonly kind = 'browser' as const;
  readonly #provider: BrowsingProvider;
  readonly #served = new Set<SandboxHandle>();

  /** The provider that provisioned the handles this adapter will be given. One that cannot browse is refused: its browser would run where the product can replace it. */
  constructor(provider: SandboxProvider) {
    if (!canBrowse(provider)) {
      refuse('unsupported-feature', `sandbox provider ${provider.id} has no browse, so a browser would run where the product can replace it`);
    }
    this.#provider = provider;
  }

  async run(scenario: BehavioralScenario, h: SandboxHandle): Promise<CheckResult> {
    const { input, expected } = readBrowserScenario(scenario);
    if (this.#served.has(h)) {
      refuse('unrecognised-scenario', `scenario ${scenario.id}: sandbox ${h} already ran a browser scenario, whose server still holds it; each browser scenario needs a fresh sandbox`);
    }
    this.#served.add(h);
    const startedAt = new Date().toISOString();
    // A provider that cannot run the command throws, and there is no result to report: the
    // caller's gate treats a required check with no result as a failure.
    const started = await this.#provider.exec(h, [...input.serve], { detach: true });
    const base = { checkId: scenario.id, exitCode: started.exitCode, stderr: started.stderr, suiteCount: null, findings: null, property: null, startedAt };
    if (started.exitCode !== 0) {
      return {
        ...base,
        stdout: '',
        expectation: { held: false, mismatches: [{ field: 'serve', expected: 'the server started', observed: `the start exited ${String(started.exitCode)}: ${quote(started.stderr)}` }] },
        durationMs: started.durationMs,
      };
    }
    const observed = await this.#provider.browse(h, { port: input.port, readyWithinMs: input.readyWithinMs, steps: input.steps });
    return {
      ...base,
      // The browser's whole report is the evidence, the browser that made it included; the expectation is the verdict drawn from it.
      stdout: JSON.stringify(observed),
      expectation: compareBrowser(expected, input.mayBlock, observed),
      durationMs: started.durationMs + observed.durationMs,
    };
  }
}

/** A sandbox that can serve the product: Node, no egress, and a clock long enough for a step to wait out its target. */
function serving(dirs: SandboxDirs, proxyImage: string): SandboxSpec {
  return specFor(dirs, 'ro', { image: proxyImage, limits: { cpus: 0.5, memoryMb: 256, pids: 64, wallClockMs: 120_000 } });
}

/** A provider, the sandbox image, and the browser image already held, so no sandbox's clock pays for a pull. */
async function withBrowser(
  prefix: string,
  body: (rig: { provider: LocalDockerProvider; spec: () => SandboxSpec; browserImageId: string }) => Promise<void>,
): Promise<void> {
  const { BROWSER_IMAGE, EGRESS_PROXY_IMAGE, imageId } = await import('@olympus-ai/sandbox');
  const browserImageId = await imageId(BROWSER_IMAGE, { pull: true });
  await withProvider(prefix, (provider, dirs) => body({ provider, spec: () => serving(dirs, EGRESS_PROXY_IMAGE), browserImageId }));
}

/** Runs one scenario through the adapter in a sandbox of its own. */
async function judged(rig: { provider: LocalDockerProvider; spec: () => SandboxSpec }, s: Scenario): Promise<CheckResult> {
  const { BrowserBehavioralAdapter } = await import('@olympus-ai/adapters');
  return withSandbox(rig.provider, rig.spec(), (h) => new BrowserBehavioralAdapter(rig.provider).run(s, h));
}

function fields(result: CheckResult): string[] {
  const outcome = result.expectation;
  if (outcome === null) throw new Error('a browser scenario returned no expectation');
  return outcome.held ? [] : outcome.mismatches.map((m) => m.field);
}

function evidence(result: CheckResult): BrowseResult {
  return JSON.parse(result.stdout) as BrowseResult;
}

/** The browser containers running in `h`'s network namespace. */
async function browsersOf(h: SandboxHandle): Promise<string[]> {
  const { stdout } = await run('docker', ['ps', '--all', '--filter', 'name=browse-', '--format', '{{.Names}}']);
  const names = stdout.split(/\r?\n/).map((line) => line.trim()).filter((line) => line !== '');
  const joined: string[] = [];
  for (const name of names) {
    const mode = await run('docker', ['inspect', '--format', '{{.HostConfig.NetworkMode}}', name]).then((r) => r.stdout.trim(), () => '');
    if (mode === `container:${h}`) joined.push(name);
  }
  return joined;
}

async function seenBrowser(h: SandboxHandle): Promise<string> {
  for (let i = 0; i < 100; i += 1) {
    const [name] = await browsersOf(h);
    if (name !== undefined) return name;
    await new Promise((r) => setTimeout(r, 100));
  }
  throw new Error('I3: no browser container was seen while the call ran');
}

const SLOW = { port: 8080, readyWithinMs: 10_000, steps: [{ do: 'goto', path: '/slow' }, { do: 'observe', target: { text: 'Late' } }] } as const;

/**
 * Run inside the product's container while the browser runs: every port
 * listening in the shared network namespace but the product's own, and what
 * each answers a request with. A browser driven over a port would be one of
 * them, and would answer.
 */
const LISTENERS = `
  const fs = require('node:fs');
  const ports = new Set();
  for (const file of ['/proc/net/tcp', '/proc/net/tcp6']) {
    let text = '';
    try { text = fs.readFileSync(file, 'utf8'); } catch (error) { continue; }
    for (const line of text.split('\\n').slice(1)) {
      const cols = line.trim().split(/\\s+/);
      if (cols[3] === '0A') ports.add(parseInt(cols[1].split(':').pop(), 16));
    }
  }
  ports.delete(8080);
  const ask = (port) => new Promise((resolve) => {
    const req = require('node:http').get({ host: '127.0.0.1', port, path: '/json/version', timeout: 3000 }, (res) => { res.resume(); resolve([port, res.statusCode]); });
    req.on('timeout', () => { req.destroy(); resolve([port, 'timeout']); });
    req.on('error', (error) => resolve([port, error.code || 'error']));
  });
  Promise.all([...ports].map(ask)).then((answers) => process.stdout.write(JSON.stringify(answers)));
`;

export const BROWSER_OUT_OF_THE_PRODUCTS_REACH: LocalAssertion = runtime({
  id: 'I3.browser-out-of-the-products-reach',
  title:
    "the browser is inspected while it runs: in the sandbox's network namespace and its own process namespace, nothing mounted, a read-only root, every capability " +
    'dropped and none added, no new privileges, an unprivileged user, without a file the product wrote, and with no port the product can drive it through; ' +
    'a deny-all sandbox it browsed is still --network none, and no browser outlives its call or its sandbox',
  run: async () => {
    await withBrowser('conformance-browser-reach-', async ({ provider, spec }) => {
      await withSandbox(provider, spec(), async (h) => {
        const started = await provider.exec(h, ['node', '--eval', SERVER], { detach: true });
        if (started.exitCode !== 0) throw new Error(`I3: the product's server did not start: ${started.stderr}`);
        const planted = await provider.exec(h, ['sh', '-c', 'echo forged > /tmp/browse-marker']);
        if (planted.exitCode !== 0) throw new Error("I3: could not plant the marker in the product's container");

        const call = provider.browse(h, SLOW);
        const browser = await seenBrowser(h);
        // PidMode empty is Docker's private PID namespace: the product's processes are not the browser's.
        const format =
          '{{.HostConfig.NetworkMode}}|{{.HostConfig.PidMode}}|{{json .Mounts}}|{{json .HostConfig.Binds}}|{{json .HostConfig.Tmpfs}}|{{.HostConfig.ReadonlyRootfs}}|' +
          '{{json .HostConfig.CapDrop}}|{{json .HostConfig.CapAdd}}|{{.HostConfig.Privileged}}|{{json .HostConfig.SecurityOpt}}|{{.Config.User}}';
        const inspected = (await run('docker', ['inspect', '--format', format, browser])).stdout.trim();
        const wanted = `container:${h}||[]|null|null|true|["ALL"]|null|false|["no-new-privileges"]|65534:65534`;
        if (inspected !== wanted) throw new Error(`I3: the browser ran as ${inspected}, not ${wanted}`);
        // Asked inside the browser's container, with its own node binary as a control, so an exec
        // that failed cannot pass for a file that is absent.
        const check =
          "const fs = require('node:fs'); process.stdout.write(JSON.stringify({ marker: fs.existsSync('/tmp/browse-marker'), control: fs.existsSync(process.execPath) }))";
        const inside = (await run('docker', ['exec', browser, 'node', '--eval', check])).stdout.trim();
        if (inside !== '{"marker":false,"control":true}') throw new Error(`I3: inside the browser's container, the product's marker and the control read ${inside}`);

        // From the product's side of the shared network: whatever the browser's container listens on refuses the product.
        let listeners: Array<[number, number | string]> = [];
        for (let i = 0; i < 30 && listeners.length < 2; i += 1) {
          const asked = await provider.exec(h, ['node', '--eval', LISTENERS]);
          if (asked.exitCode !== 0) throw new Error(`I3: the product's container could not list what listens beside it: ${asked.stderr}`);
          listeners = JSON.parse(asked.stdout) as Array<[number, number | string]>;
          if (listeners.length < 2) await new Promise((r) => setTimeout(r, 100));
        }
        if (listeners.length < 2) throw new Error(`I3 control: the browser's two refusing listeners were not seen from the product's container (${JSON.stringify(listeners)})`);
        const open = listeners.filter(([, status]) => status !== 403);
        if (open.length > 0) throw new Error(`I3: the product reached a listener beside it that did not refuse it: ${JSON.stringify(open)}`);

        const result = await call;
        const [seen] = result.observations;
        if (seen?.kind !== 'observed' || seen.text !== 'Late') throw new Error(`I3: the browser did not observe the product's page: ${JSON.stringify(result)}`);
        const network = (await run('docker', ['inspect', '--format', '{{.HostConfig.NetworkMode}}', h])).stdout.trim();
        if (network !== 'none') throw new Error(`I3: browsing a deny-all sandbox changed its network to ${network}`);
        const left = await browsersOf(h);
        if (left.length > 0) throw new Error(`I3: browser containers outlived their call: ${left.join(', ')}`);
      });

      // A call still running when its sandbox ends: the browser goes with it.
      const h = await provider.provision(spec());
      let call: Promise<unknown> | undefined;
      try {
        const started = await provider.exec(h, ['node', '--eval', SERVER], { detach: true });
        if (started.exitCode !== 0) throw new Error(`I3: the product's server did not start: ${started.stderr}`);
        call = provider.browse(h, SLOW).catch((error: unknown) => error);
        await seenBrowser(h);
      } finally {
        await provider.destroy(h);
      }
      await call;
      const outlived = await browsersOf(h);
      if (outlived.length > 0) throw new Error(`I3: browser containers outlived their sandbox: ${outlived.join(', ')}`);
    });
  },
});

          const found = await find(target);
          if (found.length === 1) {
            const seen = await on(found[0], DESCRIBE, []);
            if (seen.visible) return { objectId: found[0], seen: seen };
            why = shown(target) + ' is on the page and not visible';
          } else {
            why = found.length === 0 ? shown(target) + ' matched no element' : shown(target) + ' matched ' + found.length + ' elements, and none is picked';
          }
        } catch (error) {
          why = String(error && error.message || error);
        }
        if (Date.now() >= deadline) throw new Error(why);
        await sleep(100);
      }
    }
    /** Returns when no request is in flight and the document has stopped changing, or when SETTLE_MS has passed. */
    async function settle() {
      const deadline = Date.now() + SETTLE_MS;
      while (Date.now() < deadline) {
        if (inflight.size > 0) { await sleep(50); continue; }
        let quiet = false;
        try { quiet = (await evaluate(QUIET, [QUIET_MS], true)).value === true; } catch (error) { await sleep(50); }
        if (quiet && inflight.size === 0) return;
      }
    }
    // A click lands where a person's would: on whatever is at the middle of the element.
    async function click(objectId) {
      await cdp.send('DOM.scrollIntoViewIfNeeded', { objectId: objectId });
      const quads = (await cdp.send('DOM.getContentQuads', { objectId: objectId })).quads;
      if (quads.length === 0) throw new Error('the element has no area to click');
      const q = quads[0];
      await page.mouse.click((q[0] + q[2] + q[4] + q[6]) / 4, (q[1] + q[3] + q[5] + q[7]) / 4);
    }
    async function observe(target) {
      const found = await find(target);
      const seen = found.length === 1 ? await on(found[0], DESCRIBE, []) : { visible: false, text: '', value: null };
      const at = new URL(page.url());
      return { kind: 'observed', matched: found.length, visible: seen.visible, text: seen.text, value: seen.value, path: at.pathname + at.search, title: await page.title() };
    }
    async function perform(step) {
      if (step.do === 'goto') {
        // Joined as text, so the host is fixed before the path's first character whatever the path holds.
        const url = parse(origin + step.path);
        if (url === null || url.host !== new URL(origin).host) throw new Error('the path does not stay on the loopback');
        await page.goto(url.href, { waitUntil: 'load', timeout: STEP_MS });
      } else if (step.do === 'click') {
        await click((await one(step.target)).objectId);
      } else if (step.do === 'fill') {
        const field = await one(step.target);
        if (field.seen.tag === 'SELECT') {
          const refusal = await on(field.objectId, CHOOSE, [norm(step.value)]);
          if (refusal !== '') throw new Error(shown(step.target) + ' could not be set: ' + refusal);
        } else {
          const typed = field.seen.tag === 'TEXTAREA' || field.seen.editable || (field.seen.tag === 'INPUT' && !NOT_TYPED.includes(field.seen.type));
          if (!typed) throw new Error(shown(step.target) + ' is not a field that takes typed text');
          await click(field.objectId);
          await page.keyboard.down('Control');
          await page.keyboard.press('a');
          await page.keyboard.up('Control');
          await page.keyboard.press('Backspace');
          await page.keyboard.type(step.value);
        }
      } else if (step.do === 'press') {
        await page.keyboard.press(step.key);
      } else {
        throw new Error('an unknown step: ' + JSON.stringify(step.do));
      }
    }
    /** Throws when the step left the page somewhere a scenario does not go. */
    function stayed() {
      if (interrupted !== null) throw new Error(interrupted);
      const at = parse(page.url());
      const here = web(at) && isLoopback(at.hostname);
      if (web(at) && !here) navigated.add(at.hostname.toLowerCase());
      if (navigated.size > 0) throw new Error('the page left the loopback for ' + [...navigated].sort().join(', '));
      if (!here) throw new Error('the page left the loopback and is at ' + String(page.url()).slice(0, 200));
    }

    for (let i = 0; i < request.steps.length; i += 1) {
      const step = request.steps[i];
      if (result.failedStep !== null) {
        if (step.do === 'observe') result.observations.push({ kind: 'not-reached', reason: 'step ' + result.failedStep.index + ' could not be done' });
        continue;
      }
      try {
        if (step.do === 'observe') {
          await settle();
          const seen = await observe(step.target);
          stayed();
          result.observations.push(seen);
        } else {
          await perform(step);
          await settle();
          stayed();
        }
      } catch (error) {
        result.failedStep = { index: i, reason: String(error && error.message || error) };
        if (step.do === 'observe') result.observations.push({ kind: 'not-reached', reason: 'step ' + i + ' could not be done' });
      }
    }
    const hosts = new Set([...refused.keys(), ...navigated]);
    result.blocked = [...hosts].sort().map(function (h) { return { host: h, requests: refused.get(h) || 0, navigation: navigated.has(h) }; });
  }
  result.durationMs = Date.now() - started;
  // The container ends with this process, and Chrome with it.
  process.stdout.write(JSON.stringify(result), function () { process.exit(0); });
}

let request;
try { request = JSON.parse(REQUEST); } catch (error) { fail('the request is not JSON'); }
main(request).catch(function (error) { fail(String(error && error.stack || error)); });
`;

/**
 * The browser a browser scenario observes the product through (A-R3-01).
 *
 * Built beside the probe, and for the probe's reason: a browser inside the
 * product's container is one the product can replace, and what it saw would
 * be the product's report of itself.
 *
 * - One container per `browse()` call, started with
 *   `--network container:<sandbox>`. It shares the sandbox's loopback and not
 *   its filesystem or its process tree, and is gone when the call returns.
 * - It has the probe's limits and nothing more: a read-only root, every
 *   capability dropped, no new privileges, an unprivileged user, nothing
 *   mounted. It is also given a memory and a process ceiling, since a page is
 *   the agent's content and the sandbox's own limits do not reach a second
 *   container (D-R3-10).
 * - The request names a port and never a host. Everything else the page asks
 *   for is refused inside the container and reported (`browse-source.ts`).
 * - It observes and decides nothing.
 *
 * The image is the one place a browser comes from: pinned by digest, pulled
 * only by digest, and run by the content ID the engine reports for it, which
 * is what a result names.
 */
import type { BrowseBlockedHost, BrowseObservation, BrowseRequest, BrowseResult, BrowseStep } from '../types.js';
import { BROWSE_SOURCE } from './browse-source.js';
import { refuse } from './refusal.js';

/** How long a `click` or `fill` waits for its target to name exactly one visible element, and a `goto` for its page to load. */
export const BROWSE_STEP_MS = 10_000;

/** The longest the browser waits for a page to go quiet before it reads it (D-R3-07). */
export const BROWSE_SETTLE_MS = 5_000;

/** How long a document must go unchanged, with no request in flight, to count as quiet. */
export const BROWSE_QUIET_MS = 250;

/** The browser container's memory ceiling, swap included. */
export const BROWSE_MEMORY = '2g';

/** The most processes and threads the browser container may hold. */
export const BROWSE_PIDS = 1024;

const STEP_KEYS: Readonly<Record<BrowseStep['do'], readonly string[]>> = {
  goto: ['do', 'path'],
  click: ['do', 'target'],
  fill: ['do', 'target', 'value'],
  press: ['do', 'key'],
  observe: ['do', 'target'],
};

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

/** True when `text` holds a character below space, or DEL. */
function hasControl(text: string): boolean {
  for (let i = 0; i < text.length; i += 1) {
    const code = text.charCodeAt(i);
    if (code < 0x20 || code === 0x7f) return true;
  }
  return false;
}

function sameKeys(value: Record<string, unknown>, keys: readonly string[]): boolean {
  const own = Object.keys(value);
  return own.length === keys.length && keys.every((key) => own.includes(key));
}

function checkTarget(target: unknown, at: string): void {
  const forms = 'a role and name, a label, or a text';
  if (!isRecord(target)) refuse('browse', `${at} must name its element by ${forms}`);
  if (sameKeys(target, ['role', 'name'])) {
    if (typeof target.role !== 'string' || target.role.trim() === '' || hasControl(target.role)) refuse('browse', `${at}.role must be a non-empty string`);
    if (typeof target.name !== 'string') refuse('browse', `${at}.name must be a string`);
    return;
  }
  for (const form of ['label', 'text'] as const) {
    if (!sameKeys(target, [form])) continue;
    const value = target[form];
    if (typeof value !== 'string' || value.trim() === '') refuse('browse', `${at}.${form} must be a non-empty string`);
    return;
  }
  // A selector, a script, or a URL is a key none of the three forms has.
  refuse('browse', `${at} must name its element by exactly one of ${forms}; it has ${Object.keys(target).join(', ') || 'no key'}`);
}

function checkStep(step: unknown, i: number): void {
  const at = `steps[${String(i)}]`;
  if (!isRecord(step)) refuse('browse', `${at} must be an object`);
  const kind = step.do;
  if (typeof kind !== 'string' || !Object.hasOwn(STEP_KEYS, kind)) refuse('browse', `${at}.do must be one of goto, click, fill, press, observe`);
  const keys = STEP_KEYS[kind as BrowseStep['do']];
  const extra = Object.keys(step).filter((key) => !keys.includes(key));
  if (extra.length > 0) refuse('browse', `${at} has ${extra.join(', ')}, which a ${kind} step does not carry`);
  if (kind === 'goto') {
    const path = step.path;
    if (typeof path !== 'string' || !path.startsWith('/') || /\s/.test(path) || hasControl(path)) {
      refuse('browse', `${at}.path must begin with / and contain no whitespace or control characters; the host is always the sandbox's loopback`);
    }
    return;
  }
  if (kind === 'press') {
    const key = step.key;
    if (typeof key !== 'string' || key === '' || key.length > 32 || hasControl(key)) refuse('browse', `${at}.key must be the name of one key`);
    return;
  }
  checkTarget(step.target, `${at}.target`);
  if (kind === 'fill' && typeof step.value !== 'string') refuse('browse', `${at}.value must be a string`);
}

/** Refuses a request the browser could not carry out as written, naming the field, rather than doing something else. */
export function checkBrowseRequest(request: BrowseRequest): void {
  if (!Number.isInteger(request.port) || request.port < 1 || request.port > 65535) refuse('browse', 'port must be an integer from 1 to 65535');
  if (!Number.isInteger(request.readyWithinMs) || request.readyWithinMs <= 0) refuse('browse', 'readyWithinMs must be a positive integer');
  const steps: unknown = request.steps;
  if (!Array.isArray(steps) || steps.length === 0) refuse('browse', 'steps must be a non-empty list');
  // Array.from, so a hole is an entry that is checked and refused rather than one forEach skips.
  Array.from(steps as readonly unknown[]).forEach(checkStep);
  // The browser starts on no page, so anything before a goto has nothing to act on.
  if (request.steps[0].do !== 'goto') refuse('browse', 'steps[0] must be a goto: the browser starts on no page');
}

/**
 * What is written to the standard input of `node -` in the browser's
 * container: the request as a string literal, then the runner. JSON's own
 * escaping makes the request a literal and nothing else, so no part of a
 * scenario is read as code.
 */
export function browseProgram(request: BrowseRequest): string {
  return `'use strict';\nconst REQUEST = ${JSON.stringify(JSON.stringify(request))};\n${BROWSE_SOURCE}`;
}

function readObservation(value: unknown): BrowseObservation {
  if (!isRecord(value)) throw new Error('an observation is not an object');
  if (value.kind === 'not-reached' && typeof value.reason === 'string') return { kind: 'not-reached', reason: value.reason };
  if (
    value.kind === 'observed' &&
    typeof value.matched === 'number' && Number.isInteger(value.matched) && value.matched >= 0 &&
    typeof value.visible === 'boolean' &&
    typeof value.text === 'string' &&
    (value.value === null || typeof value.value === 'string') &&
    typeof value.path === 'string' &&
    typeof value.title === 'string'
  ) {
    return { kind: 'observed', matched: value.matched, visible: value.visible, text: value.text, value: value.value, path: value.path, title: value.title };
  }
  throw new Error(`an observation has an unknown shape: ${JSON.stringify(value).slice(0, 200)}`);
}

function readBlocked(value: unknown): BrowseBlockedHost {
  if (
    !isRecord(value) || typeof value.host !== 'string' || value.host === '' ||
    typeof value.requests !== 'number' || !Number.isInteger(value.requests) || value.requests < 0 || typeof value.navigation !== 'boolean'
  ) {
    throw new Error(`a refused host has an unknown shape: ${JSON.stringify(value).slice(0, 200)}`);
  }
  return { host: value.host, requests: value.requests, navigation: value.navigation };
}

/**
 * Reads what the browser's runner printed. The runner is the runtime's own
 * code, so a shape this does not know, or a report that does not add up, is
 * the runner breaking, and it throws rather than reports an observation that
 * was not made. `image` is the content ID the container was started from: the
 * provider's fact, which nothing inside the container supplies.
 */
export function readBrowseOutput(stdout: string, request: BrowseRequest, image: string): BrowseResult {
  let parsed: unknown;
  try {
    parsed = JSON.parse(stdout);
  } catch {
    throw new Error(`the browser printed something that is not JSON: ${stdout.slice(0, 200)}`);
  }
  if (
    !isRecord(parsed) || typeof parsed.ready !== 'boolean' || !Array.isArray(parsed.observations) || !Array.isArray(parsed.blocked) ||
    typeof parsed.durationMs !== 'number' || !isRecord(parsed.browser)
  ) {
    throw new Error('the browser printed a result with an unknown shape');
  }
  const { name, version } = parsed.browser;
  if (typeof name !== 'string' || name === '' || typeof version !== 'string' || version === '') throw new Error('the browser did not say what it is');

  let failedStep: BrowseResult['failedStep'] = null;
  if (parsed.failedStep !== null) {
    const failed = parsed.failedStep;
    if (!isRecord(failed) || typeof failed.index !== 'number' || !Number.isInteger(failed.index) || typeof failed.reason !== 'string') {
      throw new Error('the browser reported a failed step with an unknown shape');
    }
    if (failed.index < 0 || failed.index >= request.steps.length) throw new Error(`the browser reported step ${String(failed.index)} failed, which the request does not have`);
    failedStep = { index: failed.index, reason: failed.reason };
  }

  const observations = Array.from(parsed.observations as readonly unknown[]).map(readObservation);
  const observes = request.steps.flatMap((step, i) => (step.do === 'observe' ? [i] : []));
  const expected = parsed.ready ? observes.length : 0;
  if (observations.length !== expected) throw new Error(`the browser reported ${String(observations.length)} observations for ${String(expected)} observe steps`);
  if (!parsed.ready && failedStep !== null) throw new Error('the browser reported a failed step on a port that never answered');
  // Which observations were made follows from the step that failed, and a report that disagrees is not one to compare.
  observations.forEach((observation, n) => {
    const stepIndex = observes[n];
    const reached = failedStep === null || (stepIndex !== undefined && stepIndex < failedStep.index);
    if (reached !== (observation.kind === 'observed')) throw new Error(`the browser's observation ${String(n)} does not agree with the step it says failed`);
  });

  return {
    ready: parsed.ready,
    observations,
    failedStep,
    blocked: Array.from(parsed.blocked as readonly unknown[]).map(readBlocked),
    browser: { name, version, image },
    durationMs: parsed.durationMs,
  };
}

     const relayPlan = spec.relay === undefined ? undefined : checkRelay(spec.relay, new Set(this.#credentials.keys()), this.#hostPorts);
     if (plan.mode === 'allowlist' && relayPlan !== undefined && plan.hosts.includes(relayPlan.upstreamHost)) {
       // The proxy would be a second route to the host the relay exists to be the only route to.
       // Both are the runtime's to choose, and a spec that chose both contradicts itself (external review, codex-2).
       refuse(
         'relay',
         `egress.allow names ${relayPlan.upstreamHost}, the relay's upstream host. The relay is the sandbox's only route to its upstream, ` +
           'and an allowlist that also reaches it is a route around the grant; the spec is refused rather than half-applied.',
       );
     }
     checkLimits(spec.limits);
     checkUser(spec.user);
 
     // Re-validated rather than trusted: the table may have been built by a cast, parsed from a
     // document, or handed over by JavaScript, in which case the type-level guarantee never ran.
     const table = mountTable({ workspace: spec.mounts.workspace, others: spec.mounts.others });
     const mounts = await resolveMounts(table, this.#vaultPaths);
     const workspace = mounts.find((mount) => mount.target === table.workspace.target);
     // resolveMounts returns one entry per table entry, the workspace among them.
     if (workspace === undefined) refuse('mount', 'the resolved mount table has no workspace');
     await checkWorkspaceWritable(workspace, spec.user);
 
     // I10: no Greek name in code, container names included. This one reaches `docker ps`.
     const id = randomUUID();
     const name = `sandbox-${id}`;
     const egress = await this.#applyEgress(id, plan, relayPlan);
     try {
       return await this.#start(spec, mounts, name, egress);
     } catch (error) {
       // Whatever refused, the proxy, the relay, and their networks were created for a sandbox that
       // does not exist. They go with it: a leaked route out is worse than the failure that caused it,
       // and one that would not go is reported beside that failure, never instead of it or not at all.
       const failed = withLeftovers(error, await this.#stopSidecars(egress));
       throw failed instanceof SandboxRefusal ? new SandboxRefusal(failed.layer, failed.message, name) : failed;
     }
   }
 
   /**
    * Starts the proxy an allowlist needs and the relay a spec asked for, or
    * nothing at all.
    *
    * `deny-all` without a relay is untouched: no network is created, nothing is
    * started, and the container is given `--network none` exactly as before.
    * `deny-all` with a relay puts the sandbox on an internal network that
    * carries the relay alone (D-P12-02). Starting a proxy beside a sandbox
    * entitled to no egress would put a route out next to the one policy that
    * asked for none, and nothing here does.
    */
   async #applyEgress(id: string, plan: EgressPlan, relayPlan: RelayPlan | undefined): Promise<AppliedEgress> {
     if (plan.mode === 'deny-all') {
       if (relayPlan === undefined) return { mode: 'deny-all', network: 'none' };
       const relay = await this.#startRelay(id, relayPlan, undefined);
       return { mode: 'deny-all', network: relay.internalNetwork, relay };
     }
     let proxy: AppliedProxy;
     try {
       proxy = await startProxy(id, plan.hosts, this.#proxyOptions);
     } catch (error) {
       // I5: an allowlist this provider could not stand up is a refusal. There is no fallback to
       // deny-all, which breaks the run silently, and none to a routed network, which hands the
       // whole of it to a policy that asked for a subset.
       refuse(
         'egress',
         'the egress allowlist could not be enforced, so no sandbox was provisioned: ' +
           (error instanceof Error ? error.message : String(error)),
       );
     }
     if (relayPlan === undefined) return { mode: 'allowlist', network: proxy.internalNetwork, allow: plan.hosts, proxy };
     let relay: AppliedRelay;
     try {
       relay = await this.#startRelay(id, relayPlan, { internal: proxy.internalNetwork, outbound: proxy.outboundNetwork });
     } catch (error) {
       throw withLeftovers(error, await stopProxy(proxy, this.#proxyOptions));
     }
     return { mode: 'allowlist', network: proxy.internalNetwork, allow: plan.hosts, proxy, relay };
   }
 
   /**
    * Starts a relay, or refuses. The credential is read from this provider's
    * own map here and handed to the relay container alone; `checkRelay` has
    * already refused a name the map does not hold.
    */
   async #startRelay(id: string, plan: RelayPlan, shared: { internal: string; outbound: string } | undefined): Promise<AppliedRelay> {
     // A declared null is none (A-R21-02); `checkRelay` allowed it only for a local upstream.
     const credential = plan.credential === null ? null : this.#credentials.get(plan.credential);
     if (credential === undefined) refuse('relay', `the credential ${JSON.stringify(plan.credential)} is not held by this provider`);
     try {
       return await startRelay(id, plan, credential, this.#relayTrust, shared, this.#relayOptions);
     } catch (error) {
       refuse(
         'relay',
         'the model relay could not be started, so no sandbox was provisioned: ' + (error instanceof Error ? error.message : String(error)),
       );
     }
   }
 
   /** Removes the relay and then the proxy, in that order: the proxy's networks cannot go while the relay is on them. */
   async #stopSidecars(egress: AppliedEgress): Promise<Error | undefined> {
     const relay = relayOf(egress);
     const relayFailure = relay === undefined ? undefined : await stopRelay(relay, this.#relayOptions);
     const proxyFailure = egress.mode === 'allowlist' ? await stopProxy(egress.proxy, this.#proxyOptions) : undefined;
     return relayFailure ?? proxyFailure;
   }
 
   async #start(spec: SandboxSpec, mounts: readonly ResolvedMount[], name: string, egress: AppliedEgress): Promise<SandboxHandle> {
     const args = runArgsFor(spec, mounts, name, egress, this.#labels);
     const started = await dockerCli(this.#executable, args, { timeoutMs: LIFECYCLE_TIMEOUT_MS });
     if (started.exitCode !== 0) {
       refuse('image', `docker run exited ${String(started.exitCode)} for image ${spec.image}: ${started.stderr.trim() || started.stdout.trim()}`);
     }
     const containerId = started.stdout.trim();
     if (containerId === '') refuse('image', 'docker run reported no container id, so there is no sandbox to hand back');
 
     // A container that exits immediately — a shell-less image, a bad entrypoint — would otherwise
     // hand back a handle whose every exec fails obscurely.
     const running = await dockerCli(this.#executable, ['inspect', '--format', '{{.State.Running}}', containerId], {
       timeoutMs: LIFECYCLE_TIMEOUT_MS,
     });
     if (running.stdout.trim() !== 'true') {
       // Best effort: the refusal below is the reason the caller needs, and a container that
       // never started is not one this provider is handing out a handle to.
       await this.#remove(containerId);
       refuse('image', `the container from image ${spec.image} was not running after start; it needs a POSIX shell at \`sh\` to hold itself open`);
     }
 
     const handle = containerId as SandboxHandle;
     const sandbox: Sandbox = {
       controls: {
         containerId,
         mounts,
         egress,
         limits: { ...spec.limits },
         deadline: performance.now() + spec.limits.wallClockMs,
         runArgs: Object.freeze([...args]),
       },
       probes: new Set(),
     };
     this.#sandboxes.set(handle, sandbox);
 
     // The budget has to be enforced by something that runs on its own. Checking it inside `exec`
     // bounds only sandboxes somebody keeps calling: a task that starts a background process and
     // is never exec'd again would outlive its limit entirely, which is the whole thing the limit
     // exists to prevent. Review finding 2.
     const timer = setTimeout(() => {
       void this.#expire(sandbox, spec.limits.wallClockMs);
     }, spec.limits.wallClockMs);
     // The runtime is a service (I9), and a pending timer must never be the reason its process
     // stays alive. An unref'd timer still fires for as long as the process is running.
     timer.unref();
     sandbox.timer = timer;
     return handle;
   }
 
   async exec(h: SandboxHandle, cmd: string[], options: ExecOptions = {}): Promise<ExecResult>  {
     const sandbox = this.#require(h);
     if (sandbox.ended !== undefined) refuse('lifetime', `sandbox ${h} has ended: ${sandbox.ended}`);
     if (cmd.length === 0) refuse('handle', 'the command is empty; there is nothing to run');
     const detach = options.detach === true;
     if (detach && options.stdin !== undefined) refuse('probe', 'a detached command cannot be handed stdin; it would run without it');
     const passthrough = environmentPassthrough(options.env);

     try {

         this.#executable,

         // run as if it had been handed nothing.

     } catch (error) {

   /**
    * Posts one request to the sandbox's relay from inside the relay container
    * (A-R6-05): read-only, with no capabilities, and out of the sandbox's
    * filesystem and process tree, so nothing the task wrote answers in the
    * relay's place. The time it takes is the sandbox's, as an `exec`'s is.
    */
   async relayCall(h: SandboxHandle, request: RelayCallRequest): Promise<RelayCallResult> {
     const sandbox = this.#require(h);
     if (sandbox.ended !== undefined) refuse('lifetime', `sandbox ${h} has ended: ${sandbox.ended}`);
     const relay = relayOf(sandbox.controls.egress);
     if (relay === undefined) refuse('relay', `sandbox ${h} was provisioned without a model relay; there is nothing to call`);
     if (!request.path.startsWith('/')) refuse('relay', `the relay call's path ${JSON.stringify(request.path)} does not begin with /`);
     const remaining = await this.#remaining(sandbox, h);
     try {
       const result = await dockerCli(
         this.#executable,
         ['exec', '--interactive', relay.containerId, 'node', '--eval', RELAY_CALL_SOURCE, request.path],
         { timeoutMs: remaining, ...this.#outputCap, stdin: request.body },
       );
       if (result.exitCode !== 0) {
         throw new Error(`the relay call exited ${String(result.exitCode)}: ${result.stderr.trim() || result.stdout.trim()}`);
       }
       let answer: unknown;
       try {
         answer = JSON.parse(result.stdout);
       } catch {
         answer = undefined;
       }
       if (typeof answer !== 'object' || answer === null) {
         throw new Error(`the relay call printed ${JSON.stringify(result.stdout.slice(0, 200))}, not a status and a body`);
       }
       const { status, body } = answer as { status?: unknown; body?: unknown };
       if (typeof status !== 'number' || typeof body !== 'string') {
         throw new Error(`the relay call printed ${JSON.stringify(result.stdout.slice(0, 200))}, not a status and a body`);
       }
       return { status, body };
     } catch (error) {
       return await this.#overBudget(sandbox, h, error);
     }
   }
 
   /** What is left of the sandbox's wall clock, or a refusal that ends it when nothing is. */
   async #remaining(sandbox: Sandbox, h: SandboxHandle): Promise<number> {
     const remaining = sandbox.controls.deadline - performance.now();
     if (remaining <= 0) {
       const budget = String(sandbox.controls.limits.wallClockMs);
       await this.#end(sandbox, `its wall-clock budget of ${budget}ms was already spent`);
       refuse('lifetime', `sandbox ${h} is past its wall-clock limit of ${budget}ms; the command was not started`);
     }
     return remaining;
   }
 
   /**
    * Rethrows anything but a timeout or an output overrun; either ends the
    * sandbox and refuses. Killing the `docker exec` client does not stop the
    * command inside the container, so the container goes, and with it the
    * command (D-P8-15).
    */
   async #overBudget(sandbox: Sandbox, h: SandboxHandle, error: unknown): Promise<never> {
     if (error instanceof CliOutputExceeded) {
       const cap = String(error.maxOutputBytes);
       const { removal } = await this.#end(sandbox, `a command printed more than ${cap} bytes`);
       if (removal !== undefined) {
         refuse('output', `sandbox ${h} ran a command that printed more than ${cap} bytes, and could not be destroyed: ${removal.message}`);
       }
       refuse('output', `sandbox ${h} ran a command that printed more than ${cap} bytes; it was killed with its sandbox, and none of its output is returned`);
     }
     if (!(error instanceof CliTimeout)) throw error;
     const budget = String(sandbox.controls.limits.wallClockMs);
     // The budget is the sandbox's, not the command's: the container goes with it, so nothing
     // left running inside outlives the limit that was supposed to bound it.
     const { removal } = await this.#end(sandbox, `it exceeded its wall-clock limit of ${budget}ms`);
     if (removal !== undefined) {
       // The breach is still the headline, but a container that outlived its budget *and*
       // could not be removed is worse than one that was, and must not be reported as less.
       refuse('lifetime', `sandbox ${h} exceeded its wall-clock limit of ${budget}ms and could not be destroyed: ${removal.message}`);
     }
     refuse('lifetime', `sandbox ${h} exceeded its wall-clock limit of ${budget}ms and was destroyed`);
   }
 
   /**
    * Ends the sandbox and returns what its relay counted (D-P13-07). A sandbox
    * its wall clock already ended returns the reading taken then: the call it
    * served spent money whether or not it finished in time, and the runtime
    * records it either way (D-P13-14).
    */
   async destroy(h: SandboxHandle): Promise<Teardown> {
     const sandbox = this.#require(h);
     if (sandbox.destroyed === true) refuse('lifetime', `sandbox ${h} has already ended: ${sandbox.ended ?? 'it was destroyed'}`);
     sandbox.destroyed = true;
     const { removal, reading, egress } = await this.#end(sandbox, 'it was destroyed');
     if (removal !== undefined) throw removal;
     if (reading instanceof Error) throw reading;
     if (egress instanceof Error) throw egress;
     return { meter: reading, egress };
   }
 
   /**
    * Each value is asserted against a provisioned container by the conformance
    * registry (I8), so flipping one here without changing what the provider
    * does fails the suite. `persistent` is true because one container serves
    * every exec; the others are false or `linux` because `create()` refused to
    * build a provider for which they were not.
    */
   capabilities(): SandboxCapabilities {
     return { computerUse: false, gpu: false, os: this.#daemon.osType, persistent: true, remote: false };

   #require(h: SandboxHandle): Sandbox {
     const sandbox = this.#sandboxes.get(h);
     if (sandbox === undefined) refuse('handle', `${h} is not a handle this provider issued`);
     return sandbox;

   /**
    * Ends a sandbox and keeps its record. The record is not deleted, for two
    * reasons: a later call then says the sandbox ended and why, instead of
    * "unknown handle" for a container the runtime itself destroyed; and
    * `appliedControls()` keeps answering afterwards, so what was enforced on a
    * sandbox outlives the sandbox and can still be read when the run is
    * audited. Returns the removal failure rather than throwing it, so a caller
    * that is already refusing for a better reason can decide which to report.
    */
   async #end(sandbox: Sandbox, why: string): Promise<Ended> {
     // Whoever gets here second awaits the first removal rather than starting another, and so
     // does not return until the container is actually gone.
     if (sandbox.ending !== undefined) return sandbox.ending;
     sandbox.ended = why;
     if (sandbox.timer !== undefined) {
       clearTimeout(sandbox.timer);
       sandbox.timer = undefined;
     }
     sandbox.ending = this.#dismantle(sandbox.controls, sandbox.probes);
     return sandbox.ending;
   }
 
   /**
    * Removes everything one sandbox was given: the container, its relay, and
    * under an allowlist the proxy container, and the networks either created.
    *
    * The proxy goes with the sandbox it serves and with nothing else. A proxy
    * that outlived its sandbox would be a route out with no workload behind it
    * and nobody watching it, and the assertion that counts leaked containers is
    * there to make that a failing test rather than a thing somebody notices in
    * `docker ps` a week later.
    */
   async #dismantle(controls: AppliedControls, probes: ReadonlySet<string>): Promise<Ended> {
     // Probes first: each is in the container's namespace, and none may outlive it (D-P11-02).
     // A probe that already finished removed itself, so a failure here is not reported.
     await Promise.all([...probes].map((name) => this.#remove(name)));
     // Then the container: a network still holding an endpoint cannot be removed, and a meter read
     // while the sandbox runs can be overtaken by a process still calling the relay (D-P13-07).
     const container = await this.#remove(controls.containerId);
     const relay = relayOf(controls.egress);
     let reading: MeterReading | Error = { kind: 'unmetered' };
     if (relay !== undefined) {
       try {
         reading = await readMeter(relay, this.#relayOptions);
       } catch (error) {
         // Kept, not thrown: the relay is still torn down below, and `destroy` reports this.
         reading = new Error(`the model relay's meter could not be read: ${error instanceof Error ? error.message : String(error)}`);
       }
     }
     let egress: EgressLog | Error = { kind: 'none' };
     if (controls.egress.mode === 'allowlist') {
       try {
         egress = await readEgress(controls.egress.proxy, this.#proxyOptions);
       } catch (error) {
         // Kept, not thrown, as the meter's is: the proxy is still torn down below, and `destroy` reports this.
         egress = new Error(`the egress proxy's decisions could not be read: ${error instanceof Error ? error.message : String(error)}`);
       }
     }
     const sidecars = await this.#stopSidecars(controls.egress);
     return { removal: container ?? sidecars, reading, egress };
   }
 
   /**
    * The wall-clock timer firing. Nothing is waiting on this call, so a removal
    * failure has no caller to refuse: it is recorded on the sandbox, where the
    * next `exec` or `destroy` reports it, rather than raised as an unhandled
    * rejection.
    */
   async #expire(sandbox: Sandbox, budgetMs: number): Promise<void> {
     if (sandbox.ending !== undefined) return;
     const { removal } = await this.#end(sandbox, `it exceeded its wall-clock limit of ${String(budgetMs)}ms`);
     if (removal !== undefined) {
       sandbox.ended = `it exceeded its wall-clock limit of ${String(budgetMs)}ms and could not be destroyed: ${removal.message}`;
     }
   }
 
   async #remove(containerId: string): Promise<Error | undefined> {
     const removed = await dockerCli(this.#executable, ['rm', '--force', '--volumes', containerId], {
       timeoutMs: LIFECYCLE_TIMEOUT_MS,
     });
     if (removed.exitCode === 0) return undefined;
     return new Error(`docker rm exited ${String(removed.exitCode)} for container ${containerId}: ${removed.stderr.trim()}`);
   }

  return {

async function serving(overrides: Partial<SandboxSpec> = {}): Promise<SandboxHandle> {

async function refusal(body: () => unknown): Promise<SandboxRefusal> {
  try {
    await body();
