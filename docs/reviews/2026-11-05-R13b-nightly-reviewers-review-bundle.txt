R13b: nightly reviewers. Two job templates seat the service's reviewer driver, read-only, under `deny-all` with the model relay as the only route out. `nightly-findings-review` gives a new `findings-reviewer` role the previous UTC day's records: decisions, trigger decisions, refused-delivery counts, and R13a's findings. Each arrives as a runtime id with typed fields and never its text. `nightly-tree-review` seats R9's `security-reviewer` role over the default branch's head, with the tree as its only mount. Each seat's family is checked through `seatReviewer` against every author the day holds or every author merged since the job last completed. At L3 a seat that shares an author's family is refused. Below L3 the job records reduced independence. A reviewer answers in strict JSON. A finding becomes an `intake` candidate only where the runtime can locate it: a file and line the tree holds, under `nightly-tree-finding`, or a record of the day, under `nightly-record-finding`. Any other finding is recorded as refused, and an unreadable answer refuses the job. The reviewer's text is kept as its claim in the nightly record, which gains `review`, and no prompt is ever given it. Neither job writes under any run.

   /** Steps 3 to 6, for a body already read within its cap. */
   async deliver(policy: Policy, subscription: TriggerSubscription, signature: string | undefined, body: Uint8Array): Promise<DeliveryOutcome> {
     return this.#serial(async () => {
       const name = subscription.secret;
       const secret = name.startsWith(WEBHOOK_SECRET_PREFIX) ? (this.#secrets(name) ?? '') : '';
       if (!signatureVerifies(body, signature, secret)) return this.count(subscription.id, 'unverified');
       const digest = bodyDigest(body);
       if (await this.#vault.hasTriggerDecision(subscription.id, digest)) return this.count(subscription.id, 'replayed');
       const now = this.#now();
       const decide = (outcome: TriggerDecision['outcome']): TriggerDecision => ({
         source: subscription.id, key: digest, decidedAt: new Date(now).toISOString(), outcome, collectedBy: 'runtime',
       });
       const write = async (decision: TriggerDecision, candidate: CandidateRecord | null, result: DeliveryOutcome): Promise<DeliveryOutcome> => {
         // A key written between the check and here is the same body sent twice: the second is a replay.
         if (!(await this.#vault.recordTriggerDecision(decision, candidate))) return this.count(subscription.id, 'replayed');
         return result;
       };
       if (!policy.triggers.enabled.includes(subscription.kind)) {
         return write(decide({ kind: 'refused', cause: 'not-enabled' }), null, { outcome: 'refused', cause: 'not-enabled' });
       }
       // The validator requires the age while a non-human kind is enabled; absent, nothing is fresh (I5).
       const maxAge = policy.triggers.maxDeliveryAgeMs ?? 0;
       const extracted = extractGithub(subscription, untrustedBody(body, `github:${subscription.id}`), now, maxAge);
       if (!extracted.ok) {
         if (extracted.outcome === 'ignored') return write(decide({ kind: 'ignored' }), null, { outcome: 'ignored' });
         if (extracted.cause === 'stale') {
           return write(decide({ kind: 'refused', cause: 'stale', signedAt: new Date(extracted.signedAt).toISOString() }), null, { outcome: 'refused', cause: 'stale' });
         }
         return write(decide({ kind: 'refused', cause: 'schema', defects: extracted.defects }), null, { outcome: 'refused', cause: 'schema' });
       }
       if (await this.#overBudget(policy, now)) return write(decide({ kind: 'refused', cause: 'budget' }), null, { outcome: 'refused', cause: 'budget' });
       const template = policy.triggers.taskTemplate[extracted.kind];
       if (template === undefined) return write(decide({ kind: 'refused', cause: 'not-enabled' }), null, { outcome: 'refused', cause: 'not-enabled' });
       const id = intakeId(subscription.id, digest);
       const candidate: CandidateRecord = {
         id,
         kind: extracted.kind,
         template,
         schema: extracted.kind,
         extracted: extracted.extracted,
         authorTrust: extracted.authorTrust,
         lineage: await this.#lineage(extracted.commit, now),
         signedAt: new Date(extracted.signedAt).toISOString(),
         body: { sha256: digest, base64: Buffer.from(body).toString('base64') },
         recordedAt: new Date(now).toISOString(),
         collectedBy: 'runtime',
       };
       return write(decide({ kind: 'candidate', candidate: id }), candidate, { outcome: 'candidate', candidate: id });
     });
   }

   async missed(schedule: TriggerSchedule, first: string, last: string): Promise<void> {

         source: `schedule:${schedule.id}`, key: `missed:${first}:${last}`, decidedAt: new Date(this.#now()).toISOString(),

   async #serial<T>(work: () => Promise<T>): Promise<T> {
     const run = this.#turn.then(work, work);
     this.#turn = run.catch(() => undefined);

   }
 
   /** Candidates recorded across every kind in the sliding window, against the policy's budget (D-R22-07). */
   async #overBudget(policy: Policy, now: number): Promise<boolean> {
     const { runs, windowMs } = policy.triggers.budgetPerWindow;
     const since = now - windowMs;
     let recorded = 0;
     for (const entry of await this.#vault.readTriggerDecisions()) {
       if (entry.candidate !== null && Date.parse(entry.candidate.recordedAt) > since) recorded += 1;
     }
     return recorded >= runs;
   }
 
   /**
    * An event naming a commit an Olympus run merged carries that run's chain
    * plus the run, one deeper. Read from the runtime's own integration and
    * admission records, never from the payload's say-so.
    */
   async #lineage(commit: string | null, now: number): Promise<TriggerLineage> {
     const windowStart = new Date(now).toISOString();
     if (commit === null) return { depth: 0, chain: [], windowStart };
     for (const runId of await this.#vault.listRuns()) {
       const merged = (await readIntegration(this.#vault, runId)).some((r) => r.kind === 'merged' && r.mergeCommit === commit);
       if (!merged) continue;
       const parent = await this.#parentLineage(runId);
       return { depth: parent.depth + 1, chain: [...parent.chain, runId], windowStart };
     }
     return { depth: 0, chain: [], windowStart };
   }
 
   async #parentLineage(runId: RunId): Promise<TriggerLineage> {
     const state = await this.#vault.readRunState(runId);
     const record = JSON.parse(new TextDecoder().decode(await this.#vault.read(state.admission))) as AdmissionRecord;
     return record.run.trigger.lineage;
   }

/**
 * One decision's typed fields. Every cause is listed, so a cause added later
 * fails to compile here rather than reaching the seat unprojected. No
 * message, detail, host, path, or check id is taken: a check id may be one a
 * drafter wrote (R19), and a host is the agent's choice (I7).
 */
function decisionFields(d: EnforcementDecision, violation: IntegrityViolation | null): Record<string, string | number | null> {
  const base = { cause: d.decision.cause, decidedBy: d.decision.decidedBy, taskId: d.taskId, station: d.station };
  const decision = d.decision;
  switch (decision.cause) {
    case 'record-drift': return { ...base, problem: decision.problem };
    case 'admission-refused': return { ...base, reason: decision.refusal.reason };
    case 'resume-refused': return { ...base, reason: decision.refusal.reason };
    case 'station-refused': {
      const refusal = decision.refusal;
      if (refusal.reason === 'gate-failed') {
        return { ...base, reason: refusal.reason, failedChecks: refusal.failed.length, failedCauses: [...new Set(refusal.failed.map((f) => f.cause))].sort().join(',') };
      }
      return { ...base, reason: refusal.reason };
    }
    case 'approval-granted': return { ...base, key: decision.key };
    case 'approval-refused': return { ...base, reason: decision.reason };
    case 'approval-rejected': return { ...base, reason: decision.reason, notes: decision.notes ? 'yes' : 'no' };
    case 'draft-sent-back': return { ...base, sentBackBy: decision.sentBack.by };
    case 'draft-floor-applied': return { ...base, level: decision.level, failedConditions: [...decision.failed].sort().join(',') };
    case 'egress-connection': return { ...base, verdict: decision.connection.verdict };
    case 'violation-recorded': return { ...base, violation: violation?.kind ?? 'unreadable', role: violation?.role ?? null };
    case 'relay-refused': return { ...base, refused: decision.refused, exhausted: decision.exhausted };
    case 'tier-escalated': return { ...base, from: decision.from, to: decision.to, failedGates: decision.failedGates };
    case 'task-conflict': return { ...base, with: decision.with.length, paths: decision.paths.length };
  }
}

/** Every record the runtime made on `day`, each with its id and typed fields alone. */
export async function dayItems(vault: Vault, day: string): Promise<DayItem[]> {
  const items: DayItem[] = [];
  for (const runId of await vault.listRuns()) {
    for (const ref of await vault.readDecisions(runId)) {
      const d = JSON.parse(decode(await vault.read(ref))) as EnforcementDecision;
      if (d.decidedAt.slice(0, 10) !== day) continue;
      let violation: IntegrityViolation | null = null;
      if (d.decision.cause === 'violation-recorded') {
        try {
          violation = JSON.parse(decode(await vault.read(d.decision.violation))) as IntegrityViolation;
        } catch {
          violation = null;
        }
      }
      items.push({ id: `decision:${ref.hash}`, kind: 'decision', runId: d.runId, fields: decisionFields(d, violation) });
    }
  }
  for (const entry of await vault.readTriggerDecisions()) {
    const { decision } = entry;
    if (decision.decidedAt.slice(0, 10) !== day) continue;
    const outcome = decision.outcome;
    const fields: Record<string, string | number | null> = { source: decision.source, outcome: outcome.kind };
    if (outcome.kind === 'refused') {
      fields.cause = outcome.cause;
      if (outcome.cause === 'schema') fields.defects = outcome.defects.length;
    }
    items.push({ id: `trigger:${entry.sha256}`, kind: 'trigger-decision', runId: entry.admittedAs, fields });
  }
  for (const count of await vault.readRefusedDeliveryCounts()) {
    if (!count.hour.startsWith(day)) continue;
    items.push({ id: `deliveries:${count.hour}:${count.bucket}`, kind: 'refused-deliveries', runId: null, fields: { bucket: count.bucket, hour: count.hour, ...count.counts } });
  }
  for (const entry of await vault.readNightly()) {
    if (entry.record.date !== day) continue;
    entry.record.findings.forEach((f, index) => {
      const id = `nightly:${entry.sha256}:${String(index)}`;
      const fields: Record<string, string | number | null> = { template: entry.record.template, finding: f.kind };
      let runId: RunId | null = null;
      switch (f.kind) {
        case 'record-drift': fields.problem = f.problem; runId = f.runId; break;
        case 'rate-drift': fields.rate = f.rate; break;
        case 'security': fields.scanner = f.scanner; fields.rule = f.rule; fields.candidate = 'id' in f.candidate ? 'recorded' : f.candidate.none; break;
        case 'security-unrecordable': fields.scanner = f.scanner; fields.defects = f.defects.length; break;
      }
      items.push({ id, kind: 'nightly-finding', runId, fields });
    });
  }
  return items;
}

/**
 * Every identity that called a model for `runs` under a role that authors,
 * each once (D-R13b-05). The calls are every usage record the store holds for
 * each run, whatever its state references, so a call stored before a crash
 * took its commit is still an author (D-P13-20). A record that is not one the
 * runtime wrote for the run stops the read rather than being skipped (I5),
 * and one written before R9 names no driver, which `seatReviewer` refuses
 * rather than reading as different (A-R9-01).
 */
export async function authorsOf(vault: Vault, runs: Iterable<RunId>): Promise<SeatIdentity[]> {
  const distinct = new Map<string, SeatIdentity>();
  for (const runId of new Set(runs)) {
    for (const ref of await vault.readUsage(runId)) {
      const call = JSON.parse(decode(await vault.read(ref))) as Partial<UsageRecord> | null;
      if (call?.collectedBy !== 'runtime' || call.runId !== runId || call.role === undefined || call.model === undefined) {
        throw new Error(`a usage record of run ${runId} is not one the runtime collected for it`);
      }
      if (JUDGING_ROLES.has(call.role.id)) continue;
      const driver = typeof call.driver === 'string' ? call.driver : null;
      const seat: SeatIdentity = { role: { id: call.role.id, sha256: call.role.sha256 }, driver, model: call.model };
      distinct.set(JSON.stringify([call.role.id, call.role.sha256, driver, call.model.provider, call.model.family, call.model.model, call.model.version]), seat);
    }
  }
  return [...distinct.values()];
}

/**
 * A reviewer's final message, read strictly, as a critic's is
 * (`readCriticResult`, D-R13b-06): one JSON object whose one key is
 * `findings`, a list. A single fenced block is unwrapped. Anything else is
 * unreadable, which is a refusal and never "no findings" (I5).
 */
export function readReviewResult(narrative: string): { readable: true; findings: readonly unknown[] } | { readable: false } {
  const text = narrative.trim().replace(/^```(?:json)?\s*\n([\s\S]*)\n```$/u, '$1').trim();
  let parsed: unknown;
  try {
    parsed = JSON.parse(text);
  } catch {
    return { readable: false };
  }
  if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) return { readable: false };
  const keys = Object.keys(parsed);
  const { findings } = parsed as { findings?: unknown };
  if (keys.length !== 1 || !Array.isArray(findings)) return { readable: false };
  return { readable: true, findings };
}

/** One element's fields, without its note, which stays in the claim alone. A shape that is not an object is all defects. */
function elementFields(element: unknown, keys: readonly string[]): { fields: Record<string, unknown> } | { defects: FieldDefect[] } {
  if (typeof element !== 'object' || element === null || Array.isArray(element)) return { defects: keys.map((field) => ({ field, problem: 'missing' })) };
  const { note, ...fields } = element as Record<string, unknown>;
  if (note !== undefined && typeof note !== 'string') return { defects: [{ field: 'note', problem: 'not-a-string' }] };
  // A line is a number in the result and a decimal string in the schema; any other type is left for the schema to refuse.
  if (typeof fields.line === 'number' && Number.isSafeInteger(fields.line)) fields.line = String(fields.line);
  return { fields };
}

/** The number of lines a file holds, as an editor would number them. */
function lineCount(text: string): number {
  if (text.length === 0) return 0;
  const lines = text.split('\n').length;
  return text.endsWith('\n') ? lines - 1 : lines;
}

interface SeatRequest {
  readonly policy: Policy;
  readonly schedule: TriggerSchedule;
  readonly role: RoleId;
  readonly contract: StationContract;
  readonly kind: 'review' | 'security';
  readonly authors: readonly SeatIdentity[];
  /** The one tree the seat reads, mounted read-only. */
  readonly tree: string;
  readonly context: string;
}

type SeatOutcome =
  | { readonly ok: true; readonly review: Omit<NightlyReview, 'subject' | 'findings'>; readonly result: TaskResult }
  | { readonly ok: false; readonly refusal: NightlyRefusal; readonly review: Omit<NightlyReview, 'subject' | 'findings'> };

const UNSEATED: Omit<NightlyReview, 'subject' | 'findings'> = { role: null, seat: null, usage: null, egress: null, claim: null };

/** The seat graded against every author; `no-author` where there is none, since `seatReviewer` seats only a review of someone (I6). */
function grade(task: TaskId, kind: 'review' | 'security', authors: readonly SeatIdentity[], reviewer: SeatIdentity, level: AutonomyLevel): { ok: true; seat: NightlySeat } | { ok: false; detail: string } {
  if (authors.length === 0) return { ok: true, seat: { reviewer, authors: [], independence: 'no-author', level } };
  let seated;
  try {
    seated = seatReviewer(task, kind, authors, reviewer, level);
  } catch (error) {
    return { ok: false, detail: describe(error) };
  }
  if (!seated.ok) return { ok: false, detail: seated.message };
  return { ok: true, seat: { reviewer, authors: seated.seat.authors, independence: seated.seat.independence, level } };
}

/**
 * Seats the reviewer once. The role is compiled from the service's roles
 * directory and scoped by policy at `review` (I4); the seat is graded before
 * the call from the model the runtime resolved, and after it from the model
 * the result names, keeping the weaker (I6). The sandbox mounts `tree`
 * read-only and nothing else, under `deny-all`; the relay is the one route
 * out, metered at the scope's budget. What the call cost and what the proxy
 * saw are read at teardown whatever the call did.
 */
async function seatOnce(options: ReviewJobOptions, request: SeatRequest): Promise<SeatOutcome> {
  const seatOptions = options.seat;
  if (seatOptions === null) {
    return { ok: false, refusal: { part: 'seat', cause: 'no-reviewer', detail: 'this service names no reviewer driver and roles directory, so no seat can be filled' }, review: UNSEATED };
  }
  const { driver } = seatOptions;
  const compiled = await compileRoles(seatOptions.roles).catch((error: unknown) => ({ ok: false as const, error }));
  const role: CompiledRole | undefined = compiled.ok ? compiled.roles.find((r) => r.role === request.role) : undefined;
  if (role === undefined) {
    return { ok: false, refusal: { part: 'seat', cause: 'role-unavailable', detail: `the role ${request.role} did not compile from the roles directory` }, review: UNSEATED };
  }
  const roleVersion = { id: role.role, sha256: role.sha256 };
  const resolved = engine.resolveCapabilities(request.role, 'review', request.policy);
  if (!resolved.ok) {
    return { ok: false, refusal: { part: 'policy', cause: 'role-unavailable', detail: `policy grants ${request.role} no scope at review: ${resolved.detail}` }, review: { ...UNSEATED, role: roleVersion } };
  }
  const { scope } = resolved;
  const capability = capabilityRefusal(request.contract, driver.capabilities());
  if (capability !== undefined) {
    return { ok: false, refusal: { part: 'seat', cause: 'seat-refused', detail: `the reviewer driver lacks what the seat requires (${capability.reason})` }, review: { ...UNSEATED, role: roleVersion } };
  }
  const level = seatLevel(request.policy);
  const relay = driver.relayRequest();
  // An unmetered call spends what nothing bounds; the line runs none above L1, and neither does a night (I4).
  if (relay === null && level > 1) {
    return { ok: false, refusal: { part: 'seat', cause: 'seat-refused', detail: 'the reviewer driver calls no model through a relay, so its call would be unmetered above L1' }, review: { ...UNSEATED, role: roleVersion } };
  }
  const task = `nightly.${request.schedule.id}` as TaskId;
  const tier = stationTier(scope, 'review');
  let model: ModelIdentity;
  try {
    model = driver.resolveModel(tier);
  } catch (error) {
    return { ok: false, refusal: { part: 'seat', cause: 'seat-failed', detail: `the reviewer driver resolved no model: ${describe(error)}` }, review: { ...UNSEATED, role: roleVersion } };
  }
  const identity = (m: ModelIdentity): SeatIdentity => ({ role: roleVersion, driver: driver.provenanceId(), model: m });
  const planned = grade(task, request.kind, request.authors, identity(model), level);
  if (!planned.ok) return { ok: false, refusal: { part: 'seat', cause: 'seat-refused', detail: planned.detail }, review: { ...UNSEATED, role: roleVersion } };

  const wallClockMs = scope.budget.maxWallClockMs;
  const spec: SandboxSpec = {
    image: seatOptions.image,
    // The tree is the one mount, read-only: the seat writes nothing, and the Vault is mounted nowhere (I1, I4).
    mounts: { workspace: { source: request.tree, target: '/workspace', mode: 'ro' }, others: [] },
    // Whatever egress policy grants the role elsewhere, a nightly seat reaches the relay alone.
    egress: { mode: 'deny-all', allow: [] },
    limits: options.limits === null ? { cpus: 0, memoryMb: 0, pids: 0, wallClockMs } : { ...options.limits, wallClockMs },
    user: { ...options.user },
  };
  let handle: SandboxHandle;
  try {
    handle = await options.sandbox.provision(relay === null ? spec : {
      ...spec,
      relay: { ...relay, meter: boundTo(relay.meter, model.model), budget: { maxTokens: scope.budget.maxTokens, maxCostUsd: scope.budget.maxCostUsd } },
    });
  } catch (error) {
    return { ok: false, refusal: { part: 'seat', cause: 'seat-failed', detail: `the seat's sandbox could not be provisioned: ${describe(error)}` }, review: { ...UNSEATED, role: roleVersion, seat: planned.seat } };
  }
  let called: { ok: true; result: TaskResult } | { ok: false; error: unknown };
  try {
    const result = await driver.runTask({
      taskId: task, role, stablePrefix: request.context, variableSuffix: task, tier, tools: [...scope.tools], sandbox: handle,
      timeoutMs: wallClockMs, budget: { ...scope.budget },
    });
    called = { ok: true, result };
  } catch (error) {
    called = { ok: false, error };
  }
  let teardown: Teardown;
  try {
    teardown = await options.sandbox.destroy(handle);
  } catch (error) {
    return {
      ok: false,
      refusal: { part: 'seat', cause: 'seat-failed', detail: `the seat's sandbox could not be destroyed and read, so what the call cost is unknown: ${describe(error)}` },
      review: { ...UNSEATED, role: roleVersion, seat: planned.seat },
    };
  }
  const read = { role: roleVersion, usage: teardown.meter, egress: teardown.egress };
  if (!called.ok) {
    return { ok: false, refusal: { part: 'seat', cause: 'seat-failed', detail: `the reviewer's call did not return: ${describe(called.error)}` }, review: { ...UNSEATED, ...read, seat: planned.seat } };
  }
  const { result } = called;
  const problems = taskResultProblems(result);
  if (result.taskId !== task || problems.length > 0) {
    return { ok: false, refusal: { part: 'seat', cause: 'result-unreadable', detail: `the reviewer's result does not match the TaskResult contract: ${problems.join('; ') || 'another task'}` }, review: { ...UNSEATED, ...read, seat: planned.seat } };
  }
  const claim = { narrative: result.claim.narrative, filesChanged: [...result.claim.filesChanged] };
  // The model the result names is graded too; it can lower the grade and never raise it, and never becomes the recorded identity (I2, I6).
  const reported = grade(task, request.kind, request.authors, identity(result.model), level);
  if (!reported.ok) return { ok: false, refusal: { part: 'seat', cause: 'seat-refused', detail: reported.detail }, review: { ...UNSEATED, ...read, seat: planned.seat, claim } };
  const independence: IndependenceGrade | 'no-author' = planned.seat.independence === 'no-author' || reported.seat.independence === 'no-author'
    ? planned.seat.independence
    : weakerGrade(planned.seat.independence, reported.seat.independence);
  return { ok: true, review: { ...read, seat: { ...planned.seat, independence }, claim }, result };
}

/** A located finding's candidate, or why it has none; the key is the runtime's hash of its fields. */
async function candidateFor(
  intake: Intake, policy: Policy, schedule: TriggerSchedule, date: string, schema: JobSchema, fields: Readonly<Record<string, string>>,
): Promise<{ readonly id: CandidateId } | { readonly none: 'budget' | 'schema' | 'recorded-before' }> {
  const key = `${date}:${schema}:${sha256(JSON.stringify(Object.entries(fields).sort()))}`;
  return intake.recordFinding(policy, schedule, key, fields, schema);
}

const RESULT_NOTE = 'A finding may carry "note", a string saying what is wrong; it is kept as your claim and read by a person, never by the runtime.';

/** `nightly-findings-review` (R13b): the day before the firing, as typed fields, before a seat graded against the day's authors. */
export async function findingsReview(options: ReviewJobOptions, policy: Policy, schedule: TriggerSchedule, date: string, intake: Intake): Promise<ReviewPart> {
  const day = reviewedDay(date);
  let items: DayItem[];
  try {
    items = await dayItems(options.vault, day);
  } catch (error) {
    return { refusals: [{ part: 'subject', cause: 'store-unreadable', detail: `the records of ${day} could not be read: ${describe(error)}` }], review: empty(null) };
  }
  const subject: ReviewSubject = { kind: 'day', day, items, postMerge: 'not-recorded' };
  // A day with no record has nothing to review; it is recorded as reviewed by no seat, never as a seat that found nothing.
  if (items.length === 0) return { refusals: [], review: empty(subject) };
  let authors: SeatIdentity[];
  try {
    authors = await authorsOf(options.vault, items.flatMap((i) => (i.runId === null ? [] : [i.runId])));
  } catch (error) {
    return { refusals: [{ part: 'subject', cause: 'store-unreadable', detail: `the day's authors could not be read: ${describe(error)}` }], review: empty(subject) };
  }
  const dir = join(options.root, 'nightly', `${schedule.id}-${date}-${randomUUID()}`);
  const tree = join(dir, 'empty');
  await mkdir(tree, { recursive: true });
  const context = [
    `[day]\nThe runtime's records of ${day}, UTC. Each is an id the runtime gave it and its typed fields; nothing a record says in words is here.`,
    JSON.stringify(items),
    `[result]\nYour final message is one JSON object and nothing else: {"findings": [{"category": "<one of ${RECORD_CATEGORIES.join(', ')}>", "record": "<the id of one record above>"}]}. ${RESULT_NOTE} An empty list says you found nothing.`,
  ].join('\n\n');
  let seated: SeatOutcome;
  try {
    seated = await seatOnce(options, { policy, schedule, role: FINDINGS_REVIEWER_ROLE, contract: STATION_CONTRACTS.review, kind: 'review', authors, tree, context });
  } finally {
    await rm(dir, { recursive: true, force: true });
  }
  if (!seated.ok) return { refusals: [seated.refusal], review: { subject, ...seated.review, findings: [] } };
  const read = readReviewResult(seated.result.claim.narrative);
  if (!read.readable) {
    return { refusals: [{ part: 'seat', cause: 'result-unreadable', detail: 'the reviewer returned no result the runtime could read as a list of findings' }], review: { subject, ...seated.review, findings: [] } };
  }
  const ids = new Set(items.map((i) => i.id));
  const findings: ReviewFinding[] = [];
  for (const element of read.findings) {
    const shaped = elementFields(element, ['category', 'record']);
    if ('defects' in shaped) {
      findings.push({ kind: 'refused', cause: 'unrecordable', defects: shaped.defects });
      continue;
    }
    const checked = checkFields('nightly-record-finding', shaped.fields);
    if (!checked.ok) {
      findings.push({ kind: 'refused', cause: 'unrecordable', defects: checked.defects });
      continue;
    }
    // A finding names a record of the day it reviewed, or it is refused (R13b).
    if (!ids.has(checked.fields.record)) {
      findings.push({ kind: 'refused', cause: 'unknown-record', defects: [] });
      continue;
    }
    const category = checked.fields.category as (typeof RECORD_CATEGORIES)[number];
    const candidate = await candidateFor(intake, policy, schedule, date, 'nightly-record-finding', checked.fields);
    findings.push({ kind: 'record', category, record: checked.fields.record, candidate });
  }
  return { refusals: [], review: { subject, ...seated.review, findings } };
}

/**
 * The runs merged since the job's last completed run started, every merged
 * run on its first, and whether the head is a commit no recorded merge
 * produced (D-R13b-05).
 */
async function mergedSince(vault: Vault, schedule: TriggerSchedule, date: string, head: string): Promise<{ merged: RunId[]; since: string; unattributed: boolean }> {
  const last = (await vault.readNightly())
    .filter((e) => e.record.schedule === schedule.id && e.record.template === 'nightly-tree-review' && e.record.outcome.kind === 'completed' && e.record.date < date)
    .at(-1);
  const since = last === undefined ? 'none' : last.record.startedAt;
  const merged: RunId[] = [];
  const commits = new Set<string>();
  for (const runId of await vault.listRuns()) {
    for (const record of await readIntegration(vault, runId)) {
      if (record.kind !== 'merged') continue;
      commits.add(record.mergeCommit);
      if (since === 'none' || record.mergedAt > since) merged.push(runId);
    }
  }
  return { merged: [...new Set(merged)], since, unattributed: !commits.has(head) };
}

/** `nightly-tree-review` (R13b): R9's security reviewer over the head of the default branch, graded against every author merged since its last run. */
export async function treeReview(options: ReviewJobOptions, policy: Policy, schedule: TriggerSchedule, date: string, intake: Intake): Promise<ReviewPart> {
  const { target } = options;
  if (target === null) return { refusals: [{ part: 'target', cause: 'no-target', detail: 'this host names no default branch to review' }], review: empty(null) };
  const dir = join(options.root, 'nightly', `${schedule.id}-${date}-${randomUUID()}`);
  const tree = join(dir, 'tree');
  try {
    let commit: string;
    let files: readonly string[];
    try {
      commit = await resolveCommit(target.checkout, `refs/heads/${target.branch}`);
      await mkdir(tree, { recursive: true });
      files = (await materialize(target.checkout, commit, tree)).files;
    } catch (error) {
      return { refusals: [{ part: 'target', cause: 'target-unreadable', detail: describe(error) }], review: empty(null) };
    }
    let since: Awaited<ReturnType<typeof mergedSince>>;
    let authors: SeatIdentity[];
    try {
      since = await mergedSince(options.vault, schedule, date, commit);
      authors = await authorsOf(options.vault, since.merged);
    } catch (error) {
      return { refusals: [{ part: 'subject', cause: 'store-unreadable', detail: `the merged work and its authors could not be read: ${describe(error)}` }], review: empty(null) };
    }
    const subject: ReviewSubject = { kind: 'tree', branch: target.branch, commit, files: files.length, merged: since.merged, since: since.since, unattributed: since.unattributed };
    const context = [
      `[tree]\nThis is not a change. The working directory holds the whole tree of the default branch ${target.branch} at commit ${commit}, ${String(files.length)} files, read-only. There is no spec, diff, or evidence: ask your two questions of the tree as it stands, and name a file and a line in it for every finding.`,
      `[result]\nYour final message is one JSON object and nothing else: {"findings": [{"category": "<one of ${TREE_CATEGORIES.join(', ')}>", "path": "<a file in the tree, relative to it>", "line": <a line number in that file>}]}. ${RESULT_NOTE} An empty list says you found nothing.`,
    ].join('\n\n');
    const seated = await seatOnce(options, {
      policy, schedule, role: SECURITY_REVIEWER_ROLE, contract: securitySeatContract(STATION_CONTRACTS.review), kind: 'security', authors, tree, context,
    });
    if (!seated.ok) return { refusals: [seated.refusal], review: { subject, ...seated.review, findings: [] } };
    const read = readReviewResult(seated.result.claim.narrative);
    if (!read.readable) {
      return { refusals: [{ part: 'seat', cause: 'result-unreadable', detail: 'the reviewer returned no result the runtime could read as a list of findings' }], review: { subject, ...seated.review, findings: [] } };
    }
    const held = new Set(files);
    const findings: ReviewFinding[] = [];
    for (const element of read.findings) {
      const shaped = elementFields(element, ['category', 'path', 'line']);
      if ('defects' in shaped) {
        findings.push({ kind: 'refused', cause: 'unrecordable', defects: shaped.defects });
        continue;
      }
      const checked = checkFields('nightly-tree-finding', shaped.fields);
      if (!checked.ok) {
        findings.push({ kind: 'refused', cause: 'unrecordable', defects: checked.defects });
        continue;
      }
      // Located in the tree reviewed, or refused: a file it holds, and a line that file has (R13b).
      const line = Number(checked.fields.line);
      const path = checked.fields.path;
      const located = held.has(path) && line <= lineCount(await readFile(join(tree, path), 'utf8'));
      if (!located) {
        findings.push({ kind: 'refused', cause: 'unlocated', defects: [] });
        continue;
      }
      const category = checked.fields.category as (typeof TREE_CATEGORIES)[number];
      const candidate = await candidateFor(intake, policy, schedule, date, 'nightly-tree-finding', checked.fields);
      findings.push({ kind: 'tree', category, path, line, candidate });
    }
    return { refusals: [], review: { subject, ...seated.review, findings } };
  } finally {
    await rm(dir, { recursive: true, force: true });
  }
}

/** Runs a reviewer job; the job's id names which. */
export async function reviewJob(
  template: ReviewJobTemplateId, options: ReviewJobOptions, policy: Policy, schedule: TriggerSchedule, date: string, intake: Intake,
): Promise<ReviewPart> {
  return template === 'nightly-findings-review'
    ? findingsReview(options, policy, schedule, date, intake)
    : treeReview(options, policy, schedule, date, intake);
}

# Findings reviewer

You review one day of the runtime's records: what its controls refused,
parked, found, and counted. The message you are given is the id of the
nightly job.

## What you work from

Your context holds every record the runtime made on one day, each with an id
the runtime gave it and its typed fields: causes, reasons, counts, stations,
runs, and tasks. Nothing a record says in words is there, and nothing anyone
outside the runtime wrote. The working directory is empty and read-only:
you cannot change anything, and you are not asked to.

## What to ask of the day

- Does one refusal, park, or failed check recur across runs or tasks, so that
  a control keeps meeting the same thing?
- Does a violation, a drifted record, or a rate that moved say something a
  person should look at?
- Do refused deliveries or a scanner's findings cluster on one source?
- Does the day show a control that should have acted and did not, a gap?

## What to say

Your final message is your review, in the form the context asks for. It is
recorded for a person to read; it does not decide anything, and nothing you
say changes a run, a grant, or a status. Every finding names one record by
its id. If you found nothing, say so with an empty list. Do not soften a
finding and do not invent one.

async function reviewPolicy(level: AutonomyLevel): Promise<Policy> {
  return linePolicy({}, {
    roles: {
      ['builder' as RoleId]: lineScope(['build', 'verify'], 'standard'),
      ['reviewer' as RoleId]: lineScope(['review'], 'deep'),
      ['security-reviewer' as RoleId]: lineScope(['review'], 'deep'),
      ['findings-reviewer' as RoleId]: lineScope(['review'], 'deep'),
    },
    triggers: {
      enabled: ['human', 'scheduled'],
      entryStation: { human: 'intake', scheduled: 'intake' },
      taskTemplate: { scheduled: 'intake-candidate' },
      maxAutonomy: { human: 2, scheduled: level },
      minAuthorTrust: { human: 'owner', scheduled: 'owner' },
      maxTriggerDepth: 2,
      budgetPerWindow: { runs: 20, windowMs: 3_600_000 },
      schedules: [FINDINGS, TREE],
      maxDeliveryAgeMs: 3_600_000,
      candidateExpiryMs: 86_400_000,
    },
  });
}

function model(family: string): ModelIdentity {
  return { provider: family, family: family as ModelFamily, model: `${family}-model`, version: '1' };
}

/** A reviewer driver of a chosen family, which records every request and answers each with `narrative`. */
class ScriptedDriver implements Driver {
  readonly id = 'scripted';
  readonly contractVersion = '1';
  readonly requests: TaskRequest[] = [];
  constructor(private readonly family: string, public narrative = '{"findings":[]}') {}
  provenanceId(): string {
    return `scripted-${this.family}@1`;
  }
  capabilities(): DriverCapabilities {
    return { subagents: false, hooks: false, mcp: false, parallelism: 1, computerUse: false, steering: false, stablePrefixCaching: false };
  }
  declaredTools(): readonly string[] {
    return ['read'];
  }
  resolveModel(_tier: ModelTier): ModelIdentity {
    return model(this.family);
  }
  relayRequest(): Omit<RelaySpec, 'budget'> {
    const price = { inputPerMTok: 1, outputPerMTok: 10, cacheReadPerMTok: 0.1, cacheWritePerMTok: 1.25, cacheWrite1hPerMTok: 2 };
    return {
      upstream: 'https://api.example.invalid', paths: ['/v1/messages'], header: 'x-api-key', scheme: 'raw', credential: 'k', urlVariable: 'MODEL_BASE_URL',
      meter: { dialect: 'anthropic-messages', prices: { [model(this.family).model]: price } },
    };
  }
  runTask(req: TaskRequest): Promise<TaskResult> {
    this.requests.push(req);
    return Promise.resolve({
      taskId: req.taskId, claim: { narrative: this.narrative, filesChanged: [] }, events: [],
      usage: { inputTokens: 0, outputTokens: 0, cacheReadTokens: 0, cacheWriteTokens: 0, costUsd: 0, wallClockMs: 0 },
      model: this.resolveModel(req.tier), contractVersion: this.contractVersion,
    });
  }
  cancel(_taskId: TaskId): Promise<void> {
    return Promise.resolve();
  }
  emitArtifacts(_roles: CompiledRole[], _targetDir: string): Promise<void> {
    return Promise.resolve();
  }
  on(_kind: DriverEvent['kind'], _handler: (e: DriverEvent) => void): void {
    // Nothing fires.
  }
}

/** Records every sandbox asked for, with what its one mount held when it was provisioned. */
class RecordingProvider implements SandboxProvider {
  readonly id = 'recording';
  readonly provisioned: Array<{ readonly spec: SandboxSpec; readonly listed: readonly string[] }> = [];
  async provision(spec: SandboxSpec): Promise<SandboxHandle> {
    this.provisioned.push({ spec, listed: (await readdir(spec.mounts.workspace.source)).sort() });
    return `h${String(this.provisioned.length)}` as SandboxHandle;
  }
  exec(): Promise<ExecResult> {
    return Promise.reject(new Error('a reviewer seat runs no command of the runtime\'s'));
  }
  destroy(): Promise<Teardown> {
    return Promise.resolve({ meter: { kind: 'unmetered' }, egress: { kind: 'none' } });
  }
  capabilities() {
    return { computerUse: false, gpu: false, os: 'linux' as const, persistent: false, remote: false };
  }
}

/** A git checkout whose `main` holds three files; returns the head commit. */
async function checkout(dir: string): Promise<string> {
  await mkdir(dir, { recursive: true });
  const git = (...args: string[]) => execFileSync('git', ['-C', dir, '-c', 'user.email=r13b@example.invalid', '-c', 'user.name=r13b', '-c', 'core.autocrlf=false', ...args], { stdio: 'pipe' });
  git('init', '-q', '-b', 'main');
  await writeFile(join(dir, 'package.json'), '{ "name": "target" }\n');
  await writeFile(join(dir, 'index.js'), 'const a = 1;\nconst b = 2;\nmodule.exports = a + b;\n');
  await writeFile(join(dir, 'README.md'), '# target\n');
  git('add', '.');
  git('commit', '-q', '-m', 'target');
  return git('rev-parse', 'HEAD').toString().trim();
}

interface Rig {
  readonly vault: Vault;
  readonly intake: Intake;
  readonly jobs: NightlyJobs;
  readonly driver: ScriptedDriver;
  readonly sandbox: RecordingProvider;
  readonly head: string;
  /** Moves the runtime's clock. */
  readonly at: (iso: string) => void;
}

async function withReview<T>(prefix: string, family: string, body: (rig: Rig) => Promise<T>): Promise<T> {
  const { Intake: IntakeClass, NightlyJobs: Jobs, SCANNERS } = await api();
  const { DEFAULT_ROLES_DIRECTORY } = await import('@olympus-ai/compiler');
  const { LocalVault } = await import('@olympus-ai/vault');
  return withVaultDirs(prefix, async (dirs, base) => {
    const vault = new LocalVault(dirs);
    const target = join(base, 'target');
    const head = await checkout(target);
    const sandbox = new RecordingProvider();
    const driver = new ScriptedDriver(family);
    let clock = Date.parse(`${FIRED}T04:00:00.000Z`);
    const now = () => clock;
    const jobs = new Jobs({
      vault, sandbox, workspaces: { root: join(base, 'trees'), user: { uid: 1000, gid: 1000 } }, limits: null, scanners: SCANNERS,
      target: { checkout: target, branch: 'main' }, seat: { driver, roles: DEFAULT_ROLES_DIRECTORY, image: 'none' }, now,
    });
    const intake = new IntakeClass({ vault, secrets: () => undefined, jobs, now });
    return body({ vault, intake, jobs, driver, sandbox, head, at: (iso) => { clock = Date.parse(iso); } });
  });
}

/** One call a builder of `family` made for the run, and one its reviewer made, which is no author's (D-R13b-05). */
async function seedRun(vault: Vault, head: string, family = 'claude'): Promise<void> {
  const call = (taskId: string, station: 'build' | 'review', role: string, f: string) => vault.recordUsage({
    runId: RUN, taskId: taskId as TaskId, station, attempt: 1, model: model(f), role: { id: role as RoleId, sha256: ROLE_HASH }, driver: `scripted-${f}@1`,
    reading: { kind: 'unmetered' }, timing: { startedAt: `${DAY}T09:00:00.000Z`, endedAt: `${DAY}T09:01:00.000Z` }, collectedBy: 'runtime',
  });
  await call('build-1', 'build', 'builder', family);
  await call('review-1', 'review', 'reviewer', 'judge');
  await vault.recordIntegration({
    runId: RUN, kind: 'merged', repository: 'owner/target', baseBranch: 'main', baseCommit: head, branch: 'factory/run', commit: head,
    pullRequest: 1, url: 'https://example.invalid/pull/1', mergeCommit: head, mergedAt: `${DAY}T09:30:00.000Z`, collectedBy: 'runtime',
  });
  // A decision of the reviewed day whose host is the agent's text (I7).
  await vault.recordDecision({
    runId: RUN, taskId: 'build-1' as TaskId, station: 'build', decidedAt: `${DAY}T10:00:00.000Z`, occurrence: 'r13b-seed',
    decision: { cause: 'egress-connection', decidedBy: 'egress-proxy', connection: { verdict: 'refused', host: `${MARKER}.example`, at: `${DAY}T10:00:00.000Z` } },
    collectedBy: 'runtime',
  });
}

async function recordOn(vault: Vault, schedule: string, date: string): Promise<NightlyRecord> {
  const entry = (await vault.readNightly()).find((e) => e.record.schedule === schedule && e.record.date === date);
  if (entry === undefined) throw new Error(`R13b: no nightly record for ${schedule} on ${date}`);
  return entry.record;
}

function prompt(req: TaskRequest): string {
  return `${req.stablePrefix}\n${req.variableSuffix}`;
}

const FAMILY_DIFFERS = runtime({
  id: 'I6.nightly-reviewer-family-differs-from-every-author',
  title: "each nightly reviewer's family is checked against every author's through seatReviewer: where the policy lets scheduled work reach L3, a reviewer sharing an author's family is refused before it is called, and one of another family is seated and graded different-family; a judging seat's call is no author",
  run: async () => {
    for (const schedule of [FINDINGS, TREE]) {
      await withReview('r13b-family-same-', 'claude', async ({ vault, intake, driver, head }) => {
        await seedRun(vault, head);
        await intake.fire(await reviewPolicy(3), schedule, FIRED);
        const record = await recordOn(vault, schedule.id, FIRED);
        if (record.outcome.kind !== 'refused' || !record.outcome.refusals.some((r) => r.cause === 'seat-refused')) {
          throw new Error(`I6: ${schedule.template} seated a reviewer of its author's family at L3: ${JSON.stringify(record.outcome)}`);
        }
        if (driver.requests.length !== 0) throw new Error(`I6: ${schedule.template} called a refused seat`);
      });
      await withReview('r13b-family-other-', 'gpt', async ({ vault, intake, driver, head }) => {
        await seedRun(vault, head);
        await intake.fire(await reviewPolicy(3), schedule, FIRED);
        const record = await recordOn(vault, schedule.id, FIRED);
        const seat = record.review?.seat;
        if (record.outcome.kind !== 'completed' || seat?.independence !== 'different-family' || driver.requests.length !== 1) {
          throw new Error(`I6: ${schedule.template} did not seat a reviewer of another family: ${JSON.stringify(record.outcome)} ${JSON.stringify(seat)}`);
        }
        if (seat.authors.length !== 1 || seat.authors[0]?.model.family !== 'claude') throw new Error(`I6: the authors are not the builder alone: ${JSON.stringify(seat.authors)}`);
      });
    }
  },
});

const TYPED_FIELDS_ONLY = runtime({
  id: 'I7.nightly-reviewer-gets-typed-fields-only',
  title: "the findings reviewer is given the day's records as runtime ids and typed fields: a host the agent chose, a violation's detail, and every other text a record carries never reach its prompt",
  run: async () => {
    await withReview('r13b-typed-', 'gpt', async ({ vault, intake, driver, head }) => {
      await seedRun(vault, head);
      const violation = await vault.recordViolation({
        runId: RUN, taskId: 'build-1' as TaskId, kind: 'prompt-injection', role: 'builder' as RoleId, driverProvenanceId: 'scripted-claude@1', contractVersion: '1',
        detectedAt: `${DAY}T11:00:00.000Z`, detail: { note: MARKER },
      });
      await vault.recordDecision({
        runId: RUN, taskId: 'build-1' as TaskId, station: 'build', decidedAt: `${DAY}T11:00:00.000Z`, occurrence: 'r13b-violation',
        decision: { cause: 'violation-recorded', decidedBy: 'line', violation }, collectedBy: 'runtime',
      });
      await intake.fire(await reviewPolicy(1), FINDINGS, FIRED);
      const [request] = driver.requests;
      if (request === undefined) throw new Error('I7: the findings reviewer was never seated over a day that held records');
      const text = prompt(request);
      if (text.includes(MARKER)) throw new Error("I7: a record's text reached the findings reviewer's prompt");
      if (!text.includes('egress-connection') || !text.includes('prompt-injection')) throw new Error("I7: the day's records are not in the prompt as typed fields");
      const record = await recordOn(vault, FINDINGS.id, FIRED);
      if (record.review?.subject?.kind !== 'day' || JSON.stringify(record.review.subject).includes(MARKER)) throw new Error('I7: the recorded subject holds text a record carried');
    });
  },
});

const CLAIM_NEVER_PROMPTED = runtime({
  id: 'I7.nightly-claim-never-enters-a-prompt',

const UNLOCATED_REFUSED = runtime({
  id: 'I5.nightly-unlocated-finding-is-refused',
  title: 'a whole-tree finding naming a file the tree does not hold, a line its file does not have, or a category outside the closed list, and a findings-review finding naming no record of the day, is refused and recorded as refused, never a candidate; an unreadable result refuses the job',
  run: async () => {
    await withReview('r13b-unlocated-', 'gpt', async ({ vault, intake, driver, head }) => {
      await seedRun(vault, head);
      driver.narrative = JSON.stringify({
        findings: [
          { category: 'unchecked-input', path: 'index.js', line: 3 },
          { category: 'unchecked-input', path: 'missing.js', line: 1 },
          { category: 'unchecked-input', path: 'index.js', line: 4 },
          { category: 'anything-at-all', path: 'index.js', line: 1 },
        ],
      });
      await intake.fire(await reviewPolicy(1), TREE, FIRED);
      const tree = await recordOn(vault, TREE.id, FIRED);
      const kinds = tree.review?.findings.map((f) => (f.kind === 'refused' ? f.cause : f.kind)).join(',');

      if (candidates.length !== 1 || candidates[0]?.candidate?.schema !== 'nightly-tree-finding') throw new Error(`I5: ${String(candidates.length)} candidates for one located finding`);

      driver.narrative = JSON.stringify({ findings: [{ category: 'violation', record: `decision:${'0'.repeat(64)}` }] });

const REVIEW_CHANGES_NO_RUN = runtime({
  id: 'I2.nightly-review-changes-no-run',
  title: "neither nightly reviewer writes under any run: every run's records are the same objects after both jobs ran and recorded findings as before",
  run: async () => {
    await withReview('r13b-no-run-', 'gpt', async ({ vault, intake, driver, head }) => {
      await seedRun(vault, head);
      const runs = async () => {
        const out: string[] = [];
        for await (const object of vault.listObjects()) if (object.path.startsWith('runs/')) out.push(object.path);
        return out.sort().join('\n');
      };
      const before = await runs();
      driver.narrative = JSON.stringify({ findings: [{ category: 'unchecked-input', path: 'index.js', line: 1 }] });
      await intake.fire(await reviewPolicy(1), TREE, FIRED);
      const decision = (await vault.readDecisions(RUN))[0];
      driver.narrative = JSON.stringify({ findings: [{ category: 'violation', record: `decision:${decision?.hash ?? ''}` }] });
      await intake.fire(await reviewPolicy(1), FINDINGS, FIRED);
      const findings = await recordOn(vault, FINDINGS.id, FIRED);
      if (!findings.review?.findings.some((f) => f.kind === 'record')) throw new Error('I2: the findings review recorded no finding to test against');
      if ((await runs()) !== before) throw new Error('I2: a nightly reviewer wrote under a run');
    });
  },
});

    * file nor a directory is not a record this store writes, and stops the

   async *listObjects(): AsyncIterable<StoredObject> {
     const store = this.#store;
     async function* walk(dir: string, rel: readonly string[]): AsyncGenerator<StoredObject> {
       let entries: Dirent[];
       try {
         entries = await readdir(dir, { withFileTypes: true });
       } catch (error) {
         if (isAbsent(error) && rel.length === 0) return;
         throw error;
       }
       entries.sort((a, b) => (a.name < b.name ? -1 : a.name > b.name ? 1 : 0));
       for (const entry of entries) {
         const parts = [...rel, entry.name];
         const full = join(dir, entry.name);
         if (entry.isDirectory()) {
           yield* walk(full, parts);
           continue;

         if (!entry.isFile()) throw new Error(`LocalVault: ${parts.join('/')} in the store is neither a file nor a directory`);

     }
     yield* walk(store, []);

     if (version === null) throw Object.assign(new Error(`LocalVault: no run state for ${runId}`), { code: 'no-run-state' as const, runId }) satisfies NoRunState;
     const file = join(states, `${String(version)}.json`);
     return requireRunState(await readJson(file), file);
   }
 
   /**
    * Compare-and-swap, arbitrated by the filesystem. The next version's file is
    * created with `wx`, so of any number of writers holding the same
    * `ifVersion` exactly one creates it and the rest are refused with `EEXIST`.
    * There is deliberately no read-then-compare here: that shape loses an
    * update whenever two writers interleave between the read and the write, and
    * it passes a sequential test while doing so.
    *
    * `ifVersion` is authoritative and `s.version` is ignored; the contract left
    * the choice to this unit (D-S1, S1 spec section 2). The precondition below
    * is an extra refusal, not the concurrency control: it rejects a version
    * that never existed, so a caller cannot jump the run forward and leave a
    * hole in its history.
    */
   async commitRunState(s: RunState, ifVersion: string): Promise<RunState> {
     if (!VERSION.test(ifVersion)) {
       throw new Error(`LocalVault: '${ifVersion}' is not a run state version`);
     }
     const states = join(runDir(this.#store, s.runId), 'state');
     const current = Number(ifVersion);
     if (!Number.isSafeInteger(current)) throw new Error(`LocalVault: run state version ${ifVersion} is out of range`);
 
     const held = await highestIn(states);
     if (current === 0 ? held !== null : held === null || held < current) {
       const at = held === null ? 'no committed state' : `version ${String(held)}`;
       throw new Error(`LocalVault: run ${s.runId} has ${at}, so version ${ifVersion} was never current; nothing stored`);
     }
 
     // The history it replaces, read before anything is written: a commit may extend it and never shorten or rewrite it (codex-5).
     const previous = current === 0 ? [] : requireRunState(await readJson(join(states, `${ifVersion}.json`)), `${ifVersion}.json`).stations;
     const problem = stationsProblem(previous, (s as { stations?: unknown }).stations);
     if (problem !== undefined) throw new Error(`LocalVault: refusing run ${s.runId}'s state: ${problem}; nothing stored`);
 
     const next = String(current + 1);
     const stored: RunState = { ...s, version: next };
     if (!(await createExclusive(join(states, `${next}.json`), encode(stored)))) {
       throw new Error(
         `LocalVault: run ${s.runId} has already moved past version ${ifVersion}; another writer committed ${next} first. ` +
           'Nothing was stored. Re-read the state and commit again.',
       );
     }
     return stored;
   }
 }

   /**

    * leaves which of the two wins to P1 (docs/decisions.md, D-S1).
    */
   commitRunState(s: RunState, ifVersion: string): Promise<RunState> {
     const current = this.states.get(s.runId)?.version ?? '0';
     if (ifVersion !== current) {
       return Promise.reject(new Error(`StubVault: run ${s.runId} is at version ${current}, not ${ifVersion}; nothing stored`));
     }
     const stored = structuredClone({ ...s, version: String(Number(current) + 1) });
     this.states.set(s.runId, stored);
     this.runs.add(s.runId);
     return Promise.resolve(structuredClone(stored));
   }

       | IntentRecord | RubricRecord | DraftRecord | CriticRecord | DraftLockRecord,
   ): VaultRef {
     const bytes = new TextEncoder().encode(JSON.stringify(record));
     const ref: VaultRef = { runId, kind, hash: sha256(bytes) };
     this.blobs.set(refKey(ref), bytes);
     this.runs.add(runId);
     return ref;
   }

     try {
       return sha256(await readFile(resolve(this.root, path)));
     } catch (error) {
       if (isMissingFile(error)) return undefined;
       throw error;
     }
   }
