R6: a driver for any OpenAI-compatible endpoint (D-R6-01 to D-R6-07,
A-R6-01 to A-R6-04).

- driver-openai-compatible (new): `OpenAICompatibleDriver` runs the agent loop
  on the host and every model call and tool call inside the sandbox through
  `provider.exec`. The operator's configuration — origin, path, header,
  scheme, provider, family, version, a model per tier, prices, the output
  ceiling, parallelism, and the turn bound — is required in full and checked
  by `checkConfig`. Tools are `Bash`, `Edit`, `Read`, `Write`. A response
  naming another model, or none, or reporting no usage, refuses the task.
  Every feature capability is false; `parallelism` is configured.
- sandbox: `RelaySpec.scheme` (`'raw' | 'bearer'`) is required, and the relay
  reads a second meter dialect, `'openai-chat'`.
- driver-claude-code: its relay declares `scheme: 'raw'`.
- api: `buildGraph` attests an `OpenAICompatibleDriver` in every driver seat.
  `factory-host` reads `FACTORY_ENDPOINT_CONFIG` and `FACTORY_ENDPOINT_KEY`
  to seat it; `parseEndpointFile` and `seatedDrivers` are new.
- conformance: a second assertion under each `driver.*` claim, four external
  invariant assertions in the new package, and the relay's budget and
  fail-closed assertions under `openai-chat`. A claim assertion id may carry
  an implementation suffix, `driver.<key>.<name>`.

===== packages/drivers/openai-compatible/src/call.ts =====
/**
 * The one program the driver runs inside the sandbox to make a model call
 * (D-R6-01).
 *
 * It reads a request body on stdin, posts it to the relay at the address the
 * provider set in `RELAY_URL_VARIABLE`, and prints the status and the body as
 * one JSON line. It holds no credential: the relay discards whatever
 * authentication a client sends and writes its own (P12). It sets no timeout:
 * a long generation is bounded by the sandbox's wall clock and the relay's
 * budget, not by a client default that would cut a call the relay still
 * charges.
 *
 * Passed to `node --eval` as one argv element, with the path as the next, so
 * nothing is written to the workspace, which the agent can write (I3).
 */

/** The variable the provider sets in the sandbox to the relay's address. */
export const RELAY_URL_VARIABLE = 'MODEL_RELAY_URL';

export const CALL_SOURCE = `'use strict';
const http = require('node:http');
const base = new URL(String(process.env.${RELAY_URL_VARIABLE} || ''));
const path = String(process.argv[1] || '');
const parts = [];
process.stdin.on('data', function (c) { parts.push(c); });
process.stdin.on('end', function () {
  const body = Buffer.concat(parts);
  const q = http.request({
    host: base.hostname, port: base.port, method: 'POST', path: path, agent: false,
    headers: { 'content-type': 'application/json', 'content-length': String(body.length) },
  }, function (s) {
    const out = [];
    s.on('data', function (c) { out.push(c); });
    s.on('end', function () {
      process.stdout.write(JSON.stringify({ status: s.statusCode, body: Buffer.concat(out).toString('utf8') }));
    });
  });
  q.on('error', function (e) { process.stderr.write('model call failed: ' + String(e && e.message || e)); process.exit(2); });
  q.end(body);
});
`;

===== packages/drivers/openai-compatible/src/config.ts =====
/**
 * The operator's configuration of one endpoint, and the one function that
 * accepts it.
 *
 * Every field is required and nothing is inferred (D-R6-02). The family above
 * all: it is what reviewer independence is graded on (I6), and the driver has
 * no way to see one, so it is the operator's statement or it is nothing. A
 * default here would be a family nobody declared, a price nobody stated, or a
 * bound nobody chose.
 */
import type { ModelFamily, ModelTier } from '@olympus-ai/core';
import type { ModelPrice, RelayScheme } from '@olympus-ai/sandbox';
import { refuse } from './refusal.js';

/** The request field a call's output ceiling is sent in. Newer OpenAI models take only the second. */
export type CeilingField = 'max_tokens' | 'max_completion_tokens';

export interface EndpointConfig {
  /** An `https` origin with no path: `'https://api.example.com'`. The relay's only upstream. */
  readonly upstream: string;
  /** The chat-completions path, ending `/chat/completions`: `'/v1/chat/completions'`. The relay's only grant. */
  readonly path: string;
  /** The header the endpoint reads its credential from, such as `authorization` or `api-key`. */
  readonly header: string;
  /** How the credential is written into it (A-R6-02). */
  readonly scheme: RelayScheme;
  /** Who serves the model, as the run records it: `'openai'`, `'together'`, `'self-hosted'`. */
  readonly provider: string;
  /** The model family, as the operator declares it (I6). Lower-case letters, digits, `.`, `_`, `-`. */
  readonly family: string;
  /** A label for what the endpoint serves, recorded on every result: a date, a deployment, a server version. */
  readonly version: string;
  /** The exact model id each tier sends, never an alias (D-R6-03, D-R14-01). */
  readonly models: Readonly<Record<ModelTier, string>>;
  /** A price for every model in `models`, in US dollars per million tokens. Zero is a price; absent is not. */
  readonly prices: Readonly<Record<string, ModelPrice>>;
  /** The most output one call may generate, sent on every request in `ceilingField`. */
  readonly maxOutputTokens: number;
  readonly ceilingField: CeilingField;
  /** The most tasks this driver runs at once, each on its own sandbox (A-R4-04). */
  readonly parallelism: number;
  /** The most model calls one task may make before it is refused. */
  readonly maxTurns: number;
}

/** The configuration as the driver holds it: checked, frozen, and with the family branded once. */
export interface CheckedConfig extends Omit<EndpointConfig, 'family'> {
  readonly family: ModelFamily;
}

const TIERS: readonly ModelTier[] = ['fast', 'standard', 'deep'];
const PRICE_FIELDS: ReadonlyArray<keyof ModelPrice> = ['inputPerMTok', 'outputPerMTok', 'cacheReadPerMTok', 'cacheWritePerMTok', 'cacheWrite1hPerMTok'];
const FIELDS: ReadonlyArray<keyof EndpointConfig> = [
  'upstream', 'path', 'header', 'scheme', 'provider', 'family', 'version', 'models', 'prices',
  'maxOutputTokens', 'ceilingField', 'parallelism', 'maxTurns',
];

/** A path as the relay matches one: plain segments, no dot segment, no encoding, no trailing slash. */
const PATH = /^(?:\/[A-Za-z0-9_~-][A-Za-z0-9._~-]*)+$/u;
/** An HTTP header name (RFC 9110 token). */
const HEADER = /^[A-Za-z0-9!#$%&'*+.^_`|~-]+$/u;
const FAMILY = /^[a-z0-9][a-z0-9._-]*$/u;

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function shown(value: unknown): string {
  return value === undefined ? 'undefined' : JSON.stringify(value);
}

function text(config: Record<string, unknown>, field: keyof EndpointConfig): string {
  const value = config[field];
  if (typeof value !== 'string' || value.trim() === '' || value !== value.trim()) {
    refuse('configuration', `${field} must be a non-empty string with no surrounding space, not ${shown(value)}`);
  }
  return value;
}

function count(config: Record<string, unknown>, field: keyof EndpointConfig): number {
  const value = config[field];
  if (typeof value !== 'number' || !Number.isInteger(value) || value <= 0) {
    refuse('configuration', `${field} must be a positive integer, not ${shown(value)}`);
  }
  return value;
}

/**
 * Accepts a configuration exactly as written or refuses it naming the field
 * (I5). Typed input is checked as strictly as parsed JSON: a caller arriving
 * from untyped code, or a file, gets the same answer.
 */
export function checkConfig(input: unknown): CheckedConfig {
  if (!isRecord(input)) refuse('configuration', 'the endpoint configuration is not an object');
  const unknown = Object.keys(input).filter((key) => !(FIELDS as readonly string[]).includes(key));
  if (unknown.length > 0) {
    refuse('configuration', `the endpoint configuration holds ${unknown.map((k) => `'${k}'`).join(', ')}, which the driver does not read; a field it ignored would be a setting that did nothing`);
  }

  const upstream = text(input, 'upstream');
  let origin: URL | undefined;
  try {
    origin = new URL(upstream);
  } catch {
    origin = undefined;
  }
  if (origin?.protocol !== 'https:' || origin.origin !== upstream) {
    refuse('configuration', `upstream must be an https origin with no path, query, credentials, or trailing slash, not ${shown(upstream)}. A local endpoint over http is R21's.`);
  }

  const path = text(input, 'path');
  if (!PATH.test(path) || !path.endsWith('/chat/completions')) {
    refuse('configuration', `path must be a plain path ending /chat/completions, such as '/v1/chat/completions', not ${shown(path)}`);
  }

  const header = text(input, 'header');
  if (!HEADER.test(header)) refuse('configuration', `header ${shown(header)} is not an HTTP header name`);

  const scheme = input.scheme;
  if (scheme !== 'raw' && scheme !== 'bearer') refuse('configuration', `scheme must be 'raw' or 'bearer', not ${shown(scheme)}`);

  const provider = text(input, 'provider');
  const family = text(input, 'family');
  if (!FAMILY.test(family)) {
    refuse('configuration', `family ${shown(family)} must be lower-case letters, digits, '.', '_', and '-'; it is the operator's declaration and is compared as written (I6)`);
  }
  const version = text(input, 'version');

  const models = input.models;
  if (!isRecord(models)) refuse('configuration', 'models must map every tier to a model id');
  const extraTiers = Object.keys(models).filter((tier) => !(TIERS as readonly string[]).includes(tier));
  if (extraTiers.length > 0) refuse('configuration', `models names ${extraTiers.map((t) => `'${t}'`).join(', ')}, which is not a tier`);
  const tiers: Partial<Record<ModelTier, string>> = {};
  for (const tier of TIERS) {
    const model = models[tier];
    if (typeof model !== 'string' || model.trim() === '' || model !== model.trim()) {
      refuse('configuration', `models.${tier} must be the exact model id that tier sends, not ${shown(model)}`);
    }
    tiers[tier] = model;
  }

  const prices = input.prices;
  if (!isRecord(prices)) refuse('configuration', 'prices must give a price for every model');
  const table: Record<string, ModelPrice> = {};
  for (const [model, price] of Object.entries(prices)) {
    if (!isRecord(price)) refuse('configuration', `prices[${shown(model)}] is not a price`);
    const extra = Object.keys(price).filter((key) => !(PRICE_FIELDS as readonly string[]).includes(key));
    if (extra.length > 0) refuse('configuration', `prices[${shown(model)}] holds ${extra.join(', ')}, which is not a price field`);
    const at = (field: keyof ModelPrice): number => {
      const value = price[field];
      if (typeof value !== 'number' || !Number.isFinite(value) || value < 0) {
        refuse('configuration', `prices[${shown(model)}].${field} must be a non-negative finite number, not ${shown(value)}`);
      }
      return value;
    };
    table[model] = Object.freeze({
      inputPerMTok: at('inputPerMTok'),
      outputPerMTok: at('outputPerMTok'),
      cacheReadPerMTok: at('cacheReadPerMTok'),
      cacheWritePerMTok: at('cacheWritePerMTok'),
      cacheWrite1hPerMTok: at('cacheWrite1hPerMTok'),
    });
  }
  for (const tier of TIERS) {
    const model = tiers[tier] ?? '';
    if (!Object.hasOwn(table, model)) {
      refuse('configuration', `models.${tier} is ${shown(model)}, which prices does not name; a call whose cost cannot be counted is not made (I5)`);
    }
  }

  const ceilingField = input.ceilingField;
  if (ceilingField !== 'max_tokens' && ceilingField !== 'max_completion_tokens') {
    refuse('configuration', `ceilingField must be 'max_tokens' or 'max_completion_tokens', not ${shown(ceilingField)}`);
  }

  return Object.freeze({
    upstream,
    path,
    header,
    scheme,
    provider,
    // I6: the one place the family is branded. It is the operator's string, checked for shape and nothing else.
    family: family as ModelFamily,
    version,
    models: Object.freeze(tiers as Record<ModelTier, string>),
    prices: Object.freeze(table),
    maxOutputTokens: count(input, 'maxOutputTokens'),
    ceilingField,
    parallelism: count(input, 'parallelism'),
    maxTurns: count(input, 'maxTurns'),
  });
}

===== packages/drivers/openai-compatible/src/driver.ts =====
/**
 * The `Driver` contract over any endpoint that speaks the OpenAI-compatible
 * chat-completions API (R6).
 *
 * The endpoint is a model and nothing more; the agent loop it lacks is this
 * driver's, and it runs in the runtime's process (D-R6-01). What reaches the
 * sandbox is narrow and fixed:
 *
 * - Each model turn is one `provider.exec` of `CALL_SOURCE`, which posts the
 *   request to the model relay. The relay holds the credential the sandbox
 *   never does and enforces the task's budget (P12, P13), so a turn is metered
 *   whether the model behaves or not.
 * - Each tool call is `provider.exec` inside the sandbox. Nothing the model
 *   asks for runs on the host (I1).
 *
 * What stays on the host is what must stay out of the agent's reach: the
 * tools the model is offered, the check that a call names one it was granted,
 * and the decision to run it (I4).
 */
import { posix } from 'node:path';
import {
  DRIVER_CONTRACT_VERSION,
  type CompiledRole,
  type Driver,
  type DriverCapabilities,
  type DriverEvent,
  type ModelIdentity,
  type ModelTier,
  type TaskId,
  type TaskRequest,
  type TaskResult,
  type Usage,
} from '@olympus-ai/core';
import type { ExecOptions, ExecResult, RelaySpec, SandboxHandle, SandboxProvider } from '@olympus-ai/sandbox';
import { CALL_SOURCE, RELAY_URL_VARIABLE } from './call.js';
import { checkConfig, type CheckedConfig, type EndpointConfig } from './config.js';
import { DriverRefusal, refuse } from './refusal.js';
import { DECLARED_TOOLS, definitionsFor, runTool, type ToolOutcome } from './tools.js';

/** The name the sandbox provider holds the endpoint's credential under. A name, never a value. */
export const ENDPOINT_CREDENTIAL = 'openai-compatible-endpoint';

/** This driver's own version, stamped into its provenance. Bumped when what it sends or how it reads changes. */
export const DRIVER_VERSION = '1';

export interface OpenAICompatibleDriverOptions {
  /**
   * The provider that provisioned the handle in `TaskRequest.sandbox`. Not
   * optional and not defaulted: a driver that could fall back to running on
   * the host is the failure I1 exists to prevent.
   */
  readonly provider: SandboxProvider;
  /** The endpoint, as the operator declared it (D-R6-02). */
  readonly endpoint: EndpointConfig;
  /** The working directory inside the sandbox. Defaults to the workspace mount's usual target. */
  readonly workdir?: string;
  /** The credential name the provider holds the endpoint's key under. Defaults to `ENDPOINT_CREDENTIAL`. */
  readonly credential?: string;
}

/** One message of the conversation, in the chat API's shape. */
type ChatMessage =
  | { readonly role: 'system' | 'user'; readonly content: string }
  | { readonly role: 'assistant'; readonly content: string | null; readonly tool_calls?: readonly ToolCall[] }
  | { readonly role: 'tool'; readonly tool_call_id: string; readonly content: string };

interface ToolCall {
  readonly id: string;
  readonly type: 'function';
  readonly function: { readonly name: string; readonly arguments: string };
}

/** One answer, read strictly: the model it says it is, its usage, and its message. */
interface Turn {
  readonly model: string;
  readonly promptTokens: number;
  readonly cachedTokens: number;
  readonly completionTokens: number;
  readonly content: string | null;
  readonly toolCalls: readonly ToolCall[];
}

/** The most of an endpoint's error body a refusal quotes. */
const QUOTE_LIMIT = 500;

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function count(value: unknown): number | undefined {
  return typeof value === 'number' && Number.isInteger(value) && value >= 0 ? value : undefined;
}

function quoted(text: string): string {
  return JSON.stringify(text.length > QUOTE_LIMIT ? `${text.slice(0, QUOTE_LIMIT)}…` : text);
}

/**
 * Reads one answer, or refuses the task. A response with no usage is refused
 * rather than counted as free (D-R6-05): the relay has already charged its
 * ceiling and will forward nothing more. One naming no model is refused as
 * one naming another would be (D-R6-03).
 */
function readTurn(body: string): Turn {
  let parsed: unknown;
  try {
    parsed = JSON.parse(body);
  } catch {
    refuse('endpoint', `the endpoint answered with a body that is not JSON: ${quoted(body)}`);
  }
  if (!isRecord(parsed)) refuse('endpoint', `the endpoint answered with ${quoted(body)}, which is not a chat completion`);
  const model = parsed.model;
  if (typeof model !== 'string' || model === '') {
    refuse('identity', 'the endpoint answered without naming its model, so the model the family was declared for cannot be confirmed (I6)');
  }
  const usage = parsed.usage;
  const promptTokens = isRecord(usage) ? count(usage.prompt_tokens) : undefined;
  const completionTokens = isRecord(usage) ? count(usage.completion_tokens) : undefined;
  if (!isRecord(usage) || promptTokens === undefined || completionTokens === undefined) {
    refuse('endpoint', 'the endpoint answered without prompt_tokens and completion_tokens; an endpoint whose calls cannot be counted is not served (D-R6-05)');
  }
  const details = usage.prompt_tokens_details;
  const cachedTokens = Math.min(isRecord(details) ? (count(details.cached_tokens) ?? 0) : 0, promptTokens);
  const choices = parsed.choices;
  const first: unknown = Array.isArray(choices) ? choices[0] : undefined;
  const message = isRecord(first) ? first.message : undefined;
  if (!isRecord(message)) refuse('endpoint', `the endpoint answered with no message: ${quoted(body)}`);
  const content = message.content;
  if (content !== null && content !== undefined && typeof content !== 'string') {
    refuse('endpoint', 'the endpoint answered with a message whose content is not text');
  }
  const calls = message.tool_calls;
  const toolCalls: ToolCall[] = [];
  if (calls !== null && calls !== undefined) {
    if (!Array.isArray(calls)) refuse('endpoint', 'the endpoint answered with tool_calls that is not a list');
    for (const call of calls as unknown[]) {
      const fn = isRecord(call) ? call.function : undefined;
      if (!isRecord(call) || typeof call.id !== 'string' || !isRecord(fn) || typeof fn.name !== 'string' || typeof fn.arguments !== 'string') {
        refuse('endpoint', `the endpoint answered with a tool call the driver cannot read: ${quoted(JSON.stringify(call))}`);
      }
      toolCalls.push({ id: call.id, type: 'function', function: { name: fn.name, arguments: fn.arguments } });
    }
  }
  return { model, promptTokens, cachedTokens, completionTokens, content: typeof content === 'string' ? content : null, toolCalls };
}

/** A tool call's arguments as an object, or the error the model is shown. */
function argumentsOf(call: ToolCall): Record<string, unknown> | string {
  try {
    const parsed: unknown = JSON.parse(call.function.arguments);
    if (isRecord(parsed)) return parsed;
  } catch {
    // Fall through to the same error as a non-object.
  }
  return `error: the arguments to '${call.function.name}' are not a JSON object`;
}

export class OpenAICompatibleDriver implements Driver {
  readonly id = 'openai-compatible';
  readonly contractVersion = DRIVER_CONTRACT_VERSION;

  readonly #provider: SandboxProvider;
  readonly #config: CheckedConfig;
  readonly #workdir: string;
  readonly #credential: string;
  /** One task per sandbox at a time, whatever `parallelism` says (A-R4-04). */
  readonly #inFlight = new Map<SandboxHandle, Promise<unknown>>();
  /** Tasks running now across sandboxes, and those waiting for a slot, in arrival order. */
  #running = 0;
  readonly #waiting: Array<() => void> = [];
  readonly #handlers = new Map<DriverEvent['kind'], Array<(e: DriverEvent) => void>>();
  readonly #cancelled = new Set<TaskId>();
  /** Every request body each task sent, kept so a caller — and a conformance assertion — can read what the model was offered. */
  readonly #requests = new Map<TaskId, Array<Record<string, unknown>>>();

  constructor(options: OpenAICompatibleDriverOptions) {
    // eslint-disable-next-line @typescript-eslint/no-unnecessary-condition -- the type requires a provider, but a missing one means running tools on the host (I1). A caller arriving from untyped code is refused, not trusted.
    if (options.provider === undefined || options.provider === null) {
      refuse('provider', 'OpenAICompatibleDriver needs the SandboxProvider that provisioned the handle; there is no host fallback (I1)');
    }
    this.#provider = options.provider;
    this.#config = checkConfig(options.endpoint);
    this.#workdir = options.workdir ?? '/workspace';
    this.#credential = options.credential ?? ENDPOINT_CREDENTIAL;
  }

  provenanceId(): string {
    return `${this.id}@${DRIVER_VERSION} ${this.#config.provider}/${this.#config.version}`;
  }

  /**
   * Every feature is false, and each is proven absent (D-R6-06): an endpoint
   * the driver has not measured backs no `true`. `parallelism` is the
   * operator's number, and `#bounded` is what makes it true.
   */
  capabilities(): DriverCapabilities {
    return {
      subagents: false,
      hooks: false,
      mcp: false,
      parallelism: this.#config.parallelism,
      computerUse: false,
      steering: false,
      stablePrefixCaching: false,
    };
  }

  declaredTools(): readonly string[] {
    return DECLARED_TOOLS;
  }

  /** The relay every sandbox this driver runs in is provisioned with: the endpoint's origin, its one path, and the `openai-chat` meter (A-R6-01). */
  relayRequest(): Omit<RelaySpec, 'budget'> {
    return {
      upstream: this.#config.upstream,
      paths: [this.#config.path],
      header: this.#config.header,
      scheme: this.#config.scheme,
      credential: this.#credential,
      urlVariable: RELAY_URL_VARIABLE,
      meter: { dialect: 'openai-chat', prices: { ...this.#config.prices } },
    };
  }

  /** I6: the family is the operator's declaration, branded once in the constructor. Nothing here derives it. */
  resolveModel(tier: ModelTier): ModelIdentity {
    const model = this.#config.models[tier];
    // eslint-disable-next-line @typescript-eslint/no-unnecessary-condition -- ModelTier is a union, but a tier arriving from parsed configuration can be outside it.
    if (model === undefined) refuse('capability', `no model is mapped to tier '${tier}'`);
    return { provider: this.#config.provider, family: this.#config.family, model, version: this.#config.version };
  }

  runTask(req: TaskRequest): Promise<TaskResult> {
    return this.#serialized(req.sandbox, () => this.#runTask(req));
  }

  /**
   * `spawnSubagent` and `steer` are deliberately absent, because
   * `capabilities()` declares both false. A method that threw would be a
   * capability declared false and present.
   */

  /** Stops a task at its next turn. The sandbox is the run's, so nothing here destroys it. */
  cancel(taskId: TaskId): Promise<void> {
    this.#cancelled.add(taskId);
    return Promise.resolve();
  }

  /**
   * There is no artifact format: the role is rendered on each request and
   * nowhere else (D-R6-06), so there is nothing to write and no hook to point
   * at. Refused rather than a no-op, which would read as roles written.
   */
  emitArtifacts(roles: CompiledRole[], targetDir: string): Promise<void> {
    return Promise.reject(
      new DriverRefusal(
        'capability',
        `OpenAICompatibleDriver has no artifact format; ${String(roles.length)} role(s) for ${targetDir} were not written. ` +
          'A role reaches this driver on each request, never as a file (D-R6-06).',
      ),
    );
  }

  on(kind: DriverEvent['kind'], handler: (e: DriverEvent) => void): void {
    const existing = this.#handlers.get(kind);
    if (existing === undefined) this.#handlers.set(kind, [handler]);
    else existing.push(handler);
  }

  /** Every request body a task sent to the endpoint, in order. */
  requestsFor(taskId: TaskId): ReadonlyArray<Readonly<Record<string, unknown>>> {
    return this.#requests.get(taskId) ?? [];
  }

  async #serialized<T>(handle: SandboxHandle, body: () => Promise<T>): Promise<T> {
    // Checked again after every wait: another follower may have claimed the handle first.
    for (let held = this.#inFlight.get(handle); held !== undefined; held = this.#inFlight.get(handle)) {
      await held.catch(() => undefined);
    }
    const running = this.#bounded(body);
    this.#inFlight.set(handle, running);
    try {
      return await running;
    } finally {
      if (this.#inFlight.get(handle) === running) this.#inFlight.delete(handle);
    }
  }

  async #bounded<T>(body: () => Promise<T>): Promise<T> {
    if (this.#running >= this.#config.parallelism) await new Promise<void>((admit) => this.#waiting.push(admit));
    else this.#running += 1;
    try {
      return await body();
    } finally {
      // A slot passes straight to the next caller waiting, so a newcomer never jumps the queue.
      const next = this.#waiting.shift();
      if (next === undefined) this.#running -= 1;
      else next();
    }
  }

  #emit(event: DriverEvent): void {
    for (const handler of this.#handlers.get(event.kind) ?? []) handler(event);
  }

  /**
   * I4: a granted tool this driver does not declare is refused before the
   * task starts, so a caller that bypassed the policy engine gets no wider
   * session than policy would have allowed.
   */
  #refuseUngrantedTools(req: TaskRequest): void {
    const declared = new Set(DECLARED_TOOLS);
    const undeclared = req.tools.filter((tool) => !declared.has(tool));
    if (undeclared.length > 0) {
      refuse(
        'grant',
        `the task grants ${undeclared.map((t) => `'${t}'`).join(', ')}, which this driver does not declare. ` +
          'The task was not started: a grant this driver cannot honour is a policy that means something other than what it says (I4).',
      );
    }
  }

  async #runTask(req: TaskRequest): Promise<TaskResult> {
    this.#refuseUngrantedTools(req);
    this.#cancelled.delete(req.taskId);
    const identity = this.resolveModel(req.tier);
    const granted = new Set(req.tools);
    const tools = definitionsFor(req.tools);
    const started = Date.now();
    const deadline = started + req.timeoutMs;
    const exec = (argv: string[], options?: ExecOptions): Promise<ExecResult> => this.#provider.exec(req.sandbox, argv, options);

    // The role and the run's stable material form the system message, and this task's own text the user
    // message, as the Claude Code driver renders them (D-A-RC-03): instructions first, never a file.
    const messages: ChatMessage[] = [
      { role: 'system', content: `${req.role.instructions}\n\n${req.stablePrefix}` },
      { role: 'user', content: req.variableSuffix },
    ];
    const events: DriverEvent[] = [];
    const written = new Set<string>();
    const usage = { inputTokens: 0, outputTokens: 0, cacheReadTokens: 0, cacheWriteTokens: 0, costUsd: 0 };
    const sent: Array<Record<string, unknown>> = [];
    this.#requests.set(req.taskId, sent);
    const price = this.#config.prices[identity.model];
    // checkConfig refuses a tier model with no price; the table is a record, so the index is checked again here.
    if (price === undefined) refuse('configuration', `the model ${identity.model} has no price`);

    for (let turn = 1; ; turn += 1) {
      if (this.#cancelled.has(req.taskId)) refuse('invocation', `task ${req.taskId} was cancelled`);
      if (Date.now() >= deadline) refuse('invocation', `task ${req.taskId} ran past its ${String(req.timeoutMs)} ms bound`);
      if (turn > this.#config.maxTurns) {
        refuse('invocation', `task ${req.taskId} did not finish in ${String(this.#config.maxTurns)} model calls, the configured maxTurns`);
      }

      const body: Record<string, unknown> = {
        model: identity.model,
        messages,
        [this.#config.ceilingField]: this.#config.maxOutputTokens,
        stream: false,
        ...(tools.length === 0 ? {} : { tools }),
      };
      sent.push(JSON.parse(JSON.stringify(body)) as Record<string, unknown>);
      const answer = await this.#call(exec, body);
      const reply = readTurn(answer);
      if (reply.model !== identity.model) {
        refuse(
          'identity',
          `the endpoint answered as ${JSON.stringify(reply.model)} when ${JSON.stringify(identity.model)} was requested. ` +
            `The family '${identity.family}' was declared for the model requested, so the answer is not one the run can attribute (I6, D-R6-03).`,
        );
      }
      const input = reply.promptTokens - reply.cachedTokens;
      usage.inputTokens += input;
      usage.cacheReadTokens += reply.cachedTokens;
      usage.outputTokens += reply.completionTokens;
      usage.costUsd += (input * price.inputPerMTok + reply.cachedTokens * price.cacheReadPerMTok + reply.completionTokens * price.outputPerMTok) / 1e6;

      if (reply.toolCalls.length === 0) {
        const result: TaskResult = {
          taskId: req.taskId,
          claim: { narrative: reply.content ?? '', filesChanged: [...written].map((path) => this.#claimed(path)) },
          events,
          usage: { ...usage, wallClockMs: Date.now() - started } satisfies Usage,
          model: identity,
          contractVersion: this.contractVersion,
        };
        return result;
      }

      messages.push({ role: 'assistant', content: reply.content, tool_calls: reply.toolCalls });
      for (const call of reply.toolCalls) {
        const name = call.function.name;
        let output: string;
        if (!granted.has(name)) {
          // Never run, and recorded: the model asked for a tool policy did not give it (I4).
          output = `error: the tool '${name}' is not granted to this task`;
          const refused: DriverEvent = { at: new Date().toISOString(), kind: 'tool-call', detail: { tool: name, refused: true } };
          events.push(refused);
          this.#emit(refused);
        } else {
          const args = argumentsOf(call);
          const outcome: ToolOutcome =
            typeof args === 'string'
              ? { output: args, event: { at: new Date().toISOString(), kind: 'tool-call' as const, detail: { tool: name, error: args } } }
              : await runTool(name, args, exec, this.#workdir);
          output = outcome.output;
          events.push(outcome.event);
          this.#emit(outcome.event);
          if (outcome.written !== undefined) written.add(outcome.written);
        }
        messages.push({ role: 'tool', tool_call_id: call.id, content: output });
      }
    }
  }

  /** One model call through the relay, made from inside the sandbox. Returns the body of a success; refuses on anything else. */
  async #call(exec: (argv: string[], options?: ExecOptions) => Promise<ExecResult>, body: Record<string, unknown>): Promise<string> {
    const result = await exec(['node', '--eval', CALL_SOURCE, this.#config.path], { stdin: JSON.stringify(body) });
    if (result.exitCode !== 0) {
      refuse('invocation', `the model call inside the sandbox exited ${String(result.exitCode)}: ${result.stderr.trim()}`);
    }
    let answer: unknown;
    try {
      answer = JSON.parse(result.stdout);
    } catch {
      answer = undefined;
    }
    if (!isRecord(answer) || typeof answer.status !== 'number' || typeof answer.body !== 'string') {
      refuse('invocation', `the model call inside the sandbox printed ${quoted(result.stdout)}, not a status and a body`);
    }
    if (answer.status === 402) refuse('budget', `the relay refused the call: ${quoted(answer.body)}`);
    if (answer.status < 200 || answer.status >= 300) {
      refuse('endpoint', `the endpoint answered ${String(answer.status)}: ${quoted(answer.body)}`);
    }
    return answer.body;
  }

  /** A written path as the claim names it: relative to the working directory when inside it. */
  #claimed(path: string): string {
    const relative = posix.relative(this.#workdir, path);
    return relative.startsWith('..') || posix.isAbsolute(relative) ? path : relative;
  }
}

/**
 * One refusal type for the driver, shaped like the sandbox's and the Claude
 * Code driver's: a `layer` names which control declined, so an assertion can
 * require the right refusal rather than merely some error (I5).
 */
export type DriverRefusalLayer =
  /** No sandbox provider: there is no host fallback (I1). */
  | 'provider'
  /** The operator's configuration is missing a field, or holds one the driver cannot use as written. */
  | 'configuration'
  /** A tool the request grants is not one this driver declares (I4). */
  | 'grant'
  /** A capability the request needs is not one this driver declares. */
  | 'capability'
  /** The in-sandbox call could not be made, the task was cancelled, or it ran past its bounds. */
  | 'invocation'
  /** The endpoint answered with an error, or with a body the driver cannot read. */
  | 'endpoint'
  /** The endpoint answered as a model other than the one the request named (I6). */
  | 'identity'
  /** The relay refused the call because the task's budget is spent (P13). */
  | 'budget';

export class DriverRefusal extends Error {
  override readonly name = 'DriverRefusal';
  constructor(
    readonly layer: DriverRefusalLayer,
    message: string,
  ) {
    super(message);
  }
}

export function refuse(layer: DriverRefusalLayer, message: string): never {
  throw new DriverRefusal(layer, message);
}

===== packages/drivers/openai-compatible/src/tools.ts =====
/**
 * The tool inventory (`I4.openai-compatible-driver-tool-inventory-validated`)
 * and how each tool runs: inside the sandbox, through `provider.exec`, and
 * never on the host (I1, D-R6-01).
 *
 * The names are the Claude Code driver's own, so one policy grants either
 * driver the same tools (D-R6-04). A constant rather than a probe, because
 * `validateToolGrants` runs before any task starts.
 *
 * What a tool returns goes back to the model as text. A tool that fails —
 * a missing file, an edit that does not match, arguments that do not parse —
 * is an error the model reads and may recover from; it is not a refusal of the
 * task, as it is not for the CLI either.
 */
import { posix } from 'node:path';
import type { DriverEvent } from '@olympus-ai/core';
import type { ExecOptions, ExecResult } from '@olympus-ai/sandbox';

export const DECLARED_TOOLS: readonly string[] = ['Bash', 'Edit', 'Read', 'Write'];

/** A function the chat API offers the model, in the API's own shape. */
export interface ToolDefinition {
  readonly type: 'function';
  readonly function: {
    readonly name: string;
    readonly description: string;
    readonly parameters: Readonly<Record<string, unknown>>;
  };
}

function definition(name: string, description: string, properties: Record<string, { type: 'string'; description: string }>): ToolDefinition {
  return Object.freeze({
    type: 'function',
    function: Object.freeze({
      name,
      description,
      parameters: Object.freeze({ type: 'object', properties, required: Object.keys(properties), additionalProperties: false }),
    }),
  });
}

const DEFINITIONS: Readonly<Record<string, ToolDefinition>> = Object.freeze({
  Bash: definition('Bash', 'Runs a shell command in the working directory and returns its exit code, stdout, and stderr.', {
    command: { type: 'string', description: 'The command, run by sh -c.' },
  }),
  Edit: definition('Edit', 'Replaces the one occurrence of old_string in a file with new_string. Fails unless old_string occurs exactly once.', {
    path: { type: 'string', description: 'The file, absolute or relative to the working directory.' },
    old_string: { type: 'string', description: 'The exact text to replace. Must occur exactly once.' },
    new_string: { type: 'string', description: 'The text to put in its place.' },
  }),
  Read: definition('Read', 'Returns the contents of a file.', {
    path: { type: 'string', description: 'The file, absolute or relative to the working directory.' },
  }),
  Write: definition('Write', 'Writes a file, creating it and its directories, and replacing anything already there.', {
    path: { type: 'string', description: 'The file, absolute or relative to the working directory.' },
    content: { type: 'string', description: 'The whole content of the file.' },
  }),
});

/** The definitions of the granted tools, in inventory order, and of no others (I4). */
export function definitionsFor(granted: readonly string[]): ToolDefinition[] {
  const wanted = new Set(granted);
  return DECLARED_TOOLS.filter((name) => wanted.has(name)).map((name) => DEFINITIONS[name]).filter((d): d is ToolDefinition => d !== undefined);
}

/** How a tool reaches the sandbox: the provider's exec against the task's handle, and nothing else. */
export type SandboxExec = (argv: string[], options?: ExecOptions) => Promise<ExecResult>;

/** What one tool call did: the text the model reads, the event the run records, and the file it wrote, if any. */
export interface ToolOutcome {
  readonly output: string;
  readonly event: DriverEvent;
  readonly written?: string;
}

/** The most of any one stream returned to the model. More is cut, and the cut is said. */
export const OUTPUT_LIMIT = 64 * 1024;

function clipped(text: string): string {
  if (text.length <= OUTPUT_LIMIT) return text;
  return `${text.slice(0, OUTPUT_LIMIT)}\n[cut: ${String(text.length - OUTPUT_LIMIT)} more characters not shown]`;
}

function event(kind: DriverEvent['kind'], detail: Record<string, unknown>): DriverEvent {
  return { at: new Date().toISOString(), kind, detail };
}

/** A path inside the container, resolved against the working directory. The sandbox is the boundary, as it is for the CLI (D-R6-04). */
function resolved(workdir: string, path: string): string {
  return posix.resolve(workdir, path);
}

/** The argv that writes stdin to a path, creating its directories. The path is an argument, never shell text, and `--` ends option parsing. */
function writeArgv(path: string): string[] {
  return ['sh', '-c', 'mkdir -p -- "$(dirname -- "$1")" && cat > "$1"', 'driver', path];
}

type Args = Record<string, unknown>;

/** The string arguments a tool needs, or the error the model is shown when they are not there. */
function strings(args: Args, names: readonly string[]): Record<string, string> | string {
  const out: Record<string, string> = {};
  for (const name of names) {
    const value = args[name];
    if (typeof value !== 'string') return `error: the argument '${name}' must be a string`;
    out[name] = value;
  }
  return out;
}

/**
 * Runs one granted tool. The caller has already checked the grant; this runs
 * whatever it is handed, so it is never reached for a tool the task was not
 * granted.
 */
export async function runTool(name: string, args: Args, exec: SandboxExec, workdir: string): Promise<ToolOutcome> {
  switch (name) {
    case 'Bash': {
      const a = strings(args, ['command']);
      if (typeof a === 'string') return { output: a, event: event('tool-call', { tool: name, error: a }) };
      const command = a.command ?? '';
      // $1 is the directory and $2 the command: neither is spliced into shell text.
      const result = await exec(['sh', '-c', 'cd -- "$1" && exec sh -c "$2"', 'driver', workdir, command]);
      return {
        output: `exit code: ${String(result.exitCode)}\n--- stdout ---\n${clipped(result.stdout)}\n--- stderr ---\n${clipped(result.stderr)}`,
        event: event('command', { tool: name, command, exitCode: result.exitCode }),
      };
    }
    case 'Read': {
      const a = strings(args, ['path']);
      if (typeof a === 'string') return { output: a, event: event('tool-call', { tool: name, error: a }) };
      const path = resolved(workdir, a.path ?? '');
      const result = await exec(['cat', '--', path]);
      const output = result.exitCode === 0 ? clipped(result.stdout) : `error: reading ${path} failed: ${result.stderr.trim()}`;
      return { output, event: event('tool-call', { tool: name, path, exitCode: result.exitCode }) };
    }
    case 'Write': {
      const a = strings(args, ['path', 'content']);
      if (typeof a === 'string') return { output: a, event: event('tool-call', { tool: name, error: a }) };
      const path = resolved(workdir, a.path ?? '');
      const result = await exec(writeArgv(path), { stdin: a.content ?? '' });
      if (result.exitCode !== 0) {
        return { output: `error: writing ${path} failed: ${result.stderr.trim()}`, event: event('tool-call', { tool: name, path, exitCode: result.exitCode }) };
      }
      return { output: `wrote ${path}`, event: event('file-write', { tool: name, path }), written: path };
    }
    case 'Edit': {
      const a = strings(args, ['path', 'old_string', 'new_string']);
      if (typeof a === 'string') return { output: a, event: event('tool-call', { tool: name, error: a }) };
      const path = resolved(workdir, a.path ?? '');
      const before = a.old_string ?? '';
      if (before === '') return { output: 'error: old_string is empty', event: event('tool-call', { tool: name, path, error: 'empty old_string' }) };
      const read = await exec(['cat', '--', path]);
      if (read.exitCode !== 0) {
        return { output: `error: reading ${path} failed: ${read.stderr.trim()}`, event: event('tool-call', { tool: name, path, exitCode: read.exitCode }) };
      }
      const occurrences = read.stdout.split(before).length - 1;
      if (occurrences !== 1) {
        const error = `old_string occurs ${String(occurrences)} times in ${path}; it must occur exactly once`;
        return { output: `error: ${error}`, event: event('tool-call', { tool: name, path, error }) };
      }
      const written = await exec(writeArgv(path), { stdin: read.stdout.replace(before, () => a.new_string ?? '') });
      if (written.exitCode !== 0) {
        return { output: `error: writing ${path} failed: ${written.stderr.trim()}`, event: event('tool-call', { tool: name, path, exitCode: written.exitCode }) };
      }
      return { output: `edited ${path}`, event: event('file-write', { tool: name, path }), written: path };
    }
    default:
      return { output: `error: no tool named '${name}'`, event: event('tool-call', { tool: name, refused: true }) };
  }
}

async function refusalOf(action: () => unknown): Promise<DriverRefusal> {
  try {
    await action();
  } catch (error) {
    if (error instanceof DriverRefusal) return error;
    throw error;
  }
  throw new Error('expected a DriverRefusal, and nothing was refused');
}

/** Every key anywhere in a JSON value, at any depth. */
function keysIn(value: unknown, into = new Set<string>()): Set<string> {
  if (Array.isArray(value)) for (const item of value) keysIn(item, into);
  else if (typeof value === 'object' && value !== null) {
    for (const [key, inner] of Object.entries(value)) {
      into.add(key);
      keysIn(inner, into);
    }
  }
  return into;
}

/**
 * The driver's own behaviour beyond the registry's assertions: the
 * configuration it refuses, the answers it refuses, and the tool errors a
 * model reads and recovers from. Docker and `openssl` where a sandbox is
 * needed; no model is called.
 */
import { readFile } from 'node:fs/promises';
import { join } from 'node:path';
import { describe, expect, test } from 'vitest';
import type { TaskId } from '@olympus-ai/core';
import { CALL_SOURCE, DriverRefusal, ENDPOINT_CREDENTIAL, RELAY_URL_VARIABLE, checkConfig, type EndpointConfig } from '../src/index.js';

async function refusalOf(action: () => unknown): Promise<DriverRefusal> {

===== packages/drivers/openai-compatible/test/endpoint.ts =====
/**
 * An OpenAI-compatible endpoint the suites own: a TLS origin in its own
 * container, on a network only the relay joins, answering the chat API with a
 * conversation the test scripts.
 *
 * No model is called. The script rides in the task's user message, which the
 * driver sends on every turn: `{"turns": [...]}`, one entry per model call,
 * picked by how many assistant messages the request already holds. A turn is
 * `{"content": "..."}` to finish, `{"calls": [{"name", "arguments"}]}` to call
 * tools, and may carry `model` (answer as another model, or `null` for none),
 * `usage: false` (answer with no usage), `status` (answer with an error), and
 * `hold` (milliseconds before answering).
 *
 * It logs every request body, base64, and whether the credential arrived as
 * `Bearer <expected>`, so an assertion reads what the endpoint received rather
 * than what the driver meant to send. It never echoes the credential.
 *
 * Requires Docker and `openssl`, and fails without either, as the sandbox's
 * relay suites do.
 */
import { execFile } from 'node:child_process';
import { mkdtemp, readFile, rm } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { promisify } from 'node:util';
import { EGRESS_PROXY_IMAGE } from '@olympus-ai/sandbox';

const run = promisify(execFile);

export const ENDPOINT_HOST = 'endpoint.test';
export const ENDPOINT_PORT = 8443;
export const ENDPOINT_ORIGIN = `https://${ENDPOINT_HOST}:${String(ENDPOINT_PORT)}`;
export const ENDPOINT_PATH = '/v1/chat/completions';

export interface Certificate {
  readonly cert: string;
  readonly key: string;
}

/** A self-signed certificate for `ENDPOINT_HOST`, valid for a day, handed to the provider as `relayTrust`. */
export async function makeCertificate(): Promise<Certificate> {
  const dir = await mkdtemp(join(tmpdir(), 'endpoint-cert-'));
  try {
    await run(
      'openssl',
      [
        'req', '-x509', '-newkey', 'ec', '-pkeyopt', 'ec_paramgen_curve:P-256', '-nodes', '-days', '1',
        '-subj', `/CN=${ENDPOINT_HOST}`, '-addext', `subjectAltName=DNS:${ENDPOINT_HOST}`,
        '-keyout', join(dir, 'key.pem'), '-out', join(dir, 'cert.pem'),
      ],
      // Git for Windows' runtime rewrites an argument that starts with `/` into a Windows path.
      { env: { ...process.env, MSYS_NO_PATHCONV: '1' } },
    );
    return { cert: await readFile(join(dir, 'cert.pem'), 'utf8'), key: await readFile(join(dir, 'key.pem'), 'utf8') };
  } finally {
    await rm(dir, { recursive: true, force: true });
  }
}

/** One scripted model call. */
export interface ScriptedTurn {
  readonly content?: string;
  readonly calls?: ReadonlyArray<{ readonly name: string; readonly arguments: Record<string, unknown> | string }>;
  readonly model?: string | null;
  readonly usage?: false | { readonly prompt: number; readonly completion: number; readonly cached?: number };
  readonly status?: number;
  readonly hold?: number;
}

/** The user message that scripts a conversation. */
export function script(turns: readonly ScriptedTurn[]): string {
  return JSON.stringify({ turns });
}

const ENDPOINT_SOURCE = `'use strict';
const expected = String(process.env.ENDPOINT_EXPECTED || '');
require('node:https').createServer({ key: process.env.ENDPOINT_KEY, cert: process.env.ENDPOINT_CERT }, function (q, s) {
  const parts = [];
  q.on('data', function (c) { parts.push(c); });
  q.on('end', function () {
    const raw = Buffer.concat(parts);
    console.log('ENDPOINT ' + q.method + ' ' + q.url + ' ' + String(q.headers.authorization === 'Bearer ' + expected) + ' ' + raw.toString('base64'));
    let request = {};
    try { request = JSON.parse(raw.toString('utf8')); } catch (error) { request = {}; }
    const messages = Array.isArray(request.messages) ? request.messages : [];
    const user = messages.find(function (m) { return m && m.role === 'user'; });
    let turns = [];
    try { turns = JSON.parse(user.content).turns || []; } catch (error) { turns = []; }
    const index = messages.filter(function (m) { return m && m.role === 'assistant'; }).length;
    const t = turns[index] || { content: 'script exhausted' };
    setTimeout(function () {
      if (t.status) {
        s.writeHead(t.status, { 'content-type': 'application/json' });
        s.end(JSON.stringify({ error: { type: 'server_error', message: 'scripted failure' } }));
        return;
      }
      const message = { role: 'assistant', content: t.content === undefined ? null : t.content };
      if (Array.isArray(t.calls)) {
        message.tool_calls = t.calls.map(function (c, i) {
          return { id: 'call_' + String(index) + '_' + String(i), type: 'function',
            function: { name: c.name, arguments: typeof c.arguments === 'string' ? c.arguments : JSON.stringify(c.arguments) } };
        });
      }
      const answer = { id: 'chat_' + String(index), object: 'chat.completion',
        choices: [{ index: 0, message: message, finish_reason: message.tool_calls ? 'tool_calls' : 'stop' }] };
      if (t.model !== null) answer.model = t.model === undefined ? request.model : t.model;
      if (t.usage !== false) {
        const u = t.usage || { prompt: 100, completion: 10, cached: 0 };
        answer.usage = { prompt_tokens: u.prompt, completion_tokens: u.completion, prompt_tokens_details: { cached_tokens: u.cached || 0 } };
      }
      const body = JSON.stringify(answer);
      s.writeHead(200, { 'content-type': 'application/json', 'content-length': Buffer.byteLength(body) });
      s.end(body);
    }, typeof t.hold === 'number' ? t.hold : 0);
  });
}).listen(${String(ENDPOINT_PORT)}, '0.0.0.0');
`;

/** Starts the endpoint on `network` under `ENDPOINT_HOST`. The sandbox is never on that network. */
export async function startEndpoint(name: string, network: string, certificate: Certificate, expected: string): Promise<void> {
  await run(
    'docker',
    [
      'run', '--detach', '--init', '--name', name, '--network', network, '--network-alias', ENDPOINT_HOST,
      '--env', 'ENDPOINT_KEY', '--env', 'ENDPOINT_CERT', '--env', 'ENDPOINT_EXPECTED',
      EGRESS_PROXY_IMAGE, 'node', '--eval', ENDPOINT_SOURCE,
    ],
    { env: { ...process.env, ENDPOINT_KEY: certificate.key, ENDPOINT_CERT: certificate.cert, ENDPOINT_EXPECTED: expected } },
  );
  const probe =
    `require('node:net').connect(${String(ENDPOINT_PORT)}, '127.0.0.1')` +
    ".on('connect', function () { process.exit(0); }).on('error', function () { process.exit(1); });";
  for (let attempt = 0; attempt < 40; attempt += 1) {
    try {
      await run('docker', ['exec', name, 'node', '--eval', probe]);
      return;
    } catch {
      await new Promise((done) => setTimeout(done, 250));
    }
  }
  throw new Error(`the endpoint container ${name} never listened`);
}

/** One request the endpoint received: its path, whether the bearer credential matched, and its body. */
export interface Received {
  readonly method: string;
  readonly path: string;
  readonly bearerMatches: boolean;
  readonly body: Record<string, unknown>;
}

export async function received(name: string): Promise<Received[]> {
  const { stdout } = await run('docker', ['logs', name], { maxBuffer: 64 * 1024 * 1024 });
  return stdout
    .split(/\r?\n/u)
    .map((line) => /^ENDPOINT (\S+) (\S+) (true|false) (\S*)$/u.exec(line))
    .filter((m): m is RegExpExecArray => m !== null)
    .map((m) => ({
      method: m[1] ?? '',
      path: m[2] ?? '',
      bearerMatches: m[3] === 'true',
      body: JSON.parse(Buffer.from(m[4] ?? '', 'base64').toString('utf8') || '{}') as Record<string, unknown>,
    }));
}

const run = promisify(execFile);

export const WORKDIR = '/workspace';

/** A dollar per million input tokens, ten per million output, half a dollar per million cached. Round numbers, checked by hand. */

/** One exec the driver made, as the provider was asked to run it. */
export interface SeenExec {
  readonly handle: SandboxHandle;
  readonly argv: readonly string[];
  readonly startedAt: number;
  endedAt?: number;
}

export interface Harness {
  readonly provider: LocalDockerProvider;
  readonly driver: OpenAICompatibleDriver;
  /** The credential the provider holds for the relay. A canary: never a real key. */
  readonly secret: string;
  /** Every exec any driver built from this harness made, in order. */
  readonly execs: SeenExec[];
  /** A fresh sandbox with its relay reaching the endpoint, and its workspace on the host. */
  sandbox(): Promise<{ handle: SandboxHandle; workspace: string }>;
  request(handle: SandboxHandle, overrides?: Partial<TaskRequest>): TaskRequest;
  /** What the endpoint received, in order. */
  received(): Promise<Received[]>;
  /** A driver over the same provider with a different configuration. */
  driverWith(endpoint: EndpointConfig): OpenAICompatibleDriver;
}

let certificate: Certificate | undefined;

/** A provider whose exec is watched: what the driver ran, where, and when. */
function watched(provider: LocalDockerProvider, execs: SeenExec[]): LocalDockerProvider {
  return new Proxy(provider, {
    get(target, property, receiver) {
      if (property === 'exec') {
        return async (handle: SandboxHandle, argv: string[], options?: ExecOptions): Promise<ExecResult> => {
          const seen: SeenExec = { handle, argv: [...argv], startedAt: Date.now() };
          execs.push(seen);
          try {
            return await target.exec(handle, argv, options);
          } finally {
            seen.endedAt = Date.now();
          }
        };
      }
      const value: unknown = Reflect.get(target, property, receiver);
      return typeof value === 'function' ? (value as (...args: unknown[]) => unknown).bind(target) : value;
    },
  });
}

export async function withHarness<T>(body: (h: Harness) => Promise<T>, endpoint: EndpointConfig = CONFIG): Promise<T> {
  certificate ??= await makeCertificate();
  const secret = `canary-${randomUUID()}`;
  const id = randomUUID().slice(0, 8);
  const network = `oac-endpoint-${id}`;
  const endpointName = `oac-endpoint-${id}`;
  const base = await mkdtemp(join(tmpdir(), 'openai-compatible-'));
  const provider = await LocalDockerProvider.create({ vaultPaths: [], credentials: { [ENDPOINT_CREDENTIAL]: secret }, relayTrust: certificate.cert });
  const execs: SeenExec[] = [];
  const observed = watched(provider, execs);
  const live: SandboxHandle[] = [];
  await run('docker', ['network', 'create', network]);
  try {
    await startEndpoint(endpointName, network, certificate, secret);
    const driverWith = (config: EndpointConfig): OpenAICompatibleDriver => new OpenAICompatibleDriver({ provider: observed, endpoint: config, workdir: WORKDIR });
    const driver = driverWith(endpoint);
    return await body({
      provider,
      driver,
      secret,
      execs,
      driverWith,
      async sandbox() {
        const workspace = join(base, `ws-${String(live.length)}`);
        await mkdir(workspace, { recursive: true });
        const handle = await provider.provision({
          image: EGRESS_PROXY_IMAGE,
          mounts: { workspace: { source: workspace, target: WORKDIR, mode: 'rw' }, others: [] },
          egress: { mode: 'deny-all', allow: [] },
          limits: { cpus: 0.5, memoryMb: 256, pids: 64, wallClockMs: BUDGET.maxWallClockMs },
          user: { uid: 0, gid: 0 },
          relay: { ...driver.relayRequest(), budget: { maxTokens: BUDGET.maxTokens, maxCostUsd: BUDGET.maxCostUsd } },
        });
        live.push(handle);
        const relay = relayOf(provider.appliedControls(handle).egress);
        if (relay === undefined) throw new Error('the sandbox was provisioned without its relay');
        await run('docker', ['network', 'connect', network, relay.containerId]);
        return { handle, workspace };
      },
      request: (handle, overrides = {}) => ({
        taskId: 'task' as TaskId,
        role: ROLE,
        stablePrefix: 'the stable prefix',
        variableSuffix: '{"turns":[{"content":"done"}]}',
        tier: 'standard',
        tools: [],
        sandbox: handle,
        timeoutMs: BUDGET.maxWallClockMs,
        budget: BUDGET,
        ...overrides,
      }),
      received: () => received(endpointName),
    });
  } finally {
    for (const handle of live) await provider.destroy(handle).catch(() => undefined);
    await run('docker', ['rm', '--force', endpointName]).catch(() => undefined);
    await run('docker', ['network', 'rm', network]).catch(() => undefined);
    await rm(base, { recursive: true, force: true });
  }
}

/** The execs that were model calls: the fixed caller, run by node. */
export function modelCalls(execs: readonly SeenExec[]): SeenExec[] {
  return execs.filter((e) => e.argv[0] === 'node' && e.argv[1] === '--eval');
}

/** The most of `execs` that were running at one moment. */
export function mostAtOnce(execs: readonly SeenExec[]): number {
  let most = 0;
  for (const e of execs) {
    const overlapping = execs.filter((o) => o.startedAt <= e.startedAt && (o.endedAt ?? Infinity) > e.startedAt).length;
    most = Math.max(most, overlapping);
  }
  return most;
}

/** The argument vector and environment of every process the container user can read, read from inside (D-P5-20). */
export const PROCESS_DUMP = 'for p in /proc/[0-9]*; do cat "$p/environ" "$p/cmdline" 2>/dev/null; printf "\\n"; done';

/** Whether `secret` is anywhere in a container's own filesystem, searched from the host. Nothing is sent into the container. */
export function exportContains(container: string, secret: string): Promise<boolean> {
  const needle = Buffer.from(secret, 'utf8');
  return new Promise((resolve, reject) => {
    const child = spawn('docker', ['export', container], { stdio: ['ignore', 'pipe', 'pipe'], windowsHide: true });
    let tail = Buffer.alloc(0);
    let found = false;
    child.stdout.on('data', (chunk: Buffer) => {
      if (found) return;
      const window = Buffer.concat([tail, chunk]);
      if (window.includes(needle)) found = true;
      else tail = window.subarray(Math.max(0, window.length - needle.length + 1));
    });
    child.on('error', reject);
    child.on('close', (code) => {
      if (code !== 0 && !found) reject(new Error(`docker export exited ${String(code)}`));
      else resolve(found);
    });
  });
}

export async function dockerOut(args: string[]): Promise<string> {
  const { stdout } = await run('docker', args, { maxBuffer: 64 * 1024 * 1024 });
  return stdout;
}

async function refusalOf(action: () => unknown): Promise<DriverRefusal> {
  try {
    await action();
  } catch (error) {
    if (error instanceof DriverRefusal) return error;
    throw error;
  }
  throw new Error('expected a DriverRefusal, and nothing was refused');
}

function functionNames(body: Record<string, unknown>): string[] {
  const tools = body.tools;
  if (!Array.isArray(tools)) return [];
  return (tools as Array<{ function: { name: string } }>).map((t) => t.function.name);
}

invariantTest(
  'I1.openai-compatible-driver-executes-inside-the-sandbox',
  "every tool call and every model call of a task runs inside the provisioned sandbox through provider.exec, the host's own filesystem is never written, and a driver with no provider cannot be constructed",
  async () => {
    await withHarness(async (h) => {
      const { handle, workspace } = await h.sandbox();
      const marker = `/tmp/ran-inside-${String(Date.now())}`;
      const result = await h.driver.runTask(
        h.request(handle, {
          taskId: 'i1-tools' as TaskId,
          tools: ['Bash', 'Edit', 'Read', 'Write'],
          variableSuffix: script([
            { calls: [{ name: 'Write', arguments: { path: 'notes/a.txt', content: 'alpha beta\n' } }] },
            { calls: [{ name: 'Edit', arguments: { path: 'notes/a.txt', old_string: 'beta', new_string: 'gamma' } }] },
            { calls: [{ name: 'Read', arguments: { path: `${WORKDIR}/notes/a.txt` } }, { name: 'Bash', arguments: { command: `cat /etc/hostname > ${marker} && pwd` } }] },
            { content: 'finished' },
          ]),
        }),

      // The workspace mount holds what the tools wrote, so the writes landed in the container's view of it.

      expect(result.claim.filesChanged).toStrictEqual(['notes/a.txt']);
      expect(result.events.map((e) => [e.kind, e.detail.tool])).toStrictEqual([
        ['file-write', 'Write'],
        ['file-write', 'Edit'],
        ['tool-call', 'Read'],
        ['command', 'Bash'],
      ]);

      // The command ran in the container: its marker is there, holding the container's own hostname.
      const inside = await h.provider.exec(handle, ['cat', marker]);
      expect(inside.exitCode).toBe(0);
      const hostname = (await dockerOut(['inspect', '--format', '{{.Config.Hostname}}', h.provider.appliedControls(handle).containerId])).trim();
      expect(inside.stdout.trim()).toBe(hostname);

      // What the model read back was the container's file, and Bash ran in the working directory.
      const sent = (await h.received()).map((r) => r.body);
      expect(sent).toHaveLength(4);
      const toolResults = (sent[3]?.messages as Array<{ role: string; content: string }>).filter((m) => m.role === 'tool').map((m) => m.content);
      expect(toolResults[toolResults.length - 2]).toBe('alpha gamma\n');
      expect(toolResults[toolResults.length - 1]).toContain(`--- stdout ---\n${WORKDIR}\n`);

      // Every exec the driver made was against this sandbox's handle: four model calls and the tools' own.
      expect(h.execs.every((e) => e.handle === handle)).toBe(true);
      expect(modelCalls(h.execs)).toHaveLength(4);
    });

    // No provider is no driver: there is no host to fall back to.

  },
