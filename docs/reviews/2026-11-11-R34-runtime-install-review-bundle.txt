R34: dependencies installed by the runtime from the lockfile. Policy gains `installRegistries` (A-R34-01); adapters declare `DependencyInstaller` for npm and pnpm (A-R34-02); evidence and the base record carry an `InstallRecord`. Before the base run and before any check of a task, the runtime installs the tree's lockfile in a sandbox of its own, with egress to the granted registries alone and no lifecycle scripts, and the checks run over that install with no egress. A lockfile without hashes, several, an unsupported one, a manifest with none, or no registry granted is refused, naming the cause.

/**
 * The dependency installs the runtime runs before any check (R34, D-R34-06).
 * Each is declared once, here: the lockfile it reads, the files copied into
 * its sandbox, the image it is pinned to, the arguments, and how its
 * lockfile lists each package and the hash it records. A policy names the
 * registries the install may reach and nothing else about it (A-R34-01).
 *
 * No lifecycle script runs: npm's and pnpm's are disabled by flag, and pnpm's
 * `.pnpmfile.cjs`, which pnpm executes despite `--ignore-scripts`, is
 * disabled by its own flag and is never copied in. Every package must carry
 * an integrity hash and resolve from a registry, so the installer verifies
 * every byte it writes; a lockfile where one does not is refused (I5).
 */
import { DEPENDENCY_AUDIT_IMAGE, PNPM_AUDIT_VERSION } from './audit.js';
import type { DependencyInstaller, LockedPackage } from './types.js';

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

/** Why a package resolved from `resolved` cannot be verified as a registry download, or null when it can. */
function notRegistry(resolved: string): string | null {
  return resolved.startsWith('https://') ? null : `it resolves from ${resolved}, not from a registry`;
}

/** An integrity string as both lockfiles write one: `<algorithm>-<base64>`, several separated by spaces. */
const INTEGRITY = /^(?:sha(?:256|384|512)-[A-Za-z0-9+/]+={0,2})(?: sha(?:256|384|512)-[A-Za-z0-9+/]+={0,2})*$/u;

function integrityOf(value: unknown): string | null {
  return typeof value === 'string' && INTEGRITY.test(value) ? value : null;
}

/** The failures a whole install's output can name, by the code npm or pnpm prints. */
const HASH_FAILED = /\bEINTEGRITY\b|ERR_PNPM_TARBALL_INTEGRITY|integrity checksum failed/u;
const NO_LOCKFILE_MATCH = /\bEUSAGE\b|ERR_PNPM_OUTDATED_LOCKFILE|ERR_PNPM_LOCKFILE_CONFIG_MISMATCH|ERR_PNPM_NO_LOCKFILE/u;
const UNREACHABLE = /ECONNREFUSED|ECONNRESET|EAI_AGAIN|ENOTFOUND|ETIMEDOUT|EHOSTUNREACH|ENETUNREACH|getaddrinfo|socket hang up|\b40[37]\b|ERR_PNPM_META_FETCH_FAIL|ERR_PNPM_FETCH_40[37]/u;

function failure(output: string): string | null {
  if (HASH_FAILED.test(output)) return 'a package failed the integrity hash its lockfile records';
  if (NO_LOCKFILE_MATCH.test(output)) return 'the lockfile does not match the manifest it was installed with';
  if (UNREACHABLE.test(output)) return 'a granted registry could not be reached';
  return null;
}

/**
 * `package-lock.json`, lockfile version 2 or 3: `packages` keyed by install
 * path. The root (`""`) is the project, a `link` is a workspace package in
 * the tree, and an `inBundle` package arrives inside its parent's tarball,
 * which is hashed; none of the three is downloaded on its own.
 */
function npmPackages(document: unknown): ReturnType<DependencyInstaller['packages']> {
  if (!isRecord(document)) return { ok: false, reason: 'package-lock.json is not a JSON object' };
  const version = document.lockfileVersion;
  if (version !== 2 && version !== 3) return { ok: false, reason: `package-lock.json is lockfile version ${JSON.stringify(version)}; only versions 2 and 3 record a hash per installed package` };
  if (!isRecord(document.packages)) return { ok: false, reason: 'package-lock.json has no packages map' };
  const packages: LockedPackage[] = [];
  for (const [key, entry] of Object.entries(document.packages)) {
    if (key === '') continue;
    if (!isRecord(entry)) return { ok: false, reason: `package-lock.json's entry ${key} is not an object` };
    if (entry.link === true || entry.inBundle === true) continue;
    const integrity = integrityOf(entry.integrity);
    const resolved = typeof entry.resolved === 'string' ? notRegistry(entry.resolved) : null;
    packages.push({ key, integrity, unverifiable: integrity === null ? 'it records no integrity hash' : resolved });
  }
  return { ok: true, packages: packages.sort((a, b) => a.key.localeCompare(b.key)) };
}

/**
 * `pnpm-lock.yaml`, lockfile version 9: `packages` keyed by `name@version`,
 * each with its `resolution`. A registry package's resolution is its
 * integrity alone; a tarball, git, or directory resolution is not a
 * registry download, whatever hash it carries.
 */
function pnpmPackages(document: unknown): ReturnType<DependencyInstaller['packages']> {
  if (!isRecord(document)) return { ok: false, reason: 'pnpm-lock.yaml is not a mapping' };
  if (document.lockfileVersion !== '9.0') return { ok: false, reason: `pnpm-lock.yaml is lockfile version ${JSON.stringify(document.lockfileVersion)}; only 9.0 is read` };
  if (document.packages === undefined || document.packages === null) return { ok: true, packages: [] };
  if (!isRecord(document.packages)) return { ok: false, reason: "pnpm-lock.yaml's packages is not a mapping" };
  const packages: LockedPackage[] = [];
  for (const [key, entry] of Object.entries(document.packages)) {
    if (!isRecord(entry) || !isRecord(entry.resolution)) return { ok: false, reason: `pnpm-lock.yaml's entry ${key} has no resolution` };
    const resolution = entry.resolution;
    const integrity = integrityOf(resolution.integrity);
    let unverifiable: string | null = integrity === null ? 'it records no integrity hash' : null;
    if (typeof resolution.type === 'string') unverifiable = `it is a ${resolution.type} dependency, not a registry download`;
    else if (typeof resolution.tarball === 'string') unverifiable ??= notRegistry(resolution.tarball);
    packages.push({ key, integrity, unverifiable });
  }
  return { ok: true, packages: packages.sort((a, b) => a.key.localeCompare(b.key)) };
}

+/**
+ * A stand-in for npm's install over a sandbox that runs no image (R34): a
+ * `node` script that reads `package-lock.json` in the install's working
+ * directory and copies each package it pins, by install path, from
+ * `registry`, a directory on the host laid out as an installed tree. A
+ * package the registry does not hold fails the install. It is selected by
+ * the lockfile npm's is, so the runtime plans, refuses, and records it as
+ * it does npm's; the graph builder declares it unsafe, so no run on it goes
+ * above L1. It fetches nothing and verifies no hash.
+ */
+export function standInNpmInstaller(registry: string): DependencyInstaller {
+  const script = [
+    "const fs = require('node:fs'); const path = require('node:path');",
+    `const registry = ${JSON.stringify(registry)};`,
+    "const lock = JSON.parse(fs.readFileSync('package-lock.json', 'utf8'));",
+    "for (const key of Object.keys(lock.packages).filter((k) => k !== '')) {",
+    '  const from = path.join(registry, ...key.split("/"));',
+    "  if (!fs.existsSync(from)) { console.error(`stand-in registry holds no ${key}`); process.exit(1); }",
+    '  fs.cpSync(from, key, { recursive: true, verbatimSymlinks: true });',
+    '}',
+  ].join('\n');
+  return Object.freeze({ ...NPM_INSTALLER, image: STAND_IN_INSTALL_IMAGE, invocation: Object.freeze(['node', '-e', script] as const) });

+/**

+ * lockfile the runtime's hash rule accepts, for a fixture that installs with
+ * the stand-in.

+export function hashedLockfile(name: string, packages: readonly string[]): string {

 * check of a task, the runtime installs from the tree's lockfile in a sandbox

async function isFile(path: string): Promise<boolean> {
  return (await stat(path).catch(() => null))?.isFile() === true;
}

function sha256(bytes: string | Buffer): string {
  return createHash('sha256').update(bytes).digest('hex');
}

/** A lockfile's packages, read by the installer from the document the runtime parsed (D-R34-07). */
export function lockedPackages(installer: DependencyInstaller, text: string): ReturnType<DependencyInstaller['packages']> {
  let document: unknown;
  try {
    document = installer.format === 'json' ? (JSON.parse(text) as unknown) : (parseYaml(text) as unknown);
  } catch (error) {
    return { ok: false, reason: `${installer.lockfile} is not ${installer.format === 'json' ? 'JSON' : 'YAML'}: ${error instanceof Error ? error.message : String(error)}` };
  }
  return installer.packages(document);
}

/** At most this many packages are named in a refusal; the rest are counted. */
const NAMED = 5;

/**
 * What `tree` declares to install, read from its root alone (D-R34-04): one
 * lockfile an installer reads, with a registry granted and every package
 * hashed, or nothing to install when the root holds neither a lockfile nor a
 * manifest any installer reads. Anything else is a refusal naming what was
 * found.
 */
export async function planInstall(tree: string, installers: readonly DependencyInstaller[], registries: readonly string[]): Promise<InstallPlan> {
  const names = [...new Set([...installers.map((i) => i.lockfile), ...UNSUPPORTED_LOCKFILES])];
  const found: string[] = [];
  for (const name of names) if (await isFile(join(tree, name))) found.push(name);
  if (found.length > 1) return { kind: 'refused', reason: `the tree holds ${found.join(' and ')}; which one installs it is not decided` };
  const [lockfile] = found;
  if (lockfile === undefined) {
    const manifests = [...new Set(installers.map((i) => i.manifest))];
    for (const manifest of manifests) {
      if (await isFile(join(tree, manifest))) return { kind: 'refused', reason: `the tree holds ${manifest} and no lockfile, so its dependencies cannot be installed as pinned` };
    }
    return { kind: 'none' };
  }
  const installer = installers.find((i) => i.lockfile === lockfile);
  if (installer === undefined) return { kind: 'refused', reason: `${lockfile} is a lockfile no installer reads` };
  if (registries.length === 0) {
    return { kind: 'refused', reason: `the tree holds ${lockfile}, and policy grants no registry the install may reach (installRegistries is empty), so nothing is installed` };
  }
  const lockfileText = await readFile(join(tree, lockfile), 'utf8');
  const read = lockedPackages(installer, lockfileText);
  if (!read.ok) return { kind: 'refused', reason: read.reason };
  const unverifiable = read.packages.filter((p) => p.unverifiable !== null);
  if (unverifiable.length > 0) {
    const named = unverifiable.slice(0, NAMED).map((p) => `${p.key} (${p.unverifiable ?? ''})`).join('; ');
    const more = unverifiable.length > NAMED ? `; and ${String(unverifiable.length - NAMED)} more` : '';
    return { kind: 'refused', reason: `${lockfile} does not pin every package by a hash an install can verify: ${named}${more}` };
  }
  return { kind: 'install', installer, lockfileText, packages: read.packages };
}

/** Whether a directory name is one of the dependency trees `installers` write, in any case (D-R15-07). */
function treeNamer(installers: readonly DependencyInstaller[]): (name: string) => boolean {
  const names = new Set(installers.flatMap((i) => i.trees.map((t) => t.toLowerCase())));
  return (name) => names.has(name.toLowerCase());
}

/** Every dependency tree under `root`, by its path relative to it, outside VCS directories. Not entered. */
async function dependencyTrees(root: string, isTree: (name: string) => boolean, at = ''): Promise<string[]> {
  const out: string[] = [];
  for (const entry of await readdir(join(root, ...at.split('/').filter(Boolean)), { withFileTypes: true })) {
    if (!entry.isDirectory() || VCS_DIRECTORIES.has(entry.name)) continue;
    const path = at === '' ? entry.name : `${at}/${entry.name}`;
    if (isTree(entry.name)) out.push(path);
    else out.push(...(await dependencyTrees(root, isTree, path)));
  }
  return out;
}

/** Every file the installer reads that `tree` holds: its root files, and its nested manifest in every directory outside a dependency tree. */
async function inputFiles(tree: string, installer: DependencyInstaller, isTree: (name: string) => boolean): Promise<string[]> {
  const files: string[] = [];
  for (const name of installer.files) if (await isFile(join(tree, name))) files.push(name);
  const nested = installer.nested;
  if (nested === null) return files;
  const walk = async (at: string): Promise<void> => {
    for (const entry of await readdir(join(tree, ...at.split('/')), { withFileTypes: true })) {
      if (!entry.isDirectory() || VCS_DIRECTORIES.has(entry.name) || isTree(entry.name)) continue;
      const path = at === '' ? entry.name : `${at}/${entry.name}`;
      if (await isFile(join(tree, path, nested))) files.push(`${path}/${nested}`);
      await walk(path);
    }
  };
  await walk('');
  return files.sort();
}

/** The lockfile's diff against the base's, by package key. A base lockfile that does not read is diffed as empty. */
function lockfileDiff(head: readonly LockedPackage[], base: readonly LockedPackage[]): InstallRecord['lockfileDiff'] {
  const before = new Map(base.map((p) => [p.key, p.integrity]));
  const after = new Map(head.map((p) => [p.key, p.integrity]));
  return {
    added: [...after.keys()].filter((k) => !before.has(k)).sort(),
    removed: [...before.keys()].filter((k) => !after.has(k)).sort(),
    changed: [...after.entries()].filter(([k, v]) => before.has(k) && before.get(k) !== v).map(([k]) => k).sort(),
  };
}

/** The digest of every dependency tree an install wrote, each by its path. */
async function installedDigest(dir: string, trees: readonly string[]): Promise<string> {
  const hash = createHash('sha256');
  for (const path of trees) hash.update(`${path}\0${await treeDigest(join(dir, ...path.split('/')))}\n`);
  return hash.digest('hex');
}

/** Where an install the run already made keeps what it recorded, beside the trees it wrote. */
const RECORD = 'install.json';
const ROOT = 'root';

/** The outcome of installing into a tree: the record, nothing to install, or why its checks cannot start. */
export type InstallOutcome =
  | { readonly ok: true; readonly install: InstallRecord | null }
  | { readonly ok: false; readonly at: 'plan' | 'provision' | 'install'; readonly reason: string };

/**
 * Replaces every dependency tree in `tree` with the runtime's install of its
 * lockfile, made once per run for each set of inputs (D-R34-08). `tree` is a
 * tree the runtime built for checks to run over. A tree that declares
 * nothing to install keeps no dependency tree at all, so what the base
 * installed never decides a check (D-R34-04).
 */
export async function installInto(ctx: LineContext, tree: string, base: string, taskId: TaskId | null, station: StationId): Promise<InstallOutcome> {
  const { installers } = ctx.components;
  const isTree = treeNamer(installers);
  for (const path of await dependencyTrees(tree, isTree)) await rm(join(tree, ...path.split('/')), { recursive: true, force: true });
  const plan = await planInstall(tree, installers, ctx.policy.installRegistries);
  if (plan.kind === 'none') return { ok: true, install: null };
  if (plan.kind === 'refused') return { ok: false, at: 'plan', reason: `no dependency was installed: ${plan.reason}` };
  const { installer } = plan;

  const inputs = await inputFiles(tree, installer, isTree);
  const key = createHash('sha256');
  key.update(`${installer.stack}\0${installer.image}\0${JSON.stringify(installer.invocation)}\0${JSON.stringify(installer.env)}\n`);
  for (const path of inputs) key.update(`${path}\0${sha256(await readFile(join(tree, ...path.split('/'))))}\n`);
  const dir = installPath(ctx.components.workspaces, ctx.run.id, key.digest('hex'));

  let record: InstallRecord;
  if (await isFile(join(dir, RECORD))) {
    record = JSON.parse(await readFile(join(dir, RECORD), 'utf8')) as InstallRecord;
  } else {
    const made = await install(ctx, plan, inputs, tree, base, dir, taskId, station);
    if (!made.ok) return made;
    record = made.install;
  }
  for (const path of await dependencyTrees(join(dir, ROOT), isTree)) {
    await cp(join(dir, ROOT, ...path.split('/')), join(tree, ...path.split('/')), { recursive: true, errorOnExist: true, force: false, verbatimSymlinks: true });
  }
  return { ok: true, install: record };
}

async function install(

): Promise<{ ok: true; install: InstallRecord } | { ok: false; at: 'provision' | 'install'; reason: string }> {
  const { installer } = plan;
  const { sandbox, workspaces, profile, vault } = ctx.components;
  // Built beside its final place and renamed in, so two tasks installing the same inputs at once never share a directory.
  const building = `${dir}.${randomUUID()}`;

    await mkdir(dirname(target), { recursive: true });
    await copyFile(join(tree, ...path.split('/')), target);

    egress: { mode: 'allowlist', allow: [...ctx.policy.installRegistries] },

  } catch (error) {

  }
  if (failed !== undefined) {
    await rm(building, { recursive: true, force: true });
    return { ok: false, at: 'install', reason: failed };
  }

    image: installer.image,

    baseLockfileSha256: baseText === null ? null : sha256(baseText),

    installedTreeSha256: await installedDigest(work, trees),
    collectedBy: 'runtime',
  };
  await writeFile(join(building, RECORD), JSON.stringify(record));
  try {

  } catch {

    await rm(building, { recursive: true, force: true });

async function files(entries: Record<string, string>): Promise<void> {
  for (const [path, text] of Object.entries(entries)) {
    await mkdir(join(root, ...path.split('/').slice(0, -1)), { recursive: true });
    await writeFile(join(root, ...path.split('/')), text);
  }
}

async function refusal(registries = GRANTED): Promise<string> {
  const plan = await planInstall(root, INSTALLERS, registries);
  if (plan.kind !== 'refused') throw new Error(`expected a refusal, got ${plan.kind}`);
  return plan.reason;
}

/**
 * R34: the runtime installs a tree's dependencies from its lockfile before
 * any check runs, in a sandbox of its own whose egress is the registries
 * policy grants, and every check runs over that install with no egress.
 *
 * The rig's stand-in installer copies what a lockfile pins from the
 * fixture's own tree, offline (`standInNpmInstaller`); the base's own
 * `node_modules` is never what a check runs over. The lifecycle-script
 * assertion runs npm's declared invocation in its declared image, with no
 * network, against a dependency whose `postinstall` would leave a mark.
 */
import { execFile } from 'node:child_process';
import { mkdir, readdir, writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import { promisify } from 'node:util';
import type { ExecOptions, ExecResult, SandboxCapabilities, SandboxHandle, SandboxProvider, SandboxSpec, Teardown } from '@olympus-ai/sandbox';

const api = async () => import('@olympus-ai/api');

const docker = promisify(execFile);

/** Fails on a tree whose install does not hold `extra`, and passes on one that does. */
const NEEDS_EXTRA = { id: 'needs-extra', kind: 'behavioral', onBase: 'fails', command: ['node', '-e', "require('extra')"], required: true, timeoutMs: 10_000 };
const EXTRA = 'node_modules/extra';

/** Every file under `root`, by its path relative to it. */
async function listing(root: string, at = ''): Promise<string[]> {
  const out: string[] = [];
  for (const entry of await readdir(join(root, ...at.split('/').filter(Boolean)), { withFileTypes: true })) {
    const path = at === '' ? entry.name : `${at}/${entry.name}`;
    if (entry.isDirectory()) out.push(...(await listing(root, path)));
    else out.push(path);
  }
  return out.sort();
}

/**
 * The stub sandbox, recording each sandbox asked for and the files its
 * workspace held when it was provisioned; an exec in a sandbox of `scripted`'s
 * image is answered with its result instead of run.
 */
class Recording implements SandboxProvider {
  readonly id = 'recording-stub';
  readonly provisioned: Array<{ readonly spec: SandboxSpec; readonly listed: readonly string[] }> = [];
  readonly #scriptedHandles = new Set<SandboxHandle>();
  constructor(private readonly inner: SandboxProvider, private readonly scripted?: { readonly image: string; readonly result: ExecResult }) {}
  async provision(spec: SandboxSpec): Promise<SandboxHandle> {
    this.provisioned.push({ spec, listed: await listing(spec.mounts.workspace.source) });
    const handle = await this.inner.provision(spec);
    if (spec.image === this.scripted?.image) this.#scriptedHandles.add(handle);
    return handle;
  }
  exec(h: SandboxHandle, cmd: string[], options?: ExecOptions): Promise<ExecResult> {
    if (this.#scriptedHandles.has(h) && this.scripted !== undefined) return Promise.resolve(this.scripted.result);
    return this.inner.exec(h, cmd, options);
  }
  destroy(h: SandboxHandle): Promise<Teardown> {
    return this.inner.destroy(h);
  }
  capabilities(): SandboxCapabilities {
    return this.inner.capabilities();
  }
}

async function recording(scripted?: { image: string; result: ExecResult }): Promise<Recording> {
  const { StubSandboxProvider } = await import('@olympus-ai/sandbox');
  return new Recording(new StubSandboxProvider(), scripted);
}

/**
 * A base that declares dependencies: a manifest and `lockfile`, beside a
 * `.npmrc` and a `.pnpmfile.cjs` no install may read. The fixture's tree
 * also holds `extra` under `node_modules`, where the stand-in installs it
 * from, so a check that ran over the base's own trees would find it.
 */
async function withDependencies(rig: LineRig, lockfile: string): Promise<void> {
  const root = rig.dirs.artifacts;
  await writeFile(join(root, 'package.json'), JSON.stringify({ name: 'hello', private: true }));
  await writeFile(join(root, 'package-lock.json'), lockfile);
  await writeFile(join(root, '.npmrc'), 'registry=https://registry.example.invalid/\n');
  await writeFile(join(root, '.pnpmfile.cjs'), "require('node:fs').writeFileSync('/tmp/pnpmfile-ran', '')\n");
  await mkdir(join(root, EXTRA), { recursive: true });
  await writeFile(join(root, EXTRA, 'index.js'), "module.exports = 'extra';\n");
}

/** The rig's stub builder, made to write `files` into its tree as an agent would. */
async function writing(files: Readonly<Record<string, string>>, sandbox: SandboxProvider): Promise<Driver> {
  const driver = await stubDriver();
  return new Proxy(driver, {
    get(target, key, receiver) {
      if (key !== 'runTask') return Reflect.get(target, key, receiver) as unknown;
      return async (req: TaskRequest): Promise<TaskResult> => {
        if (req.role.role === 'builder') {
          const script = `const fs = require('node:fs'); for (const [f, t] of Object.entries(${JSON.stringify(files)})) fs.writeFileSync(f, t)`;
          const wrote = await sandbox.exec(req.sandbox, ['node', '-e', script]);
          if (wrote.exitCode !== 0) throw new Error(`the builder's write did not land: ${wrote.stderr}`);
        }
        return target.runTask(req);
      };
    },
  });
}

/** Runs the rig's change adding `extra` to the lockfile, and returns the sandboxes asked for and the task's last evidence. */
async function addingExtra(rig: LineRig): Promise<{ sandbox: Recording; bundle: EvidenceBundle | undefined; status: string | undefined; baseRecord: BaseRunRecord }> {
  const { startRun } = await api();
  const { hashedLockfile } = await adapters();
  await withDependencies(rig, hashedLockfile('hello', []));
  await writeManifest(rig.dirs, [NEEDS_EXTRA]);
  const sandbox = await recording();
  const driver = await writing({ 'package-lock.json': hashedLockfile('hello', [EXTRA]) }, sandbox);
  const components = await rig.components({ sandbox, driver, reviewer: driver }, { work: false });
  await startRun(await rig.request(components));
  const state = await rig.state();
  const bundles = await Promise.all(state.evidenceRefs.map((r) => readRecord<EvidenceBundle>(components.vault, r)));
  const baseRun = await components.vault.readBaseRun(rig.runId, null);
  if (baseRun === null) throw new Error('R34: the run wrote no base record');
  return { sandbox, bundle: bundles.at(-1), status: state.tasks[HELLO_TASK], baseRecord: await readRecord<BaseRunRecord>(components.vault, baseRun) };
}
