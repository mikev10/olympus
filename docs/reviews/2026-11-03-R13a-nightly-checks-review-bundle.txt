R13a: nightly checks without a model. Two job templates, `nightly-integrity` and `nightly-security`, run on a schedule instead of recording a candidate, and each writes one `NightlyRecord` beside the runs. The integrity job re-hashes every object the Vault holds, which the new `listObjects` lists. It also checks every ref and every in-flight run's locks, and compares R2's rates with the previous night's snapshot. The security job runs R17's scanners over the default branch's head under `deny-all`, and audits its npm or pnpm lockfile in a sandbox that can reach only the advisory host the policy names. A part that cannot finish is recorded as a refusal, never as finding nothing. Each finding becomes an `intake` candidate whose fields are typed under the `nightly-finding` schema, and drift that names a run goes into P14's record as `record-drift`. `GET /nightly` returns the records. `TriggerPolicy` gains an optional `nightly` block, `CandidateRecord` gains `schema`, and `TriggerDecision` gains the `job` outcome.

/**
 * The dependency audits the nightly sweep runs (R13a, D-R13a-06). Each is
 * declared once, here: the stack, the root files it reads, the image it is
 * pinned to, the arguments, and how its report becomes findings. A policy
 * names the one host its sandbox may reach and nothing else about it
 * (D-A-NR-04, D-A-NR-08).
 *
 * A report is read only when it is whole. An output that is not one, which
 * is what an unreachable advisory host produces, is never an empty list
 * (I5).
 */
import { createHash } from 'node:crypto';
import type { AuditFinding, AuditStack, DependencyAuditor } from './types.js';

/** `node:22-bookworm-slim`, by digest: Node 22.23.2 and npm 10.9.8, the image the drivers are built from. */
export const DEPENDENCY_AUDIT_IMAGE = 'node@sha256:83f487e0a63425e5b4d146fb5e5be574bcbe1b7b843d3ebafdd95eaf7767a7e5';

/** The pnpm an audit runs, by exact version: corepack would otherwise fetch the newest. */
export const PNPM_AUDIT_VERSION = '10.18.3';

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function parseJson(text: string): unknown {
  try {
    return JSON.parse(text) as unknown;
  } catch {
    return undefined;
  }
}

function fingerprint(stack: AuditStack, rule: string, name: string): string {
  return createHash('sha256').update(`${stack}\0${rule}\0${name}`).digest('hex');
}

const GHSA = /GHSA-[0-9a-z]{4}-[0-9a-z]{4}-[0-9a-z]{4}/u;

/** An advisory's id as a rule: its GHSA id, else its numeric id under a fixed prefix. */
function ruleOf(url: unknown, ghsa: unknown, id: unknown): string | undefined {
  if (typeof ghsa === 'string' && GHSA.test(ghsa)) return ghsa;
  if (typeof url === 'string') {
    const match = GHSA.exec(url)?.[0];
    if (match !== undefined) return match;
  }
  return typeof id === 'number' && Number.isInteger(id) && id > 0 ? `advisory-${String(id)}` : undefined;
}

/** A failed request, as npm and pnpm print one: the request, or a network error code. */
const REQUEST_FAILED = /request to |ECONNREFUSED|ECONNRESET|EAI_AGAIN|ENOTFOUND|ETIMEDOUT|EHOSTUNREACH|ENETUNREACH|getaddrinfo|socket hang up|\b40[37]\b/u;

/** Whether an output that is not a whole report says the advisory request failed. */
function unreachable(stdout: string, parsed: unknown): boolean {
  if (isRecord(parsed) && Object.hasOwn(parsed, 'error')) return REQUEST_FAILED.test(JSON.stringify(parsed));
  return REQUEST_FAILED.test(stdout);
}

function escape(text: string): string {
  return text.replace(/[.*+?^${}()|[\]\\]/gu, '\\$&');
}

/** The first line, 1-based, matching `pattern`; null where none does. */
function lineOf(text: string, pattern: RegExp): number | null {
  const lines = text.split(/\r?\n/u);
  const index = lines.findIndex((line) => pattern.test(line));
  return index === -1 ? null : index + 1;
}

/**
 * `npm audit --json --package-lock-only`: report version 2. Each vulnerable
 * package lists the advisories against it under `via`; a `via` that is a
 * string names another vulnerable package, which carries the advisory itself.
 */
export const NPM_AUDITOR: DependencyAuditor = Object.freeze({
  stack: 'npm',
  lockfile: 'package-lock.json',
  files: Object.freeze(['package.json', 'package-lock.json']),
  image: DEPENDENCY_AUDIT_IMAGE,
  invocation: Object.freeze(['npm', 'audit', '--json', '--package-lock-only', '--fetch-retries=0'] as const),
  parse(stdout: string, lockfileText: string) {
    const parsed = parseJson(stdout);
    if (!isRecord(parsed) || Object.hasOwn(parsed, 'error') || parsed.auditReportVersion !== 2 || !isRecord(parsed.vulnerabilities) || !isRecord(parsed.metadata)) {
      return { ok: false, unreachable: unreachable(stdout, parsed) } as const;
    }
    const findings = new Map<string, AuditFinding>();
    for (const entry of Object.values(parsed.vulnerabilities)) {
      if (!isRecord(entry) || !Array.isArray(entry.via)) return { ok: false, unreachable: false } as const;
      for (const via of entry.via as unknown[]) {
        if (typeof via === 'string') continue;
        if (!isRecord(via) || typeof via.name !== 'string') return { ok: false, unreachable: false } as const;
        const rule = ruleOf(via.url, undefined, via.source);
        if (rule === undefined) return { ok: false, unreachable: false } as const;
        const name = via.name;
        const print = fingerprint('npm', rule, name);
        if (findings.has(print)) continue;
        const line = lineOf(lockfileText, new RegExp(`"(?:[^"]*/)?node_modules/${escape(name)}"\\s*:\\s*\\{`, 'u'));
        findings.set(print, { rule, packageName: name, line, fingerprint: print });
      }
    }
    return { ok: true, findings: [...findings.values()].sort((a, b) => a.fingerprint.localeCompare(b.fingerprint)) } as const;
  },
});

/** `pnpm audit --json`: `advisories` by id, each naming its module and the versions found. */
export const PNPM_AUDITOR: DependencyAuditor = Object.freeze({
  stack: 'pnpm',
  lockfile: 'pnpm-lock.yaml',
  files: Object.freeze(['package.json', 'pnpm-lock.yaml', 'pnpm-workspace.yaml']),
  image: DEPENDENCY_AUDIT_IMAGE,
  invocation: Object.freeze(['corepack', `pnpm@${PNPM_AUDIT_VERSION}`, 'audit', '--json'] as const),
  parse(stdout: string, lockfileText: string) {
    const parsed = parseJson(stdout);
    if (!isRecord(parsed) || Object.hasOwn(parsed, 'error') || !isRecord(parsed.advisories) || !isRecord(parsed.metadata)) {
      return { ok: false, unreachable: unreachable(stdout, parsed) } as const;
    }
    const findings = new Map<string, AuditFinding>();
    for (const advisory of Object.values(parsed.advisories)) {
      if (!isRecord(advisory) || typeof advisory.module_name !== 'string' || !Array.isArray(advisory.findings)) return { ok: false, unreachable: false } as const;
      const rule = ruleOf(advisory.url, advisory.github_advisory_id, advisory.id);
      if (rule === undefined) return { ok: false, unreachable: false } as const;
      const name = advisory.module_name;
      const print = fingerprint('pnpm', rule, name);
      if (findings.has(print)) continue;
      const first: unknown = advisory.findings[0];
      const version = isRecord(first) && typeof first.version === 'string' ? first.version : null;
      const line = version === null ? null : lineOf(lockfileText, new RegExp(`^\\s+['"]?/?${escape(name)}@${escape(version)}(?:\\(.*\\))?['"]?:\\s*$`, 'u'));
      findings.set(print, { rule, packageName: name, line, fingerprint: print });
    }
    return { ok: true, findings: [...findings.values()].sort((a, b) => a.fingerprint.localeCompare(b.fingerprint)) } as const;
  },
});

/**
 * The dependency auditors (R13a, D-R13a-06). Outputs are as npm 10.9.8 and
 * pnpm 10.18.3 printed them in the pinned image (`docs/decisions.md`,
 * D-R13a-02). A report is read only when whole; anything else is never an
 * empty list (I5).
 */
import { describe, expect, test } from 'vitest';
import { NPM_AUDITOR, PNPM_AUDITOR } from '../src/index.js';

const NPM_LOCK = '{\n  "packages": {\n    "": {},\n    "node_modules/minimist": {\n      "version": "1.2.5"\n    }\n  }\n}\n';
const NPM_REPORT = JSON.stringify({
  auditReportVersion: 2,
  vulnerabilities: {
    minimist: {
      name: 'minimist', severity: 'critical',
      via: [{ source: 1097678, name: 'minimist', title: 'Prototype Pollution in minimist', url: 'https://github.com/advisories/GHSA-xvch-5gv4-984h' }],
      effects: [], nodes: ['node_modules/minimist'],
    },
    parent: { name: 'parent', severity: 'critical', via: ['minimist'], effects: [] },
  },
  metadata: { vulnerabilities: { critical: 1, total: 1 } },
});

const PNPM_LOCK = "lockfileVersion: '9.0'\n\npackages:\n\n  minimist@1.2.5:\n    resolution: {integrity: sha512-x}\n";
const PNPM_REPORT = JSON.stringify({
  advisories: {
    1097678: { id: 1097678, module_name: 'minimist', github_advisory_id: 'GHSA-xvch-5gv4-984h', url: 'https://github.com/advisories/GHSA-xvch-5gv4-984h', findings: [{ version: '1.2.5', paths: ['.>minimist'] }] },
  },
  metadata: { vulnerabilities: { critical: 1 } },
});

describe('npm audit', () => {
  test('a whole report is one finding per advisory and package, located in the lockfile', () => {
    const parsed = NPM_AUDITOR.parse(NPM_REPORT, NPM_LOCK);
    expect(parsed.ok).toBe(true);
    if (!parsed.ok) return;
    expect(parsed.findings).toHaveLength(1);
    expect(parsed.findings[0]).toMatchObject({ rule: 'GHSA-xvch-5gv4-984h', packageName: 'minimist', line: 4 });
  });

  test('an empty report is no findings, and is whole', () => {
    const parsed = NPM_AUDITOR.parse(JSON.stringify({ auditReportVersion: 2, vulnerabilities: {}, metadata: {} }), NPM_LOCK);
    expect(parsed).toEqual({ ok: true, findings: [] });
  });

  test('a failed advisory request is unreachable, never zero findings', () => {
    const out = JSON.stringify({ message: 'request to https://registry.npmjs.org/-/npm/v1/security/audits/quick failed, reason: getaddrinfo EAI_AGAIN registry.npmjs.org', error: { summary: '', detail: '' } });
    expect(NPM_AUDITOR.parse(out, NPM_LOCK)).toEqual({ ok: false, unreachable: true });
  });

  test('output that is not a report is not one', () => {
    expect(NPM_AUDITOR.parse('npm ERR! something', NPM_LOCK)).toEqual({ ok: false, unreachable: false });
    expect(NPM_AUDITOR.parse('', NPM_LOCK)).toEqual({ ok: false, unreachable: false });
  });

  test('a package the lockfile does not name has no line, never an invented one', () => {
    const parsed = NPM_AUDITOR.parse(NPM_REPORT, '{}');
    expect(parsed.ok && parsed.findings[0]?.line).toBeNull();
  });
});

describe('pnpm audit', () => {
  test('a whole report is located at the package entry', () => {
    const parsed = PNPM_AUDITOR.parse(PNPM_REPORT, PNPM_LOCK);
    expect(parsed.ok).toBe(true);
    if (!parsed.ok) return;
    expect(parsed.findings[0]).toMatchObject({ rule: 'GHSA-xvch-5gv4-984h', packageName: 'minimist', line: 5 });
  });

  test('an unreachable registry is unreachable', () => {
    const out = JSON.stringify({ error: { code: 'ECONNREFUSED', message: 'request to http://127.0.0.1:9/-/npm/v1/security/audits failed, reason: connect ECONNREFUSED 127.0.0.1:9' } });
    expect(PNPM_AUDITOR.parse(out, PNPM_LOCK)).toEqual({ ok: false, unreachable: true });
  });

  test('the version is pinned in the invocation', () => {
    expect(PNPM_AUDITOR.invocation.join(' ')).toMatch(/^corepack pnpm@\d+\.\d+\.\d+ audit --json$/u);
  });
});

/**
 * The nightly jobs that call no model (R13a). A schedule that names a job
 * template runs one of them in place of recording a candidate; each writes
 * one `NightlyRecord` beside the runs, and each reads the schedule's id from
 * the event and nothing else (A-R13a-01).
 *
 * - `nightly-integrity` re-hashes every object the store holds, checks every
 *   ref a run's state names, verifies the locks of a run still in flight, and
 *   checks R2's rates against the last snapshot (D-A-NR-03, D-A-NR-07).
 * - `nightly-security` runs R17's scanners over the head of the target's
 *   default branch under `deny-all`, and audits its lockfile in a separate
 *   sandbox granted the advisory host alone (D-A-NR-04, D-R13a-04).

 * beside it, and never "nothing found" (I5). A finding becomes an `intake`

 * (D-A-SC-03, I2). Every field written is the runtime's.

/** Endings no later step changes: a snapshot covers these runs alone (D-R13a-03). */
export const SETTLED: ReadonlySet<RunEnding> = new Set<RunEnding>(['refused', 'merged', 'cancelled']);

/** How long one scan or the audit may run. */
export const NIGHTLY_STEP_TIMEOUT_MS = 15 * 60_000;

function sha256(bytes: Uint8Array): string {
  return createHash('sha256').update(bytes).digest('hex');
}

function describe(error: unknown): string {
  if (error instanceof Error) return error.message;
  return typeof error === 'string' ? error : 'unknown error';
}

/** Every leaf of a JSON value, by its dotted path, as JSON: two snapshots compare figure by figure. */
export function flattenFigures(value: unknown, prefix = '', out: Record<string, string> = {}): Record<string, string> {
  if (value !== null && typeof value === 'object') {
    const entries = Array.isArray(value) ? value.map((v, i) => [String(i), v] as const) : Object.entries(value as Record<string, unknown>);
    if (entries.length === 0) out[prefix] = JSON.stringify(value);
    for (const [key, child] of entries) flattenFigures(child, prefix === '' ? key : `${prefix}.${key}`, out);
    return out;
  }
  out[prefix] = value === undefined ? 'undefined' : JSON.stringify(value);
  return out;
}

/** Every ref a run's state names, each with the path it is stored under. */
function refsOf(state: RunState): VaultRef[] {
  const refs: VaultRef[] = [state.admission, ...Object.values(state.results), ...state.evidenceRefs, ...state.violations, ...state.usage];
  for (const draft of Object.values(state.drafts)) {
    for (const attempt of draft.attempts) {
      if (attempt.draft !== null) refs.push(attempt.draft);
      if (attempt.critic !== null && attempt.critic !== 'not-seated') refs.push(attempt.critic);
      if (attempt.baseRun !== null) refs.push(attempt.baseRun);
    }
    if (draft.locked !== null) refs.push(draft.locked);
  }
  return refs;
}

function pathOf(ref: VaultRef): string {
  return ref.kind === 'admission' ? `runs/${ref.runId}/admission.json` : `runs/${ref.runId}/objects/${ref.kind}/${ref.hash}.json`;
}

/** A refused-delivery count is rewritten while its hour is open: one for the hour the last night started in, or later, may still move. */
function stillOpen(path: string, priorStartedAt: string): boolean {
  const hour = /^intake\/counts\/([^/]+)\//u.exec(path)?.[1];
  return hour !== undefined && hour >= priorStartedAt.slice(0, 13);
}

type Drift = Extract<NightlyFinding, { kind: 'record-drift' }>;

function drift(runId: RunId | null, record: string, problem: RecordDriftProblem, expected: string, observed: string): Drift {
  return { kind: 'record-drift', runId, record, problem, expected, observed };
}

/** The refusals as the record's tuple, or null where there are none. */
function refusedWith(refusals: readonly NightlyRefusal[]): NightlyRecord['outcome'] {
  const [first, ...rest] = refusals;
  return first === undefined ? { kind: 'completed' } : { kind: 'refused', refusals: [first, ...rest] };
}

export class NightlyJobs {
  readonly #options: NightlyOptions;
  readonly #now: () => number;

  constructor(options: NightlyOptions) {
    this.#options = options;
    this.#now = options.now ?? Date.now;
  }

  /** Runs the schedule's job for `date` and writes its record; returns the record and its hash. */
  async run(policy: Policy, schedule: TriggerSchedule & { readonly template: JobTemplateId }, date: string, intake: Intake): Promise<{ record: NightlyRecord; sha256: string }> {
    const startedAt = new Date(this.#now()).toISOString();
    const part = schedule.template === 'nightly-integrity'
      ? await this.#integrity(schedule, date)
      : await this.#security(policy, schedule, date, intake);
    const record: NightlyRecord = {
      schedule: schedule.id, template: schedule.template, date, startedAt, endedAt: new Date(this.#now()).toISOString(),
      outcome: refusedWith(part.refusals), findings: part.findings,
      integrity: part.integrity, security: part.security, collectedBy: 'runtime',
    };
    return { record, sha256: await this.#options.vault.recordNightly(record) };
  }

  async #integrity(schedule: TriggerSchedule, date: string): Promise<Part> {
    const { vault } = this.#options;
    const findings: NightlyFinding[] = [];
    const refusals: NightlyRefusal[] = [];
    const prior = [...(await vault.readNightly())]
      .filter((e) => e.record.schedule === schedule.id && e.record.integrity !== null && e.record.date < date)
      .at(-1);

    // Re-hash: the store names every object, and each content-addressed one is checked against its name.
    const hashes: Record<string, string> = {};
    let objects = 0;
    let storeRead = true;
    try {
      for await (const object of vault.listObjects()) {
        objects += 1;
        const observed = sha256(object.bytes);
        hashes[object.path] = observed;
        if (object.named === 'content') {
          const expected = /([0-9a-f]{64})\.json$/u.exec(object.path)?.[1] ?? '';
          if (observed !== expected) findings.push(drift(object.runId, object.path, 'hash-mismatch', expected, observed));
        }
      }
    } catch (error) {
      storeRead = false;
      refusals.push({ part: 'store', cause: 'store-unreadable', detail: describe(error) });
    }

    // Against the last night: a record that changed or went missing since it was hashed.
    const priorCoverage = prior?.record.integrity ?? null;
    if (storeRead && prior !== undefined && priorCoverage !== null) {
      for (const [path, expected] of Object.entries(priorCoverage.hashes)) {
        if (stillOpen(path, prior.record.startedAt)) continue;
        const observed = hashes[path];
        const runId = /^runs\/([^/]+)\//u.exec(path)?.[1] as RunId | undefined;
        if (observed === undefined) findings.push(drift(runId ?? null, path, 'missing', expected, 'missing'));
        else if (observed !== expected) findings.push(drift(runId ?? null, path, 'changed-since-snapshot', expected, observed));
      }
    }

    // Refs, locks, and rates, run by run.
    const records = new Map<RunId, RunRecords>();
    let ratesRead = true;
    for (const runId of await vault.listRuns()) {
      let state: RunState | null = null;
      try {
        state = await vault.readRunState(runId);
      } catch (error) {
        if (!isNoRunState(error)) {
          refusals.push({ part: 'store', cause: 'store-unreadable', detail: `run ${runId}: ${describe(error)}` });
          continue;
        }
      }
      if (storeRead && state !== null) {
        for (const ref of refsOf(state)) {
          const path = pathOf(ref);
          const observed = hashes[path] ?? (ref.kind === 'base-run' ? hashes[`runs/${runId}/base-run.json`] : undefined);
          if (observed === undefined) findings.push(drift(runId, path, 'missing', ref.hash, 'missing'));
          else if (observed !== ref.hash) findings.push(drift(runId, path, 'hash-mismatch', ref.hash, observed));
        }
      }
      let run: RunRecords;
      try {
        run = await readRunRecords(vault, runId);
      } catch (error) {
        ratesRead = false;
        refusals.push({ part: 'rates', cause: 'rates-unreadable', detail: `run ${runId}: ${describe(error)}` });
        continue;
      }
      records.set(runId, run);
      // A lock governs only while its run can still move; a settled run's artifacts may be edited for the next (D-R13a-05).
      if (SETTLED.has(run.ending)) continue;
      try {
        if ((await vault.readLocks(runId)) === null) continue;
        const verdict = await vault.verifyLocks(runId);
        if (!verdict.ok) for (const t of verdict.tampered) findings.push(drift(runId, `runs/${runId}/locks: ${t.path}`, 'lock-unverified', t.expected, t.actual));
      } catch (error) {
        refusals.push({ part: 'locks', cause: 'locks-unreadable', detail: `run ${runId}: ${describe(error)}` });
      }
    }

    // Rates: the last snapshot's runs, derived again from the records as they now stand (D-A-NR-03).
    let rates: RateSnapshot | null = null;
    if (ratesRead) {
      const priorRates = priorCoverage?.rates ?? null;
      if (priorRates !== null) {
        const again: RunRecords[] = [];
        for (const runId of priorRates.runs) {
          const run = records.get(runId);
          if (run === undefined) findings.push({ kind: 'rate-drift', rate: `runs.${runId}`, snapshot: '"settled"', observed: 'absent' });
          else {
            if (!SETTLED.has(run.ending)) findings.push({ kind: 'rate-drift', rate: `runs.${runId}`, snapshot: '"settled"', observed: JSON.stringify(run.ending) });
            again.push(run);
          }
        }
        const figures = flattenFigures(deriveOutcomes(again));
        for (const rate of new Set([...Object.keys(priorRates.figures), ...Object.keys(figures)])) {
          const before = priorRates.figures[rate] ?? 'absent';
          const now = figures[rate] ?? 'absent';
          if (before !== now) findings.push({ kind: 'rate-drift', rate, snapshot: before, observed: now });
        }
      }
      const settled = [...records.values()].filter((r) => SETTLED.has(r.ending));
      rates = { runs: settled.map((r) => r.runId).sort(), figures: flattenFigures(deriveOutcomes(settled)) };
    }

    // Drift that names a run is also in that run's P14 record (A-R13a-04).
    for (const finding of findings) {
      if (finding.kind !== 'record-drift' || finding.runId === null) continue;
      await vault.recordDecision({
        runId: finding.runId, taskId: null, station: null, decidedAt: new Date(this.#now()).toISOString(), occurrence: randomUUID(),
        decision: { cause: 'record-drift', decidedBy: 'nightly', record: finding.record, problem: finding.problem, expected: finding.expected, observed: finding.observed },
        collectedBy: 'runtime',
      });
    }

    const integrity: IntegrityCoverage = {
      objects, hashes, comparedWith: prior === undefined ? 'none' : { record: prior.sha256 }, rates,
    };
    return { findings, refusals, integrity, security: null };
  }

  async #security(policy: Policy, schedule: TriggerSchedule, date: string, intake: Intake): Promise<Part> {
    const { target } = this.#options;
    const findings: NightlyFinding[] = [];
    const refusals: NightlyRefusal[] = [];
    const nightly = policy.triggers.nightly;
    const egress: Array<SecurityCoverage['egress'][number]> = [];
    const coverage = (commit: { branch: string; commit: string } | null, ran: SecurityCoverage['ran'], audited: SecurityCoverage['audited']): SecurityCoverage => ({ target: commit, ran, audited, egress });
    if (nightly === undefined) {
      refusals.push({ part: 'policy', cause: 'no-policy', detail: 'the policy names no nightly block, so no scanner or advisory host is granted' });
      return { findings, refusals, integrity: null, security: coverage(null, [], null) };
    }
    if (target === null) {
      refusals.push({ part: 'target', cause: 'no-target', detail: 'this host names no default branch to scan' });
      return { findings, refusals, integrity: null, security: coverage(null, [], null) };
    }
    const dir = join(this.#options.workspaces.root, 'nightly', `${schedule.id}-${date}-${randomUUID()}`);
    const tree = join(dir, 'tree');
    const ran: Array<SecurityCoverage['ran'][number]> = [];
    let commit: string;
    let files: readonly string[];
    try {
      commit = await resolveCommit(target.checkout, `refs/heads/${target.branch}`);
      await mkdir(tree, { recursive: true });
      files = (await materialize(target.checkout, commit, tree)).files;
    } catch (error) {
      await rm(dir, { recursive: true, force: true });
      refusals.push({ part: 'target', cause: 'target-unreadable', detail: describe(error) });
      return { findings, refusals, integrity: null, security: coverage(null, [], null) };
    }
    const where = { branch: target.branch, commit };
    let audited: SecurityCoverage['audited'] = null;
    try {
      for (const [index, selected] of nightly.scanners.entries()) {
        const outcome = await this.#scan(tree, join(dir, `scan-${String(index)}`), selected);
        if (outcome.egress !== null) egress.push({ part: selected.kind, log: outcome.egress });
        if (!outcome.ok) {
          refusals.push({ part: selected.kind, cause: 'scan-failed', detail: outcome.reason });
          continue;
        }
        ran.push(selected.kind);
        for (const f of outcome.findings) {
          findings.push(await this.#finding(intake, policy, schedule, date, selected.kind, f.rule, f.location.path, f.location.line, f.fingerprint));
        }
      }
      const audit = await this.#audit(tree, files, join(dir, 'audit'), nightly.advisoryHost);
      if (audit.egress !== null) egress.push({ part: 'dependency-audit', log: audit.egress });
      if (!audit.ok) refusals.push({ part: 'dependency-audit', cause: audit.cause, detail: audit.detail });
      else {
        ran.push('dependency-audit');
        audited = { lockfile: audit.auditor.lockfile, stack: audit.auditor.stack };
        for (const f of audit.findings) {
          findings.push(await this.#finding(intake, policy, schedule, date, 'dependency-audit', f.rule, audit.auditor.lockfile, f.line, f.fingerprint));
        }
      }
    } finally {
      await rm(dir, { recursive: true, force: true });
    }
    return { findings, refusals, integrity: null, security: coverage(where, ran, audited) };
  }

  /** One R17 scanner over the tree, as `verify` runs it: a copy, `deny-all`, the pinned image, the tree the only data mount. */
  async #scan(
    tree: string, scratch: string, selected: NightlyScanner,
  ): Promise<({ ok: true; findings: readonly Finding[] } | { ok: false; reason: string }) & { egress: EgressLog | null }> {
    const scanner = scannerFor(selected.kind, this.#options.scanners);
    const check: SecurityCheckSpec = { id: `nightly-${selected.kind}`, kind: selected.kind, required: true, timeoutMs: NIGHTLY_STEP_TIMEOUT_MS, image: scanner.image, config: selected.config };
    const scanned = await scanTree({
      sandbox: this.#options.sandbox, tree, scratch: { dir: scratch, tree: join(scratch, 'tree'), config: join(scratch, 'config') },
      check, scanner, limits: this.#options.limits, user: this.#options.workspaces.user,
    });
    if (!scanned.ok) return { ok: false, reason: scanned.reason, egress: scanned.at === 'exec' ? scanned.egress : null };
    if (scanned.result.findings === null) return { ok: false, reason: `the scanner exited ${String(scanned.result.exitCode)} without a whole report`, egress: scanned.egress };

      mounts: { workspace: { source: scratch, target: '/workspace', mode: 'ro' }, others: [] },
      egress: { mode: 'allowlist', allow: [advisoryHost] },
      limits: this.#options.limits === null ? { cpus: 0, memoryMb: 0, pids: 0, wallClockMs: NIGHTLY_STEP_TIMEOUT_MS } : { ...this.#options.limits, wallClockMs: NIGHTLY_STEP_TIMEOUT_MS },
      user: { ...this.#options.workspaces.user },
    };
    let handle: SandboxHandle;
    try {
      handle = await this.#options.sandbox.provision(spec);
    } catch (error) {
      return { ok: false, cause: 'audit-failed', detail: `its sandbox could not be provisioned: ${describe(error)}`, egress: null };
    }
    let outcome: { ok: true; auditor: DependencyAuditor; findings: readonly AuditFinding[] } | { ok: false; cause: NightlyRefusal['cause']; detail: string };

    } finally {

    }

  }

  async #finding(
    intake: Intake, policy: Policy, schedule: TriggerSchedule, date: string,
    scanner: SecurityCoverage['ran'][number], rule: string, path: string, line: number | null, fingerprint: string,
  ): Promise<NightlyFinding> {

  }

interface Provisioned {
  readonly spec: SandboxSpec;
  /** The workspace mount's entries, read when it was provisioned. */
  readonly listed: readonly string[];
}

/** Records every sandbox asked for; answers the scanner with an empty report and the audit with `audit`. */
class ScriptedProvider implements SandboxProvider {
  readonly id = 'scripted';
  readonly provisioned: Provisioned[] = [];
  readonly execs: string[][] = [];
  readonly #images = new Map<string, string>();
  constructor(private readonly audit: { stdout: string; exitCode: number }, private readonly auditImage: string) {}
  async provision(spec: SandboxSpec): Promise<SandboxHandle> {
    this.provisioned.push({ spec, listed: (await readdir(spec.mounts.workspace.source)).sort() });
    const handle = `h${String(this.provisioned.length)}`;
    this.#images.set(handle, spec.image);
    return handle as SandboxHandle;
  }
  exec(h: SandboxHandle, cmd: string[]): Promise<ExecResult> {
    this.execs.push([...cmd]);
    const audit = this.#images.get(h) === this.auditImage;
    return Promise.resolve(audit ? { ...this.audit, stderr: '', durationMs: 1 } : { exitCode: 0, stdout: '[]', stderr: '', durationMs: 1 });
  }
  destroy(): Promise<Teardown> {
    return Promise.resolve({ meter: { kind: 'unmetered' }, egress: { kind: 'none' } });
  }
  capabilities() {
    return { computerUse: false, gpu: false, os: 'linux' as const, persistent: false, remote: false };
  }
}

/** A git checkout whose `main` holds a manifest and an npm lockfile. */
async function checkout(dir: string): Promise<void> {
  await mkdir(dir, { recursive: true });
  const git = (...args: string[]) => execFileSync('git', ['-C', dir, '-c', 'user.email=r13a@example.invalid', '-c', 'user.name=r13a', '-c', 'core.autocrlf=false', ...args], { stdio: 'pipe' });
  git('init', '-q', '-b', 'main');
  await writeFile(join(dir, 'package.json'), '{ "name": "target", "dependencies": { "minimist": "1.2.5" } }\n');
  await writeFile(join(dir, 'package-lock.json'), LOCKFILE);
  await writeFile(join(dir, 'index.js'), 'module.exports = 1;\n');
  git('add', '.');
  git('commit', '-q', '-m', 'target');
}

interface Rig {
  readonly vault: Vault;
  readonly store: string;
  readonly jobs: NightlyJobs;
  readonly intake: Intake;
  readonly sandbox: ScriptedProvider;
  readonly policy: Policy;
}

async function withNightly<T>(prefix: string, audit: { stdout: string; exitCode: number }, body: (rig: Rig) => Promise<T>): Promise<T> {
  const { Intake: IntakeClass, NightlyJobs: Jobs, SCANNERS } = await api();
  const { DEPENDENCY_AUDIT_IMAGE } = await adapters();
  const { LocalVault } = await import('@olympus-ai/vault');
  return withVaultDirs(prefix, async (dirs, base) => {
    const vault = new LocalVault(dirs);
    const target = join(base, 'target');
    await checkout(target);
    const sandbox = new ScriptedProvider(audit, DEPENDENCY_AUDIT_IMAGE);
    const jobs = new Jobs({
      vault, sandbox, workspaces: { root: join(base, 'trees'), user: { uid: 1000, gid: 1000 } }, limits: null, scanners: SCANNERS,
      target: { checkout: target, branch: 'main' },
    });
    const intake = new IntakeClass({ vault, secrets: () => undefined, jobs });
    return body({ vault, store: dirs.store, jobs, intake, sandbox, policy: await nightlyPolicy() });
  });
}

/** A run refused at admission: one decision, so it is settled and its rates are fixed (D-R13a-03). */
async function refusedRun(vault: Vault): Promise<void> {
  await vault.recordDecision({
    runId: RUN, taskId: null, station: null, decidedAt: '2026-11-01T00:00:00.000Z', occurrence: 'r13a-seed',
    decision: { cause: 'admission-refused', decidedBy: 'admission', refusal: { reason: 'invalid-request', problems: [] } },
    collectedBy: 'runtime',
  });
}

async function decisionFile(store: string): Promise<string> {
  const dir = join(store, 'runs', RUN, 'objects', 'decision');
  const [name] = await readdir(dir);
  if (name === undefined) throw new Error('R13a: the seeded decision was not stored');
  return join(dir, name);
}

/** The record a firing wrote, by its date. */
async function recordOn(vault: Vault, schedule: string, date: string): Promise<NightlyRecord> {
  const entry = (await vault.readNightly()).find((e) => e.record.schedule === schedule && e.record.date === date);
  if (entry === undefined) throw new Error(`R13a: no nightly record for ${schedule} on ${date}`);
  return entry.record;
}

   /**
    * Compare-and-swap, arbitrated by the filesystem. The next version's file is
    * created with `wx`, so of any number of writers holding the same
    * `ifVersion` exactly one creates it and the rest are refused with `EEXIST`.
    * There is deliberately no read-then-compare here: that shape loses an
    * update whenever two writers interleave between the read and the write, and
    * it passes a sequential test while doing so.

    * the choice to this unit (D-S1, S1 spec section 2). The precondition below
    * is an extra refusal, not the concurrency control: it rejects a version
    * that never existed, so a caller cannot jump the run forward and leave a
    * hole in its history.
    */
   async commitRunState(s: RunState, ifVersion: string): Promise<RunState> {
     if (!VERSION.test(ifVersion)) {
       throw new Error(`LocalVault: '${ifVersion}' is not a run state version`);
     }
     const states = join(runDir(this.#store, s.runId), 'state');
     const current = Number(ifVersion);
     if (!Number.isSafeInteger(current)) throw new Error(`LocalVault: run state version ${ifVersion} is out of range`);
 
     const held = await highestIn(states);
     if (current === 0 ? held !== null : held === null || held < current) {
       const at = held === null ? 'no committed state' : `version ${String(held)}`;
       throw new Error(`LocalVault: run ${s.runId} has ${at}, so version ${ifVersion} was never current; nothing stored`);
     }
 
     // The history it replaces, read before anything is written: a commit may extend it and never shorten or rewrite it (codex-5).
     const previous = current === 0 ? [] : requireRunState(await readJson(join(states, `${ifVersion}.json`)), `${ifVersion}.json`).stations;
     const problem = stationsProblem(previous, (s as { stations?: unknown }).stations);
     if (problem !== undefined) throw new Error(`LocalVault: refusing run ${s.runId}'s state: ${problem}; nothing stored`);
 
     const next = String(current + 1);
     const stored: RunState = { ...s, version: next };
     if (!(await createExclusive(join(states, `${next}.json`), encode(stored)))) {
       throw new Error(
         `LocalVault: run ${s.runId} has already moved past version ${ifVersion}; another writer committed ${next} first. ` +
           'Nothing was stored. Re-read the state and commit again.',

     }
